  private static final class C5CanaryStream {
    private static final int MAX_VALUES = 16, MAX_VALUE_CHARS = 4096, VARIANTS = 15;
    private static final int MAX_PATTERN_BYTES = MAX_VALUE_CHARS * 12;
    private static final int MAX_TOTAL_PATTERN_BYTES = MAX_VALUES * VARIANTS * MAX_PATTERN_BYTES;
    private final List<byte[]> patterns = new java.util.ArrayList<>();
    private int[][] prefixes = new int[0][];
    private int[] states = new int[0];
    private final int[][] firstByteIndices = new int[256][];
    private int[] activeIndices = new int[0], scratchCandidates = new int[0];
    private boolean indicesReady;
    private final boolean[] startBytes = new boolean[256];
    private int activeStates;
    private long encodedBytes, writeCalls, hits, invalidWrites;
    private int patternBytes;
    private boolean failed, inputClosed, wiped;

    C5CanaryStream(List<String> values) {
      try {
        if (values == null || values.isEmpty() || values.size() > MAX_VALUES) {
          throw new IllegalArgumentException("C5 stream value inventory bound");
        }
        for (String value : values) {
          if (value == null || value.isEmpty() || value.length() > MAX_VALUE_CHARS) {
            throw new IllegalArgumentException("C5 stream value bound");
          }
          List<byte[]> variants = canaryByteVariants(value);
          if (variants.size() != VARIANTS) {
            throw new IllegalArgumentException("C5 stream variant count");
          }
          for (byte[] variant : variants) {
            if (variant.length == 0
                || variant.length > MAX_PATTERN_BYTES
                || patternBytes > MAX_TOTAL_PATTERN_BYTES - variant.length) {
              throw new IllegalArgumentException("C5 stream pattern bound");
            }
            patterns.add(variant);
            patternBytes += variant.length;
          }
        }
        prefixes = new int[patterns.size()][];
        states = new int[patterns.size()];
        for (int index = 0; index < patterns.size(); index++) {
          byte[] pattern = patterns.get(index);
          startBytes[Byte.toUnsignedInt(pattern[0])] = true;
          int[] prefix = new int[pattern.length];
          for (int offset = 1, matched = 0; offset < pattern.length; offset++) {
            while (matched > 0 && pattern[offset] != pattern[matched]) {
              matched = prefix[matched - 1];
            }
            if (pattern[offset] == pattern[matched]) {
              matched++;
            }
            prefix[offset] = matched;
          }
          prefixes[index] = prefix;
        }
        activeIndices = new int[patterns.size()];
        scratchCandidates = new int[patterns.size()];
        int[] bucketOffsets = new int[256];
        try {
          for (byte[] pattern : patterns) {
            bucketOffsets[Byte.toUnsignedInt(pattern[0])]++;
          }
          for (int octet = 0; octet < firstByteIndices.length; octet++) {
            firstByteIndices[octet] = new int[bucketOffsets[octet]];
          }
          java.util.Arrays.fill(bucketOffsets, 0);
          for (int index = 0; index < patterns.size(); index++) {
            int octet = Byte.toUnsignedInt(patterns.get(index)[0]);
            firstByteIndices[octet][bucketOffsets[octet]++] = index;
          }
          indicesReady = true;
        } finally {
          java.util.Arrays.fill(bucketOffsets, 0);
        }
      } catch (RuntimeException invalid) {
        // Failure is sticky and closed; never export a payload-containing cause.
        failed = true;
      }
    }

    private long increment(long counter) {
      if (counter == Long.MAX_VALUE) {
        failed = true;
        return counter;
      }
      return counter + 1;
    }

    private void invalid() {
      failed = true;
      invalidWrites = increment(invalidWrites);
    }

    void write(int octet) {
      writeCalls = increment(writeCalls);
      if (inputClosed || wiped) {
        invalid();
      }
      consume((byte) octet); // Original OutputStream.write(int) consumes its low eight bits.
    }

    void write(byte[] bytes, int offset, int length) {
      writeCalls = increment(writeCalls);
      if (inputClosed || wiped) {
        invalid();
      }
      if (bytes == null || offset < 0 || length < 0 || offset > bytes.length - length) {
        invalid();
        return;
      }
      // No truncation/drop after a hit; every later valid byte is independently consumed.
      for (int index = offset; index < offset + length; index++) {
        consume(bytes[index]);
      }
    }

    private void consume(byte octet) {
      encodedBytes = increment(encodedBytes);
      if (wiped
          || !indicesReady
          || firstByteIndices.length != 256
          || activeIndices.length != patterns.size()
          || scratchCandidates.length != patterns.size()
          || states.length != patterns.size()
          || prefixes.length != patterns.size()
          || activeStates < 0
          || activeStates > patterns.size()) {
        failed = true;
        return;
      }
      if (activeStates == 0 && !startBytes[Byte.toUnsignedInt(octet)]) {
        return;
      }
      int[] starts = firstByteIndices[Byte.toUnsignedInt(octet)];
      if (starts == null || starts.length > patterns.size()) {
        failed = true;
        return;
      }
      // Merge the ordered OLD-state inventories completely before any KMP state changes.
      int candidateCount = 0, activeOffset = 0, startOffset = 0, previousIndex = -1;
      while (activeOffset < activeStates || startOffset < starts.length) {
        int index;
        boolean wasActive;
        if (startOffset == starts.length
            || activeOffset < activeStates && activeIndices[activeOffset] < starts[startOffset]) {
          index = activeIndices[activeOffset++];
          wasActive = true;
        } else if (activeOffset == activeStates
            || starts[startOffset] < activeIndices[activeOffset]) {
          index = starts[startOffset++];
          wasActive = false;
        } else {
          index = activeIndices[activeOffset++];
          startOffset++;
          wasActive = true;
        }
        if (index <= previousIndex
            || index >= states.length
            || candidateCount >= scratchCandidates.length
            || wasActive != (states[index] > 0)) {
          failed = true;
          return;
        }
        scratchCandidates[candidateCount++] = index;
        previousIndex = index;
      }
      int nextActive = 0;
      for (int candidate = 0; candidate < candidateCount; candidate++) {
        int index = scratchCandidates[candidate];
        byte[] pattern = patterns.get(index);
        int[] prefix = prefixes[index];
        int previous = states[index];
        int matched = previous;
        while (matched > 0 && pattern[matched] != octet) {
          matched = prefix[matched - 1];
        }
        if (pattern[matched] == octet) {
          matched++;
        }
        if (matched == pattern.length) {
          hits = increment(hits);
          matched = prefix[matched - 1];
        }
        if (previous == 0 && matched > 0) {
          activeStates++;
        } else if (previous > 0 && matched == 0) {
          activeStates--;
        }
        states[index] = matched;
        if (matched > 0) {
          activeIndices[nextActive++] = index;
        }
      }
      if (activeStates < 0 || activeStates > patterns.size() || nextActive != activeStates) {
        failed = true;
      }
    }

    void closeInput() {
      inputClosed = true;
    }

    void assertClean(boolean requireClosed) {
      assertThat(indicesReady || wiped || failed)
          .as("complete original-index dispatch metadata")
          .isTrue();
      assertThat(activeIndices.length).isEqualTo(states.length);
      assertThat(scratchCandidates.length).isEqualTo(states.length);
      int previousIndex = -1;
      for (int offset = 0; offset < activeStates; offset++) {
        int index = activeIndices[offset];
        assertThat(index > previousIndex && index < states.length && states[index] > 0)
            .as("ordered exact positive original-pattern index inventory")
            .isTrue();
        previousIndex = index;
      }
      assertThat(activeStates)
          .as("exact positive KMP state inventory")
          .isEqualTo((int) java.util.Arrays.stream(states).filter(state -> state > 0).count());
      assertThat(failed).as("original encoded stream observer/counter integrity").isFalse();
      assertThat(invalidWrites).as("original encoded stream invalid or late writes").isZero();
      assertThat(encodedBytes).as("actual original encoded bytes consumed").isPositive();
      assertThat(writeCalls).as("actual original encoded write calls consumed").isPositive();
      assertThat(hits)
          .as("released literal/encoded values absent from original encoded stream")
          .isZero();
      if (requireClosed) {
        assertThat(inputClosed).as("original capture target was actually closed").isTrue();
      }
    }

    void wipe() {
      for (byte[] pattern : patterns) {
        java.util.Arrays.fill(pattern, (byte) 0);
      }
      for (int[] prefix : prefixes) {
        if (prefix != null) {
          java.util.Arrays.fill(prefix, 0);
        }
      }
      java.util.Arrays.fill(states, 0);
      java.util.Arrays.fill(activeIndices, 0);
      java.util.Arrays.fill(scratchCandidates, 0);
      for (int[] bucket : firstByteIndices) {
        if (bucket != null) {
          java.util.Arrays.fill(bucket, 0);
        }
      }
      java.util.Arrays.fill(firstByteIndices, null);
      activeIndices = new int[0];
      scratchCandidates = new int[0];
      indicesReady = false;
      java.util.Arrays.fill(startBytes, false);
      activeStates = 0;
      patterns.clear();
      prefixes = new int[0][];
      states = new int[0];
      wiped = true;
    }
  }
