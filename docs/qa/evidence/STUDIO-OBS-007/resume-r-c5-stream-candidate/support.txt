    // Extend this existing independent support case, never add another Core/browser case.
    List<byte[]> byteLeaks = new java.util.ArrayList<>();
    for (String leak : encodedLeaks) byteLeaks.add(leak.getBytes(StandardCharsets.UTF_8));
    byteLeaks.add(HexFormat.of().parseHex("c3a9f09f9880225c41207a2b")); // Literal UTF8.
    byteLeaks.add(HexFormat.of().parseHex("e9003dd800de22005c00410020007a002b00")); // UTF16LE.
    byteLeaks.add(HexFormat.of().parseHex("00e9d83dde000022005c00410020007a002b")); // UTF16BE.
    byteLeaks.add("w6nwn5iAIlxBIHor".getBytes(StandardCharsets.UTF_8)); // Independent base64.
    // encodeURIComponent and form-space replacement coincide for this independent input.
    byteLeaks.add("%C3%A9%F0%9F%98%80%22%5CA%20z%2B".getBytes(StandardCharsets.UTF_8));
    assertThat(byteLeaks.size()).isEqualTo(15);
    assertThat(canaryByteVariants(value).size()).isEqualTo(15);
    for (byte[] leak : byteLeaks) {
      for (int split = 0; split <= leak.length; split++) {
        C5CanaryStream scanner = new C5CanaryStream(List.of(value));
        try {
          byte[] padded = new byte[leak.length + 6];
          java.util.Arrays.fill(padded, (byte) '~');
          System.arraycopy(leak, 0, padded, 3, leak.length);
          scanner.write(padded, 3, split);
          scanner.write(padded, 3 + split, leak.length - split);
          scanner.closeInput();
          assertThat(scanner.encodedBytes).as("every original split/offset byte consumed").isEqualTo(leak.length);
          assertThat(scanner.writeCalls).isEqualTo(2);
          assertThat(scanner.hits).as("independent canary detected across every boundary").isPositive();
          assertThatThrownBy(() -> scanner.assertClean(true)).isInstanceOf(AssertionError.class);
        } finally { scanner.closeInput(); scanner.wipe(); }
      }
      C5CanaryStream integerScanner = new C5CanaryStream(List.of(value));
      try {
        for (byte octet : leak) integerScanner.write(Byte.toUnsignedInt(octet) | 0x100);
        integerScanner.closeInput();
        assertThat(integerScanner.encodedBytes).isEqualTo(leak.length);
        assertThat(integerScanner.writeCalls).isEqualTo(leak.length);
        assertThat(integerScanner.hits).as("int writes retain low-eight-bit matching").isPositive();
        assertThatThrownBy(() -> integerScanner.assertClean(true)).isInstanceOf(AssertionError.class);
      } finally { integerScanner.closeInput(); integerScanner.wipe(); }
    }
    C5CanaryStream unrelated = new C5CanaryStream(List.of(value));
    try {
      byte[] bytes = "%C3%A8%F0%9F%98%80%22%5CA%20z%2B".getBytes(StandardCharsets.UTF_8);
      for (int offset = 0; offset < bytes.length; offset += 3)
        unrelated.write(bytes, offset, Math.min(3, bytes.length - offset));
      unrelated.closeInput();
      unrelated.assertClean(true);
    } finally { unrelated.closeInput(); unrelated.wipe(); }
    C5CanaryStream volume = new C5CanaryStream(List.of(value));
    try {
      byte[] chunk = new byte[65536];java.util.Arrays.fill(chunk, (byte) '~');
      for (int index = 0; index < 2049; index++) volume.write(chunk, 0, chunk.length);
      volume.closeInput();
      assertThat(volume.encodedBytes).as("actual bytes exceed historical aggregate cap")
          .isEqualTo(2049L * 65536).isGreaterThan(128L * 1024 * 1024);
      assertThat(volume.writeCalls).isEqualTo(2049);
      volume.assertClean(true);
    } finally { volume.closeInput(); volume.wipe(); }
    C5CanaryStream invalid = new C5CanaryStream(List.of(value));
    try {
      invalid.write(new byte[] {'~'}, 1, 1);
      invalid.write('~');
      invalid.closeInput();invalid.write('~');
      assertThat(invalid.encodedBytes).isEqualTo(2);
      assertThat(invalid.invalidWrites).isEqualTo(2);
      assertThatThrownBy(() -> invalid.assertClean(true)).isInstanceOf(AssertionError.class);
    } finally { invalid.closeInput(); invalid.wipe(); }
    C5CanaryStream notClosed = new C5CanaryStream(List.of(value));
    try {
      notClosed.write('~');
      assertThatThrownBy(() -> notClosed.assertClean(true)).isInstanceOf(AssertionError.class);
      notClosed.closeInput();notClosed.assertClean(true);
    } finally { notClosed.closeInput(); notClosed.wipe(); }
    C5CanaryStream overflow = new C5CanaryStream(List.of(value));
    try {
      overflow.encodedBytes = Long.MAX_VALUE;
      overflow.write('~');overflow.closeInput();
      assertThat(overflow.encodedBytes).isEqualTo(Long.MAX_VALUE);
      assertThat(overflow.failed).isTrue();
      assertThatThrownBy(() -> overflow.assertClean(true)).isInstanceOf(AssertionError.class);
    } finally { overflow.closeInput(); overflow.wipe(); }
