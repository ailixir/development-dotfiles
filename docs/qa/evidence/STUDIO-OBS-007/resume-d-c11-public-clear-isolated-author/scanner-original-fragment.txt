        const c5Variants=value=>{
          const utf8=Buffer.from(value,'utf8'),le=Buffer.from(value,'utf16le'),be=Buffer.from(le);be.swap16();
          const percent=[...utf8].map(byte=>String.fromCharCode(37)+byte.toString(16).padStart(2,'0').toUpperCase()).join('');
          const unicode=[...Array(value.length)].map((_,index)=>String.fromCharCode(92)+'u'+value.charCodeAt(index).toString(16).padStart(4,'0')).join('');
          const json=JSON.stringify(value),form=new URLSearchParams({v:value}).toString().slice(2);
          return [utf8,le,be,...[utf8.toString('base64'),utf8.toString('hex'),utf8.toString('hex').toUpperCase(),
            percent,percent.toLowerCase(),encodeURIComponent(value),form,form.replaceAll('+',String.fromCharCode(37)+'20'),
            unicode,unicode.toUpperCase().replaceAll('U','u'),json,json.slice(1,-1)].map(text=>Buffer.from(text,'utf8'))];
        };
        const c5ProfileHitCategory=relative=>{
          const parts=relative.split('/'),name=parts.at(-1);
          if(parts.some(part=>['Cache','Cache_Data'].includes(part)))return 0;
          if(parts.some(part=>['Code Cache','GPUCache','DawnCache','ShaderCache','GrShaderCache'].includes(part)))return 1;
          if(parts.some(part=>['Local Storage','Session Storage','IndexedDB','WebStorage','CacheStorage'].includes(part)))return 2;
          if(parts.includes('Sessions'))return 3;
          if(['History','History-journal'].includes(name))return 4;
          if(['Preferences','Secure Preferences','Local State'].includes(name))return 5;
          return 6;
        };
        const c5ProfileScan=directory=>{
          let files=0,bytes=0,hits=0,links=0,failed=false,stage=0,errno=0;
          const hitCategories=Array(8).fill(0);
          let linkName=0,linkRoot=false,linkType=0,linkBytes=0,linkTargetRead=false,linkTargetIsVersion=false;
          try{
            const root=p4Fs.realpathSync(directory),stat=p4Fs.lstatSync(directory);
            const canonical=p4Fs.lstatSync(root);
            if(stat.isSymbolicLink()||!stat.isDirectory()||stat.dev!==canonical.dev||stat.ino!==canonical.ino
              ||stat.uid!==process.getuid())throw new Error('ROOT');
            const valuesStat=p4Fs.lstatSync(c5ValuesFile);
            if(valuesStat.isSymbolicLink()||!valuesStat.isFile()||(valuesStat.mode&0o777)!==0o600||valuesStat.size>32768)throw new Error('VALUES');
            const values=JSON.parse(p4Fs.readFileSync(c5ValuesFile,'utf8'));
            if(!Array.isArray(values)||values.length<1||values.length>16||values.some(value=>typeof value!=='string'||value.length>4096))throw new Error('VALUES');
            const variants=values.flatMap(c5Variants),walk=path=>{
              for(const entry of p4Fs.readdirSync(path,{withFileTypes:true})){
                const child=path+'/'+entry.name,info=p4Fs.lstatSync(child);
                if(info.isSymbolicLink()){
                  if(path!==root||!['SingletonLock','SingletonCookie','SingletonSocket','RunningChromeVersion'].includes(entry.name)
                    ||info.size>1024){
                    linkName=['SingletonLock','SingletonCookie','SingletonSocket','RunningChromeVersion'].indexOf(entry.name)+1;
                    linkRoot=path===root;linkType=1;linkBytes=Number.isFinite(info.size)?info.size:0;
                    if(linkRoot&&linkName===4&&info.size<=1024){try{const target=p4Fs.readlinkSync(child);
                      linkTargetRead=true;
                      linkTargetIsVersion=target.length<=128&&/^[0-9]{1,4}(?:\\.[0-9]{1,6}){3}$/.test(target);
                    }catch{/* Metadata loss cannot change the original SYMLINK rejection. */}}
                    throw new Error('SYMLINK');
                  }
                  const text=Buffer.from(entry.name+' '+p4Fs.readlinkSync(child),'utf8');
                  if(text.length>2048)throw new Error('LINKBOUND');
                  links++;bytes+=text.length;
                  for(const variant of variants)if(text.includes(variant)){hits++;hitCategories[7]++;}
                  continue;
                }
                if(info.isDirectory()){walk(child);continue;}
                if(!info.isFile()||info.size>67108864||bytes+info.size>268435456)throw new Error('BOUND');
                const actual=p4Fs.realpathSync(child);if(!actual.startsWith(root+'/'))throw new Error('PATH');
                const fd=p4Fs.openSync(child,p4Fs.constants.O_RDONLY|p4Fs.constants.O_NOFOLLOW);
                try{
                  const opened=p4Fs.fstatSync(fd);
                  if(!opened.isFile()||opened.dev!==info.dev||opened.ino!==info.ino
                    ||opened.size>67108864||bytes+opened.size>268435456)throw new Error('FD');
                  const buffer=Buffer.alloc(opened.size+1);let length=0;
                  while(length<buffer.length){const read=p4Fs.readSync(fd,buffer,length,buffer.length-length,null);
                    if(read===0)break;length+=read;}
                  const after=p4Fs.fstatSync(fd);
                  if(length>opened.size||after.size!==length||after.dev!==opened.dev||after.ino!==opened.ino)
                    throw new Error('CHANGED');
                  const body=buffer.subarray(0,length);files++;bytes+=body.length;
                  for(const variant of variants)if(body.includes(variant)){hits++;hitCategories[c5ProfileHitCategory(child.slice(root.length+1))]++;}
                }finally{p4Fs.closeSync(fd);}
              }
            };walk(root);
          }catch(error){
            failed=true;
            stage=['ROOT','VALUES','SYMLINK','LINKBOUND','BOUND','PATH','FD','CHANGED'].indexOf(error?.message)+1;
            errno=['EACCES','EPERM','ENOENT','ELOOP','ENOTDIR','EISDIR','EIO','EBADF','EMFILE','ENFILE'].indexOf(error?.code)+1;
            if(stage===0){const operation=['open','read','scandir','lstat','realpath','fstat','close','readlink'].indexOf(error?.syscall);
              if(operation>=0)stage=9+operation;}
          }
          try{const fd=p4Fs.openSync(c5CountsFile,p4Fs.constants.O_WRONLY|p4Fs.constants.O_CREAT|
            p4Fs.constants.O_APPEND|p4Fs.constants.O_NOFOLLOW,0o600);
            try{p4Fs.writeSync(fd,JSON.stringify({files,bytes,hits,links,failed,stage,errno,linkName,linkRoot,linkType,linkBytes,linkTargetRead,linkTargetIsVersion,hitCategories})+'\\n');}finally{p4Fs.closeSync(fd);}}
          catch{try{const fd=p4Fs.openSync(c5CountsFile+'-failed',p4Fs.constants.O_WRONLY|p4Fs.constants.O_CREAT|
            p4Fs.constants.O_APPEND|p4Fs.constants.O_NOFOLLOW,0o600);try{p4Fs.writeSync(fd,'1');}finally{p4Fs.closeSync(fd);}}catch{}}
        };
