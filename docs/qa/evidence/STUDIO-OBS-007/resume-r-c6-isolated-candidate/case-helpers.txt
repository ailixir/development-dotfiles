  // C6 adds four concrete cuts to the existing fixture. All observers forward original methods;
  // receipts below establish observation/ownership only, never execution or usage authority.
  private void p8C6Case(int kind) throws Exception {
    String value = "C6-PRIVATE-" + UUID.randomUUID() + "-+/界";
    C5Capture capture = new C5Capture(List.of(value));
    c5Capture = capture;
    Throwable primary = null;
    try {
      capture.writeProfileValues();
      p3Case(() -> p8C6Run(kind, value, capture));
    } catch (Exception | Error failure) {
      primary = failure;
      throw failure;
    } finally {
      var failures = new java.util.ArrayList<Throwable>();
      P4RevocationState owned = p4Revocation;
      if (owned != null && owned.c6Kind == 3) {
        // p3Case already attempted original owned-context cleanup. Residual actual writers
        // remain pinned by start identity; attempt every release without masking the primary.
        for (P6OwnedPid member : owned.c6BrowserTree) {
          p3AttemptTeardown(failures, () -> {
            if (member.handle().isAlive()) {
              p6SameProcess(member.handle(), member.started());
              if (!member.handle().destroyForcibly())
                throw new IllegalStateException("C6 residual owned writer kill refused");
            }
          });
        }
        p3AttemptTeardown(failures, () -> p6AwaitProcesses(owned.c6BrowserTree, true));
      }
      if (!cleanupUnconfirmed) {
        // Retain capture/hooks until actual owner closure and final producer acknowledgements.
        p3AttemptTeardown(failures, () -> p3Scan(List.of(value)));
        p3AttemptTeardown(failures, capture::writeNumericEvidence);
        p3AttemptTeardown(failures, capture::close);
        p3AttemptTeardown(failures, () -> Files.writeString(
            TEMP.resolve("p4-revocation-state.json"), "{\"active\":false}", StandardCharsets.UTF_8));
        c5Capture = null;
        p4Revocation = null;
      } else {
        failures.add(new IllegalStateException("C6 original owner cleanup remains unconfirmed"));
      }
      if (!failures.isEmpty()) {
        if (primary != null) failures.forEach(primary::addSuppressed);
        else throwTeardownFailures(failures);
      }
    }
  }

  private void p8C6Run(int kind, String value, C5Capture capture) throws Exception {
    assertThat(kind).isBetween(1, 4);
    p3Mode = P3Mode.LOGIN;
    p3Grant("CSI-456", "fixture-other");
    UUID test = p3CreateSource("CSI-456", "fixture-other", true);
    // Legitimate test configuration precedes learning/publication/admission, never alters a job.
    assertThat(jdbc.update("UPDATE ui_tests SET timeout_seconds=120 WHERE id=?", test)).isEqualTo(1);
    VaultScope scope = VaultScope.of("CSI-456", "fixture-other", test);
    p3ExpectedPassword = value;
    p3Seed(scope, value);
    P3Publication publication = null;
    if (kind <= 2) {
      publication = p4PositiveNativeControl(test);
    }
    int reads = COUNTED_VAULT.readCount(StudioVaultProvider.ReferenceKind.SECRET, scope, "PASS");
    int models = MODEL_WIRE.size();
    P4RevocationState state = new P4RevocationState(
        P4_REVOCATION_CASE.incrementAndGet(), kind <= 2 ? 2 : 1, test, jdbc);
    state.c6Kind = kind;
    state.c6Phase = p4StartObserverPhase(1);
    p4Revocation = state;
    p4WriteRevocationMode(state);
    if (kind >= 3) {
      state.guidedSession = p3Start(test, "fixture-other");
      capture.selectedGuidedSession = state.guidedSession;
    } else {
      state.execution = p3Execute(test);
      p4WriteRevocationMode(state);
      p3AssertBinding(state.execution, scope, publication);
      assertThat(dispatcher.dispatchOnce()).isEqualTo(1);
    }
    List<P6OwnedPid> browserTree = List.of();
    if (kind < 4) {
      assertThat(state.c6Pending.await(15, java.util.concurrent.TimeUnit.SECONDS))
          .as("real password focus ingress plus contemporaneous original Node pending acknowledgement")
          .isTrue();
      synchronized (state) {
        assertThat(state.failed).isFalse();
        assertThat(state.c6FocusAt).isPositive();
        assertThat(state.c6PendingAt).isGreaterThanOrEqualTo(state.c6FocusAt);
        assertThat(state.c6FillEntries).isEqualTo(1);
        assertThat(state.active).containsOnlyKeys(state.inFlightOrdinal);
        assertThat(state.active.get(state.inFlightOrdinal)).isEqualTo(1);
        if (kind == 1) {
          assertThat(state.c6DeadlineMarginMs)
              .as("actual admitted overall deadline remains strictly later than actual step deadline")
              .isPositive();
        }
        if (kind == 1) state.c6CutAt = System.nanoTime();
      }
      P6OwnedPid primaryBrowser = kind == 3 ? p8OwnedPrimaryBrowser(state) : null;
      if (kind == 3) {
        assertThat(primaryBrowser).isNotNull();
        browserTree = java.util.stream.Stream.concat(java.util.stream.Stream.of(primaryBrowser),
            primaryBrowser.handle().descendants().map(GuidedPublicationReplayIT::p6SnapshotPid)).toList();
        assertThat(browserTree.size()).isLessThanOrEqualTo(128);
        state.c6BrowserTree = browserTree;
      }
      if (kind == 2 || kind == 3) {
        // Reconfirm genuine original pending after ownership discovery and before the real cut.
        p8WritePendingProbe(state, 2);
        assertThat(state.c6CutPending.await(2, java.util.concurrent.TimeUnit.SECONDS))
            .as("contemporaneous original Node pending acknowledgement at actual fault cut").isTrue();
        synchronized (state) {
          assertThat(state.active).containsOnlyKeys(state.inFlightOrdinal);
          assertThat(state.active.get(state.inFlightOrdinal)).isEqualTo(1);
          state.c6CutAt = System.nanoTime();
          assertThat(state.c6CutAt-state.c6PendingAt).isLessThanOrEqualTo(Duration.ofMillis(250).toNanos());
        }
      }
      if (kind == 2) {
        p3Call(p3Cookie, "/api/v1/studio/executions/" + state.execution + "/cancel",
            C.object(), 202);
      } else if (kind == 3) {
        assertThat(primaryBrowser).isNotNull();
        // Inventory was captured before the final real pending ACK, not after its cut clock.
        p6SameProcess(primaryBrowser.handle(), primaryBrowser.started());
        assertThat(primaryBrowser.handle().destroyForcibly()).isTrue();
        assertThat(System.nanoTime()-state.c6PendingAt)
            .as("actual verified-primary signal stayed contemporaneous with original pending ACK")
            .isLessThanOrEqualTo(Duration.ofMillis(250).toNanos());
        assertThat(guidedWorker.isAlive()).as("genuine browser loss leaves owning worker alive").isTrue();
      }
      // kind1 has no synthetic timeout: the unchanged original step timer supplies the cut.
    }
    if (kind <= 2) {
      JsonNode result = p3AwaitExecution(state.execution, "ERROR");
      JsonNode report = p3RunnerReport(state.execution);
      assertThat(report.path("outcome").path("errorCode").asText())
          .isEqualTo(kind == 1 ? "OUTCOME_UNKNOWN" : "CANCELLED");
      if (kind == 2) assertThat(result.path("errorCode").asText()).isEqualTo("CANCELLED");
      assertThat(report.path("steps").get(1).path("failure").asText())
          .isEqualTo(kind == 1 ? "OUTCOME_UNKNOWN" : "CANCELLED");
      assertThat(report.path("steps").get(2).path("status").asText()).isEqualTo("SKIPPED");
      assertThat(report.path("steps").get(2).path("startedAtMs").isNull()).isTrue();
      assertThat(report.path("secretUse").has("PASS")).as("unknown effects stay sparse").isFalse();
      assertThat(MODEL_WIRE.size()).isEqualTo(models);
      p3AssertBinding(state.execution, scope, publication);
    } else {
      awaitState(state.guidedSession, "UNKNOWN_OUTCOME");
      List<Map<String, Object>> consumed = jdbc.queryForList(
          "SELECT document::text AS ledger FROM studio_guided_actions WHERE session_id=? AND step_id=?"
              + " AND proposal->'operation'->'value'->>'kind'='secretRef'", state.guidedSession, TWO);
      assertThat(consumed).hasSize(1);
      JsonNode ledger = C.parse(String.valueOf(consumed.getFirst().get("ledger")));
      assertThat(ledger.path("state").asText()).isEqualTo("UNKNOWN_OUTCOME");
      assertThat(ledger.path("outcomeCode").asText()).isEqualTo("ACK_LOST");
      assertThat(ledger.path("dispatchCount").asInt()).isEqualTo(1);
      assertThat(jdbc.queryForObject(
          "SELECT count(*) FROM studio_guided_actions WHERE session_id=? AND step_id=?",
          Long.class, state.guidedSession, THREE)).isZero();
      p3AwaitTeardown(state.guidedSession);
    }
    assertThat(COUNTED_VAULT.readCount(StudioVaultProvider.ReferenceKind.SECRET, scope, "PASS") - reads)
        .isEqualTo(1);
    String ownerColumn = kind <= 2 ? "execution_id" : "session_id";
    Object owner = kind <= 2 ? state.execution : state.guidedSession;
    assertThat(jdbc.queryForObject("SELECT count(*) FROM studio_secret_use_audit WHERE " + ownerColumn
        + "=? AND row_kind='RELEASE' AND outcome='RESOLVED' AND secret_id='PASS'", Long.class, owner))
        .isEqualTo(1);
    // Generic wire ACK_LOST is not evidence of acknowledgement-loss usage, USED or REFUSED.
    assertThat(jdbc.queryForObject("SELECT count(*) FROM studio_secret_use_audit WHERE " + ownerColumn
        + "=? AND row_kind='USAGE' AND usage_outcome IN ('USED','REFUSED','NOT_USED','ACK_LOST')"
        + " AND secret_id='PASS'", Long.class, owner)).isZero();
    p3AssertAuditScope(state.guidedSession, state.execution, scope);
    p3CleanupOwnedResources();
    if (!browserTree.isEmpty()) p6AwaitProcesses(browserTree, true);
    p8AwaitC6Closed(state);
    if (kind == 2) {
      assertThat(state.cutoffAt-state.c6CutAt)
          .as("real public cancel to valid original child Stop includes network and queues")
          .isPositive().isLessThanOrEqualTo(Duration.ofSeconds(2).toNanos());
    }
    p4SealPrivateProducersAfterOwnerCleanup(state);
    assertThat(state.active).isEmpty();
    assertThat(state.c6FillEntries).isEqualTo(kind == 4 ? 0 : 1);
    assertThat(P3_RECEIPTS).isEmpty();
    assertThat(P3_REPORT_REQUESTS.get()).isZero();
    if (kind == 4) {
      p4AssertRegisteredAndNoPasswordEffects();
      assertThat(state.c6RawCalls).isEqualTo(1);
      assertThat(state.c6LiteralHits).isEqualTo(1);
      assertThat(state.c6EncodedHits).isEqualTo(2);
      assertThat(state.c6CallbackRedacted).isEqualTo(1);
      p8AssertInjectedProducedSinks(state, value);
    } else {
      // No absent page fence is promoted into a zero-write assertion.
      assertThat(state.c6NativeClose).isTrue();
      if (!state.c6FenceAvailable) assertThat(P8_C6_CLOSED_UNKNOWN).contains(state.c6Phase);
      if (kind == 3) {
        assertThat(state.c6RawCalls).isEqualTo(1);
        assertThat(state.c6LiteralHits).isZero();
        assertThat(state.c6EncodedHits).isZero();
      }
    }
    p3Scan(List.of(value));
    p3WriteEvidence("c6-" + kind, C.object().put("kind", kind)
        .put("realFocusIngress", state.c6FocusAt > 0).put("originalPendingAck", state.c6PendingAt > 0)
        .put("sameJavaClockFocusToAckNanos", state.c6PendingAt == 0 ? 0 : state.c6PendingAt-state.c6FocusAt)
        .put("actualStepOverallMarginMs", state.c6DeadlineMarginMs)
        .put("originalNativeFillEntries", state.c6FillEntries).put("originalCloseConfirmed", state.c6NativeClose)
        .put("pageEffectFenceAvailable", state.c6FenceAvailable).put("effectKnowledgeUnknown", !state.c6FenceAvailable)
        .put("selectedInterruptedNotification", state.c6InterruptedNotification)
        .put("pendingSamples", state.c6PendingSamples).put("rawCaptured", state.c6RawCalls).put("rawLiteralHits", state.c6LiteralHits)
        .put("rawEncodedHits", state.c6EncodedHits).put("producedRedactedCallback", state.c6CallbackRedacted)
        .put("realReleasedReads", 1).put("ownedBrowserTreeGone", !browserTree.isEmpty()));
  }

  private static void p8WritePendingProbe(P4RevocationState state, int probe) throws IOException {
    Path root = TEMP.resolve("p4-revocation-producers");
    p4RequirePrivateProducerRoot(root);
    int producer = state.path == 1 ? 1 : 3;
    Path target = root.resolve(state.ordinal + "-" + producer + "-seal.json");
    Path temporary = root.resolve(state.ordinal + "-" + producer + "-probe-" + probe + ".tmp");
    Files.createFile(temporary, java.nio.file.attribute.PosixFilePermissions.asFileAttribute(
        java.nio.file.attribute.PosixFilePermissions.fromString("rw-------")));
    Files.writeString(temporary, C.object().put("case", state.ordinal).put("epoch", state.ordinal)
        .put("producer", producer).put("probe", probe).toString(), StandardCharsets.UTF_8);
    Files.move(temporary, target, java.nio.file.StandardCopyOption.ATOMIC_MOVE,
        java.nio.file.StandardCopyOption.REPLACE_EXISTING);
  }

  private void p8AwaitC6Closed(P4RevocationState state) throws Exception {
    long deadline = System.nanoTime() + Duration.ofSeconds(15).toNanos();
    while (System.nanoTime() < deadline) {
      synchronized (state) {
        if (state.c6NativeClose && state.restored && state.active.isEmpty()
            && (state.path == 2 || state.closed)) break;
      }
      Thread.sleep(20);
    }
    assertThat(state.c6NativeClose).as("original SDK close receipt").isTrue();
    assertThat(state.restored).as("exact descriptors restored after original dispatch lifetime").isTrue();
    assertThat(state.active).isEmpty();
    if (state.path == 1) assertThat(state.closed).isTrue();
    assertThat(state.failed).isFalse();
    assertThat(P4_NOTIFICATION_FAILED.get()).as("no unexplained page observer loss").isFalse();
    assertThat(Files.exists(TEMP.resolve("p4-native-observer-failed")))
        .as("malformed ACK/fence and original close failures remain sticky").isFalse();
  }

  private P6OwnedPid p8OwnedPrimaryBrowser(P4RevocationState state) throws Exception {
    // A strict original SDK/runtime inode binding exists before any signal. Paths remain transient.
    Path root = TEMP.resolve("p4-revocation-producers");
    p4RequirePrivateProducerRoot(root);
    Path samplePath = root.resolve(state.ordinal + "-1.json");
    JsonNode sample = null;
    long deadline = System.nanoTime() + Duration.ofSeconds(2).toNanos();
    while (System.nanoTime() < deadline) {
      if (Files.exists(samplePath, java.nio.file.LinkOption.NOFOLLOW_LINKS)) {
        assertThat(Files.isSymbolicLink(samplePath)).isFalse();
        assertThat(Files.isRegularFile(samplePath, java.nio.file.LinkOption.NOFOLLOW_LINKS)).isTrue();
        assertThat(Files.size(samplePath)).isLessThanOrEqualTo(1024);
        assertThat(Files.getPosixFilePermissions(samplePath)).isEqualTo(
            java.nio.file.attribute.PosixFilePermissions.fromString("rw-------"));
        sample = C.parse(Files.readString(samplePath));
        if (sample.path("contextBound").asBoolean()) break;
      }
      Thread.sleep(10);
    }
    assertThat(sample).as("original SDK launch and selected runtime identity sample").isNotNull();
    assertThat(sample.path("case").asInt()).isEqualTo(state.ordinal);
    assertThat(sample.path("producer").asInt()).isEqualTo(1);
    assertThat(sample.path("epoch").asInt()).isEqualTo(state.ordinal);
    assertThat(sample.path("contextBound").asBoolean()).isTrue();
    assertThat(sample.path("failed").isBoolean() && !sample.path("failed").asBoolean()).isTrue();
    assertThat(sample.size()).isEqualTo(13);
    long device = sample.path("profileDev").asLong(-1), inode = sample.path("profileIno").asLong(-1);
    assertThat(inode).isPositive();
    JsonNode runtime = C.parse(Files.readString(Path.of("../studio-runner/dist/runtime-versions.json")));
    JsonNode provenance = runtime.path("observedBrowserProvenance");
    assertThat(provenance.path("channel").asText()).isEqualTo("chromium");
    assertThat(provenance.path("headless").asBoolean()).isTrue();
    assertThat(provenance.path("chromiumSandbox").asBoolean()).isTrue();
    assertThat(provenance.path("args").size()).isZero();
    assertThat(provenance.path("revision").asText()).isEqualTo(runtime.path("observedChromiumBuild").asText());
    Path browsers = Path.of(System.getenv().getOrDefault("PLAYWRIGHT_BROWSERS_PATH",
        Path.of(System.getProperty("user.home"), "Library", "Caches", "ms-playwright").toString()));
    Path pinned = browsers.resolve("chromium-" + provenance.path("revision").asText())
        .resolve(provenance.path("executableRelativePath").asText()).toRealPath();
    assertThat(HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(Files.readAllBytes(pinned))))
        .isEqualTo(provenance.path("executableSha256").asText());
    Instant workerStarted = guidedWorker.info().startInstant().orElseThrow();
    List<P6OwnedPid> matches = new java.util.ArrayList<>();
    for (ProcessHandle candidate : guidedWorker.descendants().toList()) {
      P6OwnedPid actual = p6SnapshotPid(candidate);
      String command = candidate.info().command().orElse("");
      String[] arguments = candidate.info().arguments().orElse(new String[0]);
      if (command.isEmpty() || !Path.of(command).toRealPath().equals(pinned)
          || java.util.Arrays.stream(arguments).anyMatch(argument -> argument.startsWith("--type="))) continue;
      List<String> profiles = java.util.Arrays.stream(arguments)
          .filter(argument -> argument.startsWith("--user-data-dir=")).toList();
      if (profiles.size() != 1) continue;
      Path profile = Path.of(profiles.getFirst().substring("--user-data-dir=".length()));
      if (Files.isSymbolicLink(profile) || !Files.isDirectory(profile, java.nio.file.LinkOption.NOFOLLOW_LINKS)
          || !profile.toRealPath().equals(profile)) continue;
      if (((Number) Files.getAttribute(profile, "unix:dev")).longValue() != device
          || ((Number) Files.getAttribute(profile, "unix:ino")).longValue() != inode) continue;
      assertThat(Files.getOwner(profile)).isEqualTo(Files.getOwner(TEMP));
      assertThat(Files.getPosixFilePermissions(profile)).isEqualTo(
          java.nio.file.attribute.PosixFilePermissions.fromString("rwx------"));
      p6SameProcess(candidate, actual.started());
      p6SameProcess(guidedWorker.toHandle(), workerStarted);
      assertThat(guidedWorker.descendants().anyMatch(child -> child.pid() == candidate.pid())).isTrue();
      matches.add(actual);
    }
    assertThat(matches).as("exact selected-context pinned full Chromium primary; never whole-worker substitute").hasSize(1);
    return matches.getFirst();
  }

  private static void p8CaptureInjectedCallback(P4RevocationState state, byte[] bytes, int status) {
    try {
      if (status < 200 || status >= 300 || bytes.length < 1 || bytes.length > 262144)
        throw new IllegalArgumentException("C6 actual callback acceptance/bound mismatch");
      JsonNode outcome = C.parseWire(bytes);
      if (!outcome.path("errorMessage").isTextual()) return; // Earlier successes are not positives.
      if (!outcome.path("sessionId").asText().equals(state.guidedSession)
          || !outcome.path("ledger").path("state").asText().equals("UNKNOWN_OUTCOME")
          || !outcome.path("ledger").path("outcomeCode").asText().equals("ACK_LOST")
          || outcome.path("ledger").path("dispatchCount").asInt() != 1)
        throw new IllegalArgumentException("C6 consumed failure identity mismatch");
      String message = outcome.path("errorMessage").asText();
      if (!message.contains("C6_RELEASED") || !message.contains("REDACTED")
          || message.getBytes(StandardCharsets.UTF_8).length > 2048 || message.length() > 2048)
        throw new IllegalArgumentException("C6 real callback is not bounded/redacted");
      C5Capture capture = c5Capture;
      if (capture == null) throw new IllegalArgumentException("C6 original sink capture unavailable");
      for (String value : capture.values) assertJoinedSinkAbsent(Map.of("c6-actual-callback", bytes), value);
      synchronized (state) { state.c6CallbackRedacted++; }
    } catch (RuntimeException | AssertionError failure) {
      state.failed = true; // Sticky even if the original HTTP endpoint returns an error or retries.
    }
  }

  private void p8AssertInjectedProducedSinks(P4RevocationState state, String value) throws Exception {
    Path diagnostic = TEMP.resolve("journal").resolve(C.hash(JSON.valueToTree(state.guidedSession)) + ".diagnostic.jsonl");
    assertThat(Files.isSymbolicLink(diagnostic)).isFalse();
    assertThat(Files.isRegularFile(diagnostic, java.nio.file.LinkOption.NOFOLLOW_LINKS)).isTrue();
    assertThat(Files.size(diagnostic)).isPositive().isLessThanOrEqualTo(262144);
    List<String> lines = Files.readAllLines(diagnostic, StandardCharsets.UTF_8);
    assertThat(lines).hasSize(1);
    JsonNode line = C.parse(lines.getFirst());
    assertThat(line.path("sessionId").asText()).isEqualTo(state.guidedSession);
    assertThat(line.path("stepId").asText()).isEqualTo(TWO);
    String message = line.path("errorMessage").asText();
    assertThat(message).contains("C6_RELEASED", "REDACTED");
    assertThat(message.getBytes(StandardCharsets.UTF_8).length).isLessThanOrEqualTo(2048);
    assertThat(message.length()).isLessThanOrEqualTo(2048);
    byte[] stderr = Files.readAllBytes(TEMP.resolve("guided.log"));
    assertThat(new String(stderr, StandardCharsets.UTF_8)).contains(message);
    assertJoinedSinkAbsent(Map.of("c6-original-stderr", stderr,
        "c6-real-diagnostic", Files.readAllBytes(diagnostic)), value);
  }
