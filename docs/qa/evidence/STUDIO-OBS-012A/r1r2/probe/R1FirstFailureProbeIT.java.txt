/**
 * OUT-OF-TREE diagnostic probe (dex-r1r2, STUDIO-OBS-012A R1). Never committed. Replays the
 * arrangement of five StudioScheduleApiIT methods whose first failing assertion does not print the
 * processor's return value, up to that assertion, and records what the processor actually returned.
 * Same class config as StudioScheduleApiIT. Records only; asserts nothing.
 */
@SpringBootTest(
    webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT,
    properties = {
      "studio.scheduling.enabled=true", "studio.scheduling.local-grants-enabled=true",
      "checks.scheduling.dispatcher-enabled=false", "checks.scheduling.retry-enabled=false"
    })
@ActiveProfiles("test")
@Testcontainers(disabledWithoutDocker = true)
@DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)
@Import(ObservedPublicationFixture.FreshAuthority.class)
class R1FirstFailureProbeIT {

  @DynamicPropertySource

    ObservedPublicationFixture.register(registry);
  }

  @Container
  static final PostgreSQLContainer<?> DATABASE = new PostgreSQLContainer<>("postgres:15-alpine");

  @DynamicPropertySource
  static void datasource(DynamicPropertyRegistry registry) {
    if (!DATABASE.isRunning()) {
      DATABASE.start();
    }
    registry.add("spring.datasource.url", DATABASE::getJdbcUrl);
    registry.add("spring.datasource.username", DATABASE::getUsername);
    registry.add("spring.datasource.password", DATABASE::getPassword);
    registry.add("spring.datasource.driver-class-name", () -> "org.postgresql.Driver");
  }

  static class MutableClock extends Clock {
    private volatile Instant now = Instant.parse("2026-10-06T00:00:00Z");

    @Override
    public ZoneId getZone() {
      return ZoneOffset.UTC;
    }

    @Override
    public Clock withZone(ZoneId zone) {
      return this;
    }

    @Override
    public Instant instant() {
      return now;
    }

    void set(Instant value) {
      now = value;
    }
  }

  @TestConfiguration
  static class TimeConfiguration {
    @Bean
    @Primary
    MutableClock schedulingTestClock() {
      return new MutableClock();
    }
  }

  private static final Path OUT = Path.of(System.getenv().getOrDefault("R1R2_PROBE_OUT", "/tmp/r1r2-probe.jsonl"));
  private static final AtomicInteger SEQUENCE = new AtomicInteger(71000);
  private static final String OWNER = "schedule-owner@citi.com";
  private static final String PRINCIPAL = "studio-local";
  @Autowired JdbcTemplate jdbc;
  @Autowired StudioScheduleService service;
  @Autowired StudioScheduleProcessor processor;
  @Autowired StudioSchedulingProperties properties;
  @Autowired MutableClock clock;
  @Autowired TransactionTemplate transactions;
  @SpyBean StudioReferenceResolver referenceResolver;
  private UUID testId;
  private UUID revisionId;
  private String csi;
  private Authentication owner;

  @BeforeEach
  void fixture() {
    properties.setEnabled(true);
    properties.setLocalGrantsEnabled(true);
    properties.setBatchSize(20);
    properties.setPerCsiBatchSize(2);
    properties.setPassBudgetMillis(5000);
    reset(referenceResolver);
    clock.set(Instant.parse("2026-10-06T00:00:00Z"));
    csi = Integer.toString(SEQUENCE.incrementAndGet());
    var fixtures = new StudioExecutionFixtures(jdbc);
    fixtures.seedUser(OWNER);
    fixtures.seedCsi(csi, OWNER);
    List<String> steps = List.of("Go to the home page");
    testId = fixtures.seedTest("Scheduled fixture", csi, OWNER, steps);
    revisionId =
        fixtures.confirmRevision(testId, csi, steps, StudioExecutionFixtures.NO_REFERENCES);
    owner = StudioExecutionFixtures.identity(OWNER, csi, "owner");
    properties.getLocalGrants().put(csi + "|Production", PRINCIPAL);
  }

  private StudioScheduleDtos.Schedule enable() {
    var s = service.create(testId, new StudioScheduleDtos.Create("* * * * *", "UTC", PRINCIPAL), owner);
    return service.enabled(
        testId, new StudioScheduleDtos.Enabled(s.version(), true, revisionId, List.of()), owner);
  }

  private void due(StudioScheduleDtos.Schedule s) {
    clock.set(s.nextDueAt());
  }

  private List<Map<String, Object>> occurrences(UUID test) {
    return jdbc.queryForList(
        "SELECT decision, code FROM studio_schedule_occurrences WHERE test_id=?", test);
  }

  private static void record(String test, String what, Object value) {
    String line =
        "{\"test\":\"" + test + "\",\"what\":\"" + what + "\",\"value\":\"" + String.valueOf(value).replace("\"", "'") + "\"}\n";
    System.out.print("R1PROBE " + line);
    try {
      Files.writeString(OUT, line, StandardOpenOption.CREATE, StandardOpenOption.APPEND);
    } catch (java.io.IOException e) {
      throw new IllegalStateException(e);
    }
  }

  private static void await(CountDownLatch latch) {
    try {
      if (!latch.await(10, TimeUnit.SECONDS)) {
        throw new IllegalStateException("Race timeout");
      }
    } catch (InterruptedException ex) {
      Thread.currentThread().interrupt();
      throw new IllegalStateException(ex);
    }
  }

  /** writeFailuresAtHistoryLedgerAndCursorBoundariesRollbackTheWholeOccurrenceUnit, line 824. */
  @Test
  void probeWriteFailures() {
    for (String table : List.of("ui_test_executions", "studio_dispatch_ledger", "studio_schedules")) {
      fixture();
      var schedule = enable();
      due(schedule);
      String column = "ui_test_executions".equals(table) ? "ui_test_id" : "test_id";
      String cursorGuard =
          "studio_schedules".equals(table)
              ? " AND EXISTS (SELECT 1 FROM studio_dispatch_ledger WHERE test_id=NEW.test_id)"
              : "";
      jdbc.execute(
          "CREATE FUNCTION scheduling_write_failure() RETURNS TRIGGER AS $$ BEGIN IF NEW."
              + column + "='" + testId + "'::uuid" + cursorGuard
              + " THEN RAISE EXCEPTION 'fixture persistence boundary'; END IF; RETURN NEW; END; $$"
              + " LANGUAGE plpgsql");
      jdbc.execute(
          "CREATE TRIGGER scheduling_write_failure BEFORE "
              + ("studio_schedules".equals(table) ? "UPDATE" : "INSERT") + " ON " + table
              + " FOR EACH ROW EXECUTE FUNCTION scheduling_write_failure()");
      try {
        record("writeFailures[" + table + "]", "processor.process", processor.process(testId));
      } catch (RuntimeException e) {
        record("writeFailures[" + table + "]", "processor.process threw", e.getClass().getName() + ": " + e.getMessage());
      } finally {
        jdbc.execute("DROP TRIGGER scheduling_write_failure ON " + table);
        jdbc.execute("DROP FUNCTION scheduling_write_failure()");
      }
      record("writeFailures[" + table + "]", "occurrences", occurrences(testId));
      return; // the original fails inside the first iteration
    }
  }

  /** replicasRacingAndRollbackCannotCreateDuplicateOrPartialAcceptedWork, line 287. */
  @Test
  void probeReplicasRacing() throws Exception {
    var s = enable();
    due(s);
    var start = new CountDownLatch(1);
    UUID target = testId;
    var a = CompletableFuture.supplyAsync(() -> { await(start); return processor.process(target); });
    var b = CompletableFuture.supplyAsync(() -> { await(start); return processor.process(target); });
    start.countDown();
    record("replicasRacing", "process a", a.get(10, TimeUnit.SECONDS));
    record("replicasRacing", "process b", b.get(10, TimeUnit.SECONDS));
    record("replicasRacing", "occurrences", occurrences(target));
  }

  /** deadlineOverrunAdvancesCursorOnlyPastAttemptedCsi, line 733. */
  @Test
  void probeDeadlineOverrun() {
    var slow = enable();
    String slowCsi = csi;
    fixture();
    enable();
    UUID fastTest = testId;
    clock.set(slow.nextDueAt());
    properties.setPassBudgetMillis(25);
    doAnswer(
            invocation -> {
              if (slowCsi.equals(invocation.getArgument(0))) {
                Thread.sleep(100);
              }
              return null;
            })
        .when(referenceResolver)
        .preflight(anyString(), anyString(), any(UUID.class), anyString(), any(ReferenceDeclarations.class));
    for (int pass = 0; pass < 60 && occurrences(fastTest).isEmpty(); pass++) {
      processor.poll();
    }
    record("deadlineOverrun", "fast occurrences", occurrences(fastTest));
  }

  /** quotaWaitCrossingGraceOrGrantWithdrawalNeverWritesAcceptedWork, line 774 (first iteration). */
  @Test
  void probeQuotaWait() throws Exception {
    var s = enable();
    due(s);
    var held = new CountDownLatch(1);
    var release = new CountDownLatch(1);
    var blocker =
        CompletableFuture.runAsync(
            () ->
                transactions.executeWithoutResult(
                    status -> {
                      jdbc.queryForObject("SELECT pg_advisory_xact_lock(?,?)", Object.class, 0x5701, 1);
                      held.countDown();
                      await(release);
                    }));
    held.await(5, TimeUnit.SECONDS);
    UUID target = testId;
    var decision = CompletableFuture.supplyAsync(() -> processor.process(target));
    boolean waited = false;
    try {
      long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(5);
      while (System.nanoTime() < deadline && !waited) {
        waited =
            Boolean.TRUE.equals(
                jdbc.queryForObject(
                    "SELECT EXISTS (SELECT 1 FROM pg_locks WHERE locktype='advisory' AND classid=?"
                        + " AND objid=1 AND NOT granted)",
                    Boolean.class,
                    0x5701));
        Thread.sleep(10);
      }
    } finally {
      release.countDown();
    }
    blocker.get(10, TimeUnit.SECONDS);
    record("quotaWait", "processor waited on quota advisory lock", waited);
    record("quotaWait", "processor.process", decision.get(10, TimeUnit.SECONDS));
    record("quotaWait", "occurrences", occurrences(target));
  }

  /** failedCsiRotatesAwayRatherThanStarvingOtherEligibleWork, line 666. */
  @Test
  void probeFailedCsiRotates() {
    var failed = enable();
    String failedCsi = csi;
    fixture();
    enable();
    UUID healthyTest = testId;
    clock.set(failed.nextDueAt());
    jdbc.execute(
        "CREATE FUNCTION fail_scheduling_fixture() RETURNS TRIGGER AS $$ BEGIN IF NEW.csi_id='"
            + failedCsi
            + "' THEN RAISE EXCEPTION 'fixture write failure'; END IF; RETURN NEW; END; $$ LANGUAGE"
            + " plpgsql");
    jdbc.execute(
        "CREATE TRIGGER fail_scheduling_fixture BEFORE INSERT ON studio_schedule_occurrences FOR"
            + " EACH ROW EXECUTE FUNCTION fail_scheduling_fixture()");
    properties.setBatchSize(1);
    properties.setPerCsiBatchSize(1);
    try {
      for (int pass = 0; pass < 60 && occurrences(healthyTest).isEmpty(); pass++) {
        processor.poll();
      }
      record("failedCsiRotates", "healthy occurrences", occurrences(healthyTest));
    } finally {
      jdbc.execute("DROP TRIGGER fail_scheduling_fixture ON studio_schedule_occurrences");
      jdbc.execute("DROP FUNCTION fail_scheduling_fixture()");
    }
  }
