// eslint-disable-next-line no-restricted-imports -- Prelaunch call-count fault oracle and actual full Chromium context.
import {chromium} from 'playwright';
import * as promises from 'node:fs/promises';
import type * as PromiseFs from 'node:fs/promises';
import {join} from 'node:path';
import {tmpdir} from 'node:os';
import {afterEach,expect,it,vi} from 'vitest';
import corpus from '../../../docs/specs/studio-observed-authoring/contracts/v3/publication-fixtures.json' with {type:'json'};
import type {Scope} from '../../src/executor/observedOperations.js';
import {prepareBrowserProfile,BROWSER_PREFERENCES} from '../../src/run/scratchLayout.js';
import {playwrightBrowserFactory,launchGuidedBrowser} from '../../src/run/browser.js';
import {GuidedLaunchFailure} from '../../src/run/guidedBrowser.js';
const fault=vi.hoisted(()=>({unsafePreparation:false}));
vi.mock('node:fs/promises',async importOriginal=>{const actual=await importOriginal<typeof PromiseFs>();return {...actual,async mkdtemp(prefix:string){const path=await actual.mkdtemp(prefix);if(fault.unsafePreparation&&prefix.endsWith('checks-guided-'))await actual.chmod(path,0o777);return path;}};});
afterEach(()=>{fault.unsafePreparation=false;vi.restoreAllMocks();vi.unstubAllEnvs();});
const config={sessionId:'00000000-0000-4000-8000-000000000099',scopeHash:'a'.repeat(64),approvedOrigins:['http://127.0.0.1'],rules:[],knownSecrets:[]};
const scope={...corpus.job.binding.scope,policyVersion:'observed-4-full-chromium-1'} as Scope;
it.each(['preferences-symlink','default-symlink','preferences-replacement','preferences-hardlink','default-replacement'] as const)('refuses unsafe %s before invoking any browser launch',async kind=>{
 const scratch=await promises.mkdtemp(join(tmpdir(),'prelaunch-profile-')),outside=await promises.mkdtemp(join(tmpdir(),'prelaunch-external-')),profile=prepareBrowserProfile(scratch),pref=join(profile,'Default','Preferences');const launch=vi.spyOn(chromium,'launchPersistentContext');try{
  if(kind==='preferences-symlink'){await promises.writeFile(join(outside,'Preferences'),JSON.stringify(BROWSER_PREFERENCES),{mode:0o600});await promises.rm(pref);await promises.symlink(join(outside,'Preferences'),pref);}
  if(kind==='default-symlink'){await promises.rename(join(profile,'Default'),join(outside,'Default'));await promises.symlink(join(outside,'Default'),join(profile,'Default'));}
  if(kind==='preferences-replacement'){await promises.rename(pref,pref+'.old');await promises.writeFile(pref,JSON.stringify(BROWSER_PREFERENCES),{mode:0o600});}
  if(kind==='preferences-hardlink')await promises.link(pref,join(outside,'linked'));
  if(kind==='default-replacement'){await promises.rename(join(profile,'Default'),join(outside,'Default'));await promises.mkdir(join(profile,'Default'),{mode:0o700});await promises.writeFile(pref,JSON.stringify(BROWSER_PREFERENCES),{mode:0o600});}
  await expect(playwrightBrowserFactory.launch({browser:'chromium',viewport:{width:800,height:600},recording:{width:800,height:600},scratchPath:scratch,observedScope:scope})).rejects.toThrow('PROFILE_PREFERENCES_REQUIRED');expect(launch).not.toHaveBeenCalled();
 }finally{await promises.rm(scratch,{recursive:true,force:true});await promises.rm(outside,{recursive:true,force:true});}
});
it.each(['preparation','initial-launch'] as const)('awaits actual scratch removal after guided %s rejection without an acquired context',async stage=>{
 const root=await promises.mkdtemp(join(tmpdir(),'guided-rejection-'));vi.stubEnv('TMPDIR',root);const launch=vi.spyOn(chromium,'launchPersistentContext').mockRejectedValue(Error('INJECTED_LAUNCH_REJECTION'));fault.unsafePreparation=stage==='preparation';try{await expect(launchGuidedBrowser(config,'context',()=>undefined)).rejects.toThrow(stage==='preparation'?'PROFILE_PREFERENCES_REQUIRED':'INJECTED_LAUNCH_REJECTION');expect(await promises.readdir(root)).toEqual([]);expect(launch).toHaveBeenCalledTimes(stage==='preparation'?0:1);}finally{await promises.rm(root,{recursive:true,force:true});}
});
it('retains cleanup capability after actual context acquisition and never deletes scratch before confirmed context closure',async()=>{
 const root=await promises.mkdtemp(join(tmpdir(),'guided-acquired-failure-'));vi.stubEnv('TMPDIR',root);const real=chromium.launchPersistentContext.bind(chromium);let closed=false,firstClose=true;
 vi.spyOn(chromium,'launchPersistentContext').mockImplementation(async(...args)=>{const context=await real(...args),page=context.pages()[0];if(!page)throw Error('PAGE');vi.spyOn(page,'goto').mockRejectedValue(Error('INJECTED_PROBE_FAILURE'));const close=context.close.bind(context);vi.spyOn(context,'close').mockImplementation(async()=>{if(firstClose){firstClose=false;throw Error('INJECTED_UNCONFIRMED_CLOSE');}await close();closed=true;});return context;});
 try{let failure:unknown;try{await launchGuidedBrowser(config,'context',()=>undefined);}catch(error){failure=error;}expect(failure).toBeInstanceOf(GuidedLaunchFailure);expect(closed).toBe(false);expect((await promises.readdir(root)).filter(p=>p.startsWith('checks-guided-'))).toHaveLength(1);if(!(failure instanceof GuidedLaunchFailure))throw Error('CLEANUP_CAPABILITY');await expect(failure.close()).rejects.toThrow('CONTEXT_LOST');expect(closed).toBe(false);expect((await promises.readdir(root)).filter(p=>p.startsWith('checks-guided-'))).toHaveLength(1);await failure.close();expect(closed).toBe(true);expect(await promises.readdir(root)).toEqual([]);}finally{await promises.rm(root,{recursive:true,force:true});}
});
it('refuses a Default directory redirected after actual launch rather than accepting its newly canonical outside path',async()=>{
 const root=await promises.mkdtemp(join(tmpdir(),'guided-default-escape-'));vi.stubEnv('TMPDIR',root);const real=chromium.launchPersistentContext.bind(chromium);
 vi.spyOn(chromium,'launchPersistentContext').mockImplementation(async(...args)=>{const context=await real(...args),profile=args[0],outside=join(root,'outside-default');await promises.rename(join(profile,'Default'),outside);await promises.symlink(outside,join(profile,'Default'));return context;});
 try{let failure:unknown;try{await launchGuidedBrowser(config,'context',()=>undefined);}catch(error){failure=error;}expect(failure).toBeInstanceOf(GuidedLaunchFailure);if(!(failure instanceof GuidedLaunchFailure))throw Error('CLEANUP_CAPABILITY');await failure.close();expect((await promises.readdir(root)).filter(p=>p.startsWith('checks-guided-'))).toEqual([]);}finally{await promises.rm(root,{recursive:true,force:true});}
});
it('launches the actual owned confined guided profile and removes it only after successful close',async()=>{
 const root=await promises.mkdtemp(join(tmpdir(),'guided-owned-positive-'));vi.stubEnv('TMPDIR',root);try{const browser=await launchGuidedBrowser(config,'context',()=>undefined);const evidence=browser.preferenceEvidence;if(!evidence||!browser.scratchDirectory)throw Error('EVIDENCE');expect(evidence.activeProfilePath).toBe(join(await promises.realpath(browser.scratchDirectory),'browser-profile','Default'));expect(evidence.effective).toEqual({credentialsEnableService:false,autofillProfileEnabled:false,autofillCreditCardEnabled:false,restoreOnStartup:5});expect((await promises.stat(evidence.activeProfilePath)).mode&0o777).toBe(0o700);await browser.close();expect(await promises.readdir(root)).toEqual([]);}finally{await promises.rm(root,{recursive:true,force:true});}
});
