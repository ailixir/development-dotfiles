import com.sun.source.tree.*;
import com.sun.source.util.*;
import java.nio.file.*;
import java.util.*;
import javax.lang.model.element.Modifier;
import javax.tools.*;

/** One-use, parse-only editor for the exact Checkstyle diagnostic catalog; no application loading. */
public class DiagnosedStyleEditor {
  record Edit(int start, int end, String replacement, String kind) {}
  record Diagnostic(int line, int column, String rule) {}
  static final class Model extends TreeScanner<Void, Void> {
    final CompilationUnitTree unit;
    final SourcePositions positions;
    final IdentityHashMap<Tree, Tree> parent = new IdentityHashMap<>();
    final Deque<Tree> stack = new ArrayDeque<>();
    final List<Tree> nodes = new ArrayList<>();
    Model(CompilationUnitTree unit, SourcePositions positions) {
      this.unit = unit; this.positions = positions; scan(unit, null);
    }
    @Override public Void scan(Tree node, Void unused) {
      if (node == null) return null;
      if (!stack.isEmpty()) parent.put(node, stack.peek());
      nodes.add(node); stack.push(node); super.scan(node, unused); stack.pop(); return null;
    }
    int start(Tree t) { return Math.toIntExact(positions.getStartPosition(unit, t)); }
    int end(Tree t) { return Math.toIntExact(positions.getEndPosition(unit, t)); }
    String name(Tree t) {
      if (t instanceof VariableTree v) return v.getName().toString();
      if (t instanceof ClassTree v) return v.getSimpleName().toString();
      if (t instanceof MethodTree v) return v.getName().toString();
      if (t instanceof IdentifierTree v) return v.getName().toString();
      if (t instanceof MemberSelectTree v) return v.getIdentifier().toString();
      return "";
    }
  }
  static void require(boolean ok, String why) { if (!ok) throw new IllegalStateException(why); }
  static String camel(String old) {
    StringBuilder out = new StringBuilder(); boolean upper = false;
    for (char ch : old.toCharArray()) {
      if (ch == '_') upper = true;
      else { out.append(upper ? Character.toUpperCase(ch) : ch); upper = false; }
    }
    return out.toString();
  }
  public static void main(String[] args) throws Exception {
    Path root = Path.of(args[0]);
    Map<String, List<Diagnostic>> catalog = new TreeMap<>();
    for (String line : Files.readAllLines(root.resolve("diagnostics.tsv"))) {
      String[] p = line.split("\t");
      catalog.computeIfAbsent(p[0], k -> new ArrayList<>())
          .add(new Diagnostic(Integer.parseInt(p[1]), Integer.parseInt(p[2]), p[3]));
    }
    List<String> ledger = new ArrayList<>();
    List<String> aliases = new ArrayList<>();
    List<String> scopes = new ArrayList<>();
    int braceBodies = 0, wrappers = 0, renamed = 0, emptyHooks = 0;
    for (var entry : catalog.entrySet()) {
      String path = entry.getKey(); Path input = root.resolve("before").resolve(path);
      String source = Files.readString(input);
      JavaCompiler compiler = ToolProvider.getSystemJavaCompiler();
      DiagnosticCollector<JavaFileObject> errors = new DiagnosticCollector<>();
      try (StandardJavaFileManager manager = compiler.getStandardFileManager(errors, null, null)) {
        JavacTask task = (JavacTask) compiler.getTask(null, manager, errors,
            List.of("-proc:none", "--release", "21"), null,
            manager.getJavaFileObjects(input.toFile()));
        CompilationUnitTree unit = task.parse().iterator().next();
        require(errors.getDiagnostics().isEmpty(), path + ": parse diagnostics");
        Model model = new Model(unit, Trees.instance(task).getSourcePositions());
        List<Edit> edits = new ArrayList<>();
        Set<Tree> doneBodies = Collections.newSetFromMap(new IdentityHashMap<>());
        Set<Tree> doneWrappers = Collections.newSetFromMap(new IdentityHashMap<>());
        Set<Tree> doneHooks = Collections.newSetFromMap(new IdentityHashMap<>());
        Set<String> doneAliases = new HashSet<>();
        for (Diagnostic diagnostic : entry.getValue()) {
          int at = Math.toIntExact(unit.getLineMap().getPosition(diagnostic.line(), diagnostic.column()));
          switch (diagnostic.rule()) {
            case "NeedBraces" -> {
              IfTree found = null;
              for (Tree t : model.nodes) if (t instanceof IfTree f && model.start(t) == at) found = f;
              require(found != null && !(found.getThenStatement() instanceof BlockTree), path + ": if node " + at);
              StatementTree body = found.getThenStatement();
              if (doneBodies.add(body)) {
                edits.add(new Edit(model.start(body), model.start(body), "{ ", "IF_THEN_OPEN"));
                edits.add(new Edit(model.end(body), model.end(body), " }", "IF_THEN_CLOSE")); braceBodies++;
              }
            }
            case "LocalFinalVariableName" -> {
              int end = at;
              while (end < source.length() && Character.isJavaIdentifierPart(source.charAt(end))) end++;
              String old = source.substring(at, end); String fresh = camel(old);
              require(!old.equals(fresh) && fresh.matches("[a-z][a-zA-Z0-9]*"), path + ": alias " + old);
              if (!doneAliases.add(old)) break;
              List<VariableTree> declarations = new ArrayList<>();
              for (Tree t : model.nodes) if (t instanceof VariableTree v && v.getName().contentEquals(old)) declarations.add(v);
              require(declarations.size() == 1 && declarations.get(0).getModifiers().getFlags().contains(Modifier.FINAL), path + ": unique final alias " + old);
              for (Tree t : model.nodes) require(!model.name(t).equals(fresh), path + ": alias collision " + fresh);
              Tree scope = model.parent.get(declarations.get(0));
              require(scope instanceof BlockTree, path + ": alias block scope");
              edits.add(new Edit(at, end, fresh, "ALIAS_DECLARATION")); int references = 0;
              for (Tree t : model.nodes) if (t instanceof IdentifierTree id && id.getName().contentEquals(old)) {
                require(model.start(t) >= model.start(scope) && model.end(t) <= model.end(scope), path + ": alias reference escaped " + old);
                require(source.substring(model.start(t), model.end(t)).equals(old), path + ": alias spelling");
                edits.add(new Edit(model.start(t), model.end(t), fresh, "ALIAS_REFERENCE")); references++;
              }
              aliases.add(path + "\t" + old + "\t" + fresh + "\t" + references); renamed++;
            }
            case "Indentation" -> {
              BlockTree block = null;
              for (Tree t : model.nodes) if (t instanceof BlockTree b &&
                  (model.start(t) == at || model.end(t) - 1 == at)) block = b;
              require(block != null && model.parent.get(block) instanceof BlockTree, path + ": standalone block " + at);
              require(((BlockTree) model.parent.get(block)).getStatements().contains(block), path + ": standalone statement");
              if (!doneWrappers.add(block)) break;
              String prefix = source.substring(model.start(block), Math.min(model.end(block), model.start(block) + 600));
              require(prefix.contains("POLL") || prefix.contains("IT-S") || prefix.contains("IT-R"), path + ": no introduced wait marker");
              List<String> names = new ArrayList<>();
              for (StatementTree statement : block.getStatements()) {
                if (statement instanceof VariableTree v) names.add(v.getName().toString());
                if (statement instanceof ClassTree v) names.add(v.getSimpleName().toString());
              }
              require(!names.isEmpty(), path + ": empty introduced scope");
              for (String name : names) {
                long declarations = model.nodes.stream().filter(t ->
                    (t instanceof VariableTree || t instanceof ClassTree) && model.name(t).equals(name)).count();
                require(declarations == 1, path + ": scope declaration not globally unique " + name);
                for (Tree t : model.nodes) if (t instanceof IdentifierTree && model.name(t).equals(name))
                  require(model.start(t) >= model.start(block) && model.end(t) <= model.end(block), path + ": outside binding " + name);
              }
              require(source.charAt(model.start(block)) == '{' && source.charAt(model.end(block)-1) == '}', path + ": exact wrapper braces");
              edits.add(new Edit(model.start(block), model.start(block)+1, "", "REMOVE_WAIT_SCOPE_OPEN"));
              edits.add(new Edit(model.end(block)-1, model.end(block), "", "REMOVE_WAIT_SCOPE_CLOSE"));
              scopes.add(path + "\t" + model.start(block) + "\t" + model.end(block) + "\t" + String.join(",", names)); wrappers++;
            }
            case "WhitespaceAround" -> {
              BlockTree block = null;
              for (Tree t : model.nodes) if (t instanceof BlockTree b && b.getStatements().isEmpty()
                  && model.parent.get(t) instanceof LambdaExpressionTree && (model.start(t)==at || model.end(t)-1==at)) block=b;
              require(block != null, path + ": diagnosed empty fixture lambda " + at);
              if (doneHooks.add(block)) {
                edits.add(new Edit(model.start(block)+1, model.start(block)+1, " /* no-op fixture hook */ ", "EMPTY_FIXTURE_HOOK_COMMENT")); emptyHooks++;
              }
            }
            default -> throw new IllegalStateException("Unexpected rule " + diagnostic.rule());
          }
        }
        edits.sort(Comparator.comparingInt(Edit::start).thenComparingInt(Edit::end));
        int lastEnd=-1;
        for (Edit edit : edits) { require(edit.start() >= lastEnd, path + ": overlapping source edits"); lastEnd=edit.end(); }
        StringBuilder output = new StringBuilder(source);
        ListIterator<Edit> iterator = edits.listIterator(edits.size());
        while (iterator.hasPrevious()) {
          Edit edit=iterator.previous(); output.replace(edit.start(),edit.end(),edit.replacement());
          ledger.add(path+"\t"+edit.start()+"\t"+edit.end()+"\t"+edit.kind()+"\t"+edit.replacement().replace("\t","\\t").replace("\n","\\n"));
        }
        Path target=root.resolve("candidate").resolve(path); Files.createDirectories(target.getParent()); Files.writeString(target,output);
      }
    }
    Files.write(root.resolve("edits.tsv"),ledger); Files.write(root.resolve("aliases.tsv"),aliases); Files.write(root.resolve("removed-scopes.tsv"),scopes);
    System.out.printf("{\"ifBodies\":%d,\"aliases\":%d,\"wrappers\":%d,\"emptyHooks\":%d}%n",braceBodies,renamed,wrappers,emptyHooks);
  }
}
