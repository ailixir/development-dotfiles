  public static void main(String[] args) throws Exception {
    Path root=Path.of(args[0]);
    Map<String,Set<String>> allowed=new TreeMap<>();
    Map<String,Integer> duplicates=new HashMap<>();
    for (String row:Files.readAllLines(root.resolve("approved-scopes.tsv"))) {
      String[] p=row.split("\t");
      allowed.computeIfAbsent(p[0],k->new HashSet<>()).add(p[3]);
      for(String name:p[4].split(",")) duplicates.merge(p[0]+":"+name,1,Integer::sum);
    }
    List<String> editLedger=new ArrayList<>(),aliasLedger=new ArrayList<>(),scopeLedger=new ArrayList<>(),parseLedger=new ArrayList<>();
    int scopeCount=0,renameCount=0;
    for(var entry:allowed.entrySet()) {
      String path=entry.getKey();Path input=root.resolve("before").resolve(path);String source=Files.readString(input);
      JavaCompiler compiler=ToolProvider.getSystemJavaCompiler();
      DiagnosticCollector<JavaFileObject> errors=new DiagnosticCollector<>();
      try(StandardJavaFileManager manager=compiler.getStandardFileManager(errors,null,null)) {
        JavacTask task=(JavacTask)compiler.getTask(null,manager,errors,List.of("-proc:none","--release","21"),null,manager.getJavaFileObjects(input.toFile()));
        CompilationUnitTree unit=task.parse().iterator().next();require(errors.getDiagnostics().isEmpty(),path+": parse failure");
        Model model=new Model(unit,Trees.instance(task).getSourcePositions());List<Edit> edits=new ArrayList<>();
        List<BlockTree> flattened=new ArrayList<>();Map<String,String> renamed=new TreeMap<>();Map<String,List<Integer>> referencePositions=new TreeMap<>();
        List<String> oldJumps=new ArrayList<>();
        for(Tree t:model.nodes) {
          if(t instanceof BreakTree v)oldJumps.add("break:"+v.getLabel());
          if(t instanceof ContinueTree v)oldJumps.add("continue:"+v.getLabel());
        }
        for(Tree t:model.nodes) if(t instanceof LabeledStatementTree label && entry.getValue().contains(label.getLabel().toString())) {
          String labelName=label.getLabel().toString();require(label.getStatement() instanceof BlockTree,path+": original block missing");
          BlockTree block=(BlockTree)label.getStatement();require(model.parent.get(label) instanceof BlockTree,path+": label was not a standalone statement");
          require(!oldJumps.contains("break:"+labelName)&&!oldJumps.contains("continue:"+labelName),path+": label referenced");
          require(source.substring(model.start(label),model.start(block)).replaceAll("\\s","").equals(labelName+":"),path+": label removal touches content");
          require(source.charAt(model.start(block))=='{'&&source.charAt(model.end(block)-1)=='}',path+": original scope braces");
          List<String> retained=new ArrayList<>(),moved=new ArrayList<>();
          for(StatementTree statement:block.getStatements()) if(statement instanceof VariableTree||statement instanceof ClassTree) {
            String name=model.name(statement);moved.add(name);boolean generic=duplicates.get(path+":"+name)>1;
            if(generic) {
              require(statement instanceof VariableTree && Set.of("bridgeDeadline","bridgePoll","itsPoll").contains(name),path+": unapproved reused binding "+name);
              String fresh="styleScope"+labelName.substring("styleWaitScope".length())+Character.toUpperCase(name.charAt(0))+name.substring(1);
              require(!java.util.regex.Pattern.compile("\\b"+fresh+"\\b").matcher(source).find(),path+": fresh name collision");
              require(!renamed.containsKey(fresh),path+": duplicate fresh name");
              long nestedDeclarations=model.nodes.stream().filter(n->(n instanceof VariableTree||n instanceof ClassTree)&&model.name(n).equals(name)&&model.start(n)>=model.start(block)&&model.end(n)<=model.end(block)).count();
              require(nestedDeclarations==1,path+": shadowed old binding "+name);
              VariableTree variable=(VariableTree)statement;require(variable.getInitializer()!=null,path+": missing generated initializer");
              int begin=model.start(variable),limit=model.start(variable.getInitializer());
              java.util.regex.Matcher match=java.util.regex.Pattern.compile("\\b"+name+"\\b").matcher(source.substring(begin,limit));
              require(match.find(),path+": declaration name missing");int decl=begin+match.start();require(!match.find(),path+": declaration prefix ambiguous");
              edits.add(new Edit(decl,decl+name.length(),fresh,"SCOPED_BINDING_DECLARATION"));List<Integer> references=new ArrayList<>();
              for(Tree ref:model.nodes) if(ref instanceof IdentifierTree id&&id.getName().contentEquals(name)&&model.start(ref)>=model.start(block)&&model.end(ref)<=model.end(block)) {
                Tree parent=model.parent.get(ref);
                require(!(parent instanceof MethodInvocationTree call&&call.getMethodSelect()==ref),path+": identifier could be a method "+name);
                require(source.substring(model.start(ref),model.end(ref)).equals(name),path+": reference spelling");
                edits.add(new Edit(model.start(ref),model.end(ref),fresh,"SCOPED_BINDING_REFERENCE"));references.add(model.start(ref));
              }
              require(!references.isEmpty(),path+": generated binding unreferenced");renamed.put(fresh,name);referencePositions.put(fresh,references);
              aliasLedger.add(path+"\t"+labelName+"\t"+name+"\t"+fresh+"\t"+decl+"\t"+references.size());renameCount++;
            } else {
              long declarations=model.nodes.stream().filter(n->(n instanceof VariableTree||n instanceof ClassTree)&&model.name(n).equals(name)).count();
              require(declarations==1,path+": retained declaration not unique "+name);
              for(Tree ref:model.nodes) if(ref instanceof IdentifierTree&&model.name(ref).equals(name))require(model.start(ref)>=model.start(block)&&model.end(ref)<=model.end(block),path+": retained name could capture outside references "+name);
              for(Tree ref:model.nodes) if((ref instanceof MethodTree||ref instanceof MemberSelectTree)&&model.name(ref).equals(name))throw new IllegalStateException(path+": retained name/import/member ambiguity "+name);
              retained.add(name);
            }
          }
          edits.add(new Edit(model.start(label),model.start(block),"","REMOVE_UNREFERENCED_LABEL"));
          edits.add(new Edit(model.start(block),model.start(block)+1,"","REMOVE_INTRODUCED_WRAPPER_OPEN"));
          edits.add(new Edit(model.end(block)-1,model.end(block),"","REMOVE_INTRODUCED_WRAPPER_CLOSE"));
          flattened.add(block);scopeLedger.add(path+"\t"+labelName+"\t"+model.start(block)+"\t"+model.end(block)+"\t"+String.join(",",retained)+"\t"+String.join(",",moved));scopeCount++;
        }
        require(flattened.size()==entry.getValue().size(),path+": approved scope count");
        edits.sort(Comparator.comparingInt(Edit::start).thenComparingInt(Edit::end));int last=-1;
        for(Edit edit:edits){require(edit.start()>=last,path+": overlapping edits");last=edit.end();}
        StringBuilder candidate=new StringBuilder(source);ListIterator<Edit> iterator=edits.listIterator(edits.size());
        while(iterator.hasPrevious()){Edit edit=iterator.previous();candidate.replace(edit.start(),edit.end(),edit.replacement());editLedger.add(path+"\t"+edit.start()+"\t"+edit.end()+"\t"+edit.kind()+"\t"+edit.replacement());}
        Path target=root.resolve("candidate").resolve(path);Files.createDirectories(target.getParent());Files.writeString(target,candidate);
        DiagnosticCollector<JavaFileObject> afterErrors=new DiagnosticCollector<>();
        JavacTask afterTask=(JavacTask)compiler.getTask(null,manager,afterErrors,List.of("-proc:none","--release","21"),null,manager.getJavaFileObjects(target.toFile()));
        CompilationUnitTree afterUnit=afterTask.parse().iterator().next();require(afterErrors.getDiagnostics().isEmpty(),path+": candidate parse failure");
        Model after=new Model(afterUnit,Trees.instance(afterTask).getSourcePositions());
        java.util.function.IntUnaryOperator position=old->{int delta=0;for(Edit edit:edits)if(edit.end()<=old)delta+=edit.replacement().length()-(edit.end()-edit.start());return old+delta;};
        Map<Integer,IfTree> ifs=new HashMap<>();List<String> newJumps=new ArrayList<>();
        for(Tree node:after.nodes){if(node instanceof IfTree v)ifs.put(after.start(node),v);if(node instanceof BreakTree v)newJumps.add("break:"+v.getLabel());if(node instanceof ContinueTree v)newJumps.add("continue:"+v.getLabel());if(node instanceof LabeledStatementTree v)require(!entry.getValue().contains(v.getLabel().toString()),path+": synthetic label remained");}
        int oldIfs=0;
        for(Tree node:model.nodes)if(node instanceof IfTree v){oldIfs++;IfTree fresh=ifs.get(position.applyAsInt(model.start(node)));require(fresh!=null,path+": conditional lost");require(v.getThenStatement().getKind()==fresh.getThenStatement().getKind(),path+": then branch changed");require((v.getElseStatement()==null)==(fresh.getElseStatement()==null),path+": else association changed");if(v.getElseStatement()!=null)require(v.getElseStatement().getKind()==fresh.getElseStatement().getKind()&&position.applyAsInt(model.start(v.getElseStatement()))==after.start(fresh.getElseStatement()),path+": else branch moved");}
        require(oldIfs==ifs.size()&&oldJumps.equals(newJumps),path+": conditional/jump inventory changed");
        for(var rename:renamed.entrySet()) {
          String fresh=rename.getKey();long declarations=after.nodes.stream().filter(n->n instanceof VariableTree&&after.name(n).equals(fresh)).count();require(declarations==1,path+": renamed declaration resolution");
          List<Integer> actual=new ArrayList<>();for(Tree n:after.nodes)if(n instanceof IdentifierTree&&after.name(n).equals(fresh))actual.add(after.start(n));
          List<Integer> expected=referencePositions.get(fresh).stream().map(position::applyAsInt).toList();require(actual.equals(expected),path+": changed reference binding/order "+fresh);
        }
        parseLedger.add(path+"\t"+oldIfs+"\t"+flattened.size()+"\t"+renamed.size()+"\t"+oldJumps.size()+"\tEXACT_BRANCHES_JUMPS_AND_SCOPED_REFERENCES_RETAINED");
      }
    }
    require(scopeCount==97&&renameCount==38,"Exact authorized counts");
    Files.write(root.resolve("edits.tsv"),editLedger);Files.write(root.resolve("aliases.tsv"),aliasLedger);Files.write(root.resolve("scope-proof.tsv"),scopeLedger);Files.write(root.resolve("parse-proof.tsv"),parseLedger);
    System.out.printf("{\"flattenedIntroducedScopes\":%d,\"freshScopedVariables\":%d,\"typeRenames\":0}%n",scopeCount,renameCount);
  }
