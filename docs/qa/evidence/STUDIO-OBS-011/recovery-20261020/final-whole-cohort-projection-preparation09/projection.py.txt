from pathlib import Path
from collections import Counter
import os,stat,json,hashlib,re,sys
R=Path('/Users/brunobuger/orca/workspaces/put-checks/ui-studio')
seen=set(); report={'scope':'REMAINING_FOUR_ORIGINAL_WHOLE6_COHORT_AUDIT_RECORDS_ONLY','notQaVerdict':True,'projectionComplete':False,'replayCohorts':None}
def read(ref):
    p=R/ref['path']; assert not Path(ref['path']).is_absolute() and p.resolve().is_relative_to(R) and str(p) not in seen
    current=p
    while current!=R:
        assert not current.is_symlink(); current=current.parent
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW)
    with os.fdopen(fd,'rb') as f:
        a=os.fstat(f.fileno()); assert stat.S_ISREG(a.st_mode) and a.st_size==ref['bytes'] and oct(stat.S_IMODE(a.st_mode))==ref['mode']
        raw=f.read(a.st_size+1); b=os.fstat(f.fileno())
    assert len(raw)==a.st_size and (a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns,a.st_ctime_ns)==(b.st_dev,b.st_ino,b.st_size,b.st_mtime_ns,b.st_ctime_ns)
    assert hashlib.sha256(raw).hexdigest()==ref['sha256']; seen.add(str(p)); return raw
def main():
    assert len(sys.argv)==3 and re.fullmatch('[0-9a-f]{64}',sys.argv[2])
    p=Path(sys.argv[1]); assert p.is_absolute() and p.is_file() and not p.is_symlink() and p.resolve().is_relative_to(R)
    s=p.stat(); raw=read({'path':str(p.relative_to(R)),'sha256':sys.argv[2],'bytes':s.st_size,'mode':oct(stat.S_IMODE(s.st_mode))}); plan=json.loads(raw)
    assert plan['scope']=='REMAINING_FOUR_ORIGINAL_WHOLE6_COHORT_AUDIT_RECORDS_ONLY' and plan['root']==str(R)
    data=json.loads(read(plan['receipt']));wanted=json.loads(read(plan['wantedReplayDefinitions']));before=json.loads(read(plan['sourceBefore']))
    assert data['exit']==0 and data['gradleExit']==0
    assert all(data.get(key) is True for key in ('sourceStable','distStable','testInputStable','taskCollectionComplete'))
    assert data['explicitTestTasks']==['test','guidedIncrementalTest','crossReplicaTest'] and set(data['requiredFreshExecTasks'])=={'test','guidedIncrementalTest','crossReplicaTest'}
    assert data['counts']=={'total':2792,'passed':2783,'failed':0,'errors':0,'skipped':9} and len(data['testcases'])==2792
    assert [r for r in data['guidedCohortFiles'] if Path(r['path']).name in ('cohort.json','audit-final.json')]==plan['cohortAndAuditFiles'] and len(plan['cohortAndAuditFiles'])==4
    assert all(r['bytes']<=plan['ordinaryFileBytes'] for r in plan['cohortAndAuditFiles'])
    source=next(r for r in before['files'] if r['path']==wanted['sourcePath']);expected_source=plan['expectedReplaySha256']
    if source['sha256']!=expected_source:raise ValueError('EXPECTED_STAGED_SOURCE_HASH_MISMATCH')
    root=R
    closedCohorts={};closedAudits={}
    finals={};cohorts={}
    for row in data['guidedCohortFiles']:
        path=root/row['path']
        if path.name in ('audit-final.json','cohort.json'):
            raw=read(row)
            if path.name=='cohort.json':
                cohort=json.loads(raw);task=row['task'];assert task not in cohorts
                assert all(cohort.get(key) is True for key in ('auditComplete','privacyComplete','producersDrained',
                    'finalScansAttempted','modelEvidenceComplete','budgetEvidenceComplete','responseEvidenceComplete'))
                assert cohort['retentionFailures']==0
                actual_invocations=Counter((r['method'],r['variant'],r['variantOrdinal']) for r in cohort['invocations'])
                expected_invocations=Counter((r['method'],r['variant'],r['variantOrdinal']) for r in wanted['cases'] if r['task']==task)
                assert actual_invocations==expected_invocations
                assert all(r['passed'] is True and r['afterEachComplete'] is True for r in cohort['invocations'])
                cohorts[task]=hashlib.sha256(raw).hexdigest()
                assert type(cohort['retentionFailures']) is int
                closedCohorts[task]={key:cohort[key] for key in ('auditComplete','privacyComplete','producersDrained','finalScansAttempted','modelEvidenceComplete','budgetEvidenceComplete','responseEvidenceComplete','retentionFailures')}
                closedCohorts[task].update({'invocations':len(cohort['invocations']),'allPassed':all(r['passed'] is True for r in cohort['invocations']),'allAfterEachComplete':all(r['afterEachComplete'] is True for r in cohort['invocations'])})
                continue
            audit=json.loads(raw);task=row['task'];assert task not in finals
            assert all(audit.get(key) is True for key in ('auditComplete','producersDrained','contextClosed'))
            if audit.get('exactLaneObserved') is not True:raise ValueError('EXACT_FULL_PROFILE_CONTROL_NOT_OBSERVED')
            assert audit['expectedInvocations']==wanted['counts'][task]
            assert audit['observedInvocations']==wanted['counts'][task]
            if audit['finalCorpusControlChecks']<2:raise ValueError('FINAL_CORPUS_CONTROL_BOUNDARIES_MISSING')
            assert audit['corpusControlChecks']>=2
            assert audit['contentScans']>=2
            finals[task]=hashlib.sha256(raw).hexdigest()
            assert all(type(audit[key]) is int for key in ('expectedInvocations','observedInvocations','finalCorpusControlChecks','corpusControlChecks','contentScans'))
            closedAudits[task]={key:audit[key] for key in ('auditComplete','producersDrained','contextClosed','exactLaneObserved','expectedInvocations','observedInvocations','finalCorpusControlChecks','corpusControlChecks','contentScans')}
    assert set(finals)==set(cohorts)=={'test','guidedIncrementalTest'}
    assert all(data['taskResults'][task]['freshExec'] is True for task in finals)
    projections={}
    for row in plan['cohortAndAuditFiles']:
        task=row['task']; projections.setdefault(task,{})
        if Path(row['path']).name=='cohort.json':
            projections[task]['cohortSha256']=cohorts[task]
        else: projections[task]['auditFinalSha256']=finals[task]
    report['replayCohorts']={'sourceSha256':expected_source,'taskBoundOriginalHashes':projections,'cohortFlagsAndCounts':closedCohorts,'auditFlagsAndCounts':closedAudits,'approvedAuditPrivacyProducerDrainFinalScanModelBudgetResponseRetentionZeroInvocationAfterEachExactLaneAndCorpusPredicatesSatisfied':True,'exactInvocationCounts':wanted['counts'],'noPairCliOrSyntheticReceipt':True}
    report['projectionComplete']=True; report['protectedCohortAuditFilesReadOnce']=True
    report['xmlOrGradleLogRead']=False; report['originalWholeReceiptUnmodified']=True
try:
    main()

    reason=error.args[0] if error.args and type(error.args[0]) is str and error.args[0] in allowed else 'REQUIRED_EXISTING_WHOLE_COHORT_PROJECTION_INCOMPLETE'
    report['reason']=reason
    print(json.dumps(report,separators=(',',':')));sys.exit(2)
