import os, stat, json, sys
PARENT='/private/var/folders/69/txvg5ksj635bk5xr530hcrx00000gn/T'
PREFIX='obs007-joined-'
LIMIT=65536
fd=None
result={'scope':'ONE_PARENT_DIRECT_NAMES_ONLY_RESOURCE_CENSUS','complete':False,'entryLimit':LIMIT,'directEntries':0,'prefixMatchingNames':0,'childMetadataOrBodiesRead':False,'namesExported':False,'notCustodyOrDiagnosticVerdict':True}
try:
    fd=os.open('/',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    for part in PARENT.strip('/').split('/'):
        nextfd=os.open(part,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=fd)
        os.close(fd)
        fd=nextfd
    before=os.fstat(fd)
    assert stat.S_ISDIR(before.st_mode) and before.st_uid==os.geteuid()
    with os.scandir(fd) as entries:
        for entry in entries:
            result['directEntries']+=1
            if result['directEntries']>LIMIT:
                result['reason']='PARENT_NAME_LIMIT_EXCEEDED'
                break
            if entry.name.startswith(PREFIX):
                result['prefixMatchingNames']+=1
        else:
            after=os.fstat(fd)
            signature=lambda s:(s.st_dev,s.st_ino,s.st_mode,s.st_uid,s.st_mtime_ns,s.st_ctime_ns)
            result['parentStable']=signature(before)==signature(after)
            result['complete']=result['parentStable']
            if not result['complete']:
                result['reason']='PARENT_CHANGED_DURING_NAME_ENUMERATION'
except Exception:
    result['reason']='NAME_CENSUS_INCOMPLETE'
finally:
    if fd is not None:os.close(fd)
print(json.dumps(result,separators=(',',':')))
sys.exit(0 if result['complete'] else 2)
