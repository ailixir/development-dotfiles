def read(ref):
    p=R/ref['path']; assert not Path(ref['path']).is_absolute() and p.resolve().is_relative_to(R) and str(p) not in seen
    current=p
    while current!=R:
        assert not current.is_symlink(); current=current.parent
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW)
    with os.fdopen(fd,'rb') as f:
        a=os.fstat(f.fileno()); assert stat.S_ISREG(a.st_mode) and a.st_size==ref['bytes'] and oct(stat.S_IMODE(a.st_mode))==ref['mode']
        raw=f.read(a.st_size+1); b=os.fstat(f.fileno())
    assert len(raw)==a.st_size and (a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns,a.st_ctime_ns)==(b.st_dev,b.st_ino,b.st_size,b.st_mtime_ns,b.st_ctime_ns)
    assert hashlib.sha256(raw).hexdigest()==ref['sha256']; seen.add(str(p)); return raw

def integer(value):
