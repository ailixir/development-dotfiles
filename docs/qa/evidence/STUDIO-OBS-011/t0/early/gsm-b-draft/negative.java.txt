/**
 * Real PostgreSQL negative-only compositions. No plan, obligation, manifest, admission or
 * translation-state fixture is inserted. Both a missing installed store and an installed store
 * with no state are tested. These probes never become positive planner or browser evidence.
 */
class GuidedAdmissionWithoutTranslationIT {
  private static final PostgreSQLContainer<?> POSTGRES =
      new PostgreSQLContainer<>("postgres:15-alpine");
  private static final Instant NOW = Instant.parse("2026-10-08T03:00:00Z");
  private static JsonNode documents;
  private static final String[] TABLES = {
    "studio_guided_actions", "studio_guided_obligation_admissions",
    "studio_guided_preparations", "studio_guided_translation_plans",
    "studio_guided_translation_state", "studio_guided_source_manifests",
    "studio_guided_model_attempts", "studio_guided_clarification_contexts",
    "studio_guided_materializations", "studio_guided_materialization_deliveries",
    "studio_guided_sessions"
  };

  @BeforeAll
  static void database() throws Exception {
    POSTGRES.start();
    try (var input = GuidedAdmissionWithoutTranslationIT.class.getResourceAsStream(
        "/studio/guided/session-it-documents.json")) {
      documents = new ObjectMapper().readTree(input).path("documents");
    }
  }

  @AfterAll
  static void close() {
    POSTGRES.stop();
  }

  @Test
  void sideEffectConfirmationWithoutTranslationIsAlreadyRefusedByWrapper() {
    eachComposition(f -> {
      f.observe();
      ObjectNode question = f.question(true);
      f.refuses("side-effect-wrapper-control", () ->
          f.service.sideEffectConfirmation(f.controller, f.id, 1, question));
    });
  }

  @Test
  void clarifyConfirmationWithoutTranslationRefusesBeforeRefreshContinuation() {
    eachComposition(f -> {
      f.observe();
      ObjectNode question = f.question(true);
      f.service.clarification(f.controller, f.id, 1, "ask-confirmation", question);
      // The legacy non-production SPI cannot create this confirmation through the guarded
      // translation wrapper. This negative-only progress fixture supplies no executable authority:
      // the actual session, question, owner, materialization and command gates remain real.
      f.repository.transaction(() -> {
        var current = f.repository.lock(f.id);
        assertThat(current.progress().path("clarification")).isEqualTo(question);
        current.progress().putObject("sideEffectConfirmation")
            .put("clarificationId", question.path("clarificationId").asText())
            .put("status", "PENDING");
        f.repository.save(current);
        return null;
      });
      f.refuses("clarify-confirmation", () ->
          f.service.clarify(f.user, f.id, "answer-confirmation", f.answer()));
    });
  }

  @Test
  void clarifyChoiceWithoutTranslationRefusesBeforeAnswerContinuation() {
    eachComposition(f -> {
      f.observe();
      f.service.clarification(f.controller, f.id, 1, "ask-choice", f.question(false));
      f.refuses("clarify-choice", () ->
          f.service.clarify(f.user, f.id, "answer-choice", f.answer()));
    });
  }

  @Test
  void admissionWithoutTranslationRefusesBeforeAnyActionAuthorizationOrWrite() {
    eachComposition(f -> {
      f.observe();
      ObjectNode proposal = f.proposal();
      ObjectNode proof = f.materialization.identity(f.context());
      proof.put("preparationId", UUID.randomUUID().toString())
          .put("requestDigest", GuidedContracts.hash(proposal))
          .put("evidenceHash", proposal.path("evidenceHash").asText())
          .put("authoredInputsHash", proposal.path("authoredInputsHash").asText())
          .put("expiresAt", NOW.plusSeconds(20).toString());
      // Existing authenticated preparation seam; never store-level planner/obligation seeding.
      f.service.storePreparation(f.controller, f.id, 1, proposal, proof);
      assertThat(f.count("studio_guided_preparations")).isEqualTo(1);
      f.refuses("admission", () ->
          f.service.admit(f.controller, f.id, 1, "admit-negative-probe", proposal));
    });
  }

  @Test
  void bootstrapWithoutTranslationRefusesBeforeEvidenceReceiptOrStateWrite() {
    eachComposition(f -> {
      ObjectNode context = f.context();
      ObjectNode request = f.materialization.identity(context);
      ObjectNode evidence = f.contracts.object().put("kind", "initialBlank")
          .put("sessionId", f.id).put("contextId", context.path("contextId").asText())
          .put("creationId", UUID.randomUUID().toString()).put("documentId", "blank-document")
          .put("bootstrapId", UUID.randomUUID().toString()).put("url", "about:blank")
          .putNull("origin").put("generation", 0);
      request.set("executionEvidence", evidence);
      request.put("evidenceHash", GuidedContracts.hash(evidence));
      f.contracts.validate("v3/bridge#/definitions/bootstrap", request);
      f.refuses("bootstrap", () -> f.service.bootstrap(f.worker, f.id, 1, "bootstrap", request));
    });
  }

  private static void eachComposition(Consumer<Fixture> assertion) {
    for (boolean installEmptyStore : List.of(false, true)) {
      assertion.accept(new Fixture(installEmptyStore));
    }
  }

  private static final class Fixture {
    private final GuidedContracts contracts = new GuidedContracts();
    private final Clock clock = Clock.fixed(NOW, ZoneOffset.UTC);
    private final Authentication user = new BffAuthenticationToken(new UserIdentity(
        "user-1", "fixture@example.test", "Fixture", "Test", List.of(),
        List.of(new CsiOwnershipEntry("CSI-123", "owner"))));
    private final Authentication worker =
        new GuidedInternalSecurity.Identity("worker-1", "owning-worker");
    private final Authentication controller =
        new GuidedInternalSecurity.Identity("controller-1", "backend-controller");
    private final JdbcTemplate jdbc;
    private final GuidedSessionRepository repository;
    private final NegativeAccess access;
    private final GuidedSessionService service;
    private final GuidedMaterializationService materialization;
    private final GuidedPreparationStore preparations;
    private final GuidedBindingStore bindings;
    private final String id;
    private final boolean installEmptyStore;

    Fixture(boolean installEmptyStore) {
      this.installEmptyStore = installEmptyStore;
      String database = "no_translation_" + UUID.randomUUID().toString().replace("-", "");
      var admin = new DriverManagerDataSource(
          POSTGRES.getJdbcUrl(), POSTGRES.getUsername(), POSTGRES.getPassword());
      new JdbcTemplate(admin).execute("CREATE DATABASE " + database);
      var source = new DriverManagerDataSource(
          POSTGRES.getJdbcUrl().replace(POSTGRES.getDatabaseName(), database),
          POSTGRES.getUsername(), POSTGRES.getPassword());
      Flyway.configure().dataSource(source).target("22").load().migrate();
      jdbc = new JdbcTemplate(source);
      repository = new GuidedSessionRepository(jdbc,
          new TransactionTemplate(new DataSourceTransactionManager(source)), contracts);
      access = spy(new NegativeAccess());
      service = new GuidedSessionService(repository, contracts, access,
          new GuidedSessionProperties(true, Map.of()), clock);
      service.installPresentation(new GuidedPublicPresentationProjector(jdbc, contracts));
      materialization = new GuidedMaterializationService(jdbc, contracts);
      preparations = spy(new GuidedPreparationStore(jdbc, contracts));
      bindings = spy(new GuidedBindingStore(jdbc, contracts, materialization));
      service.installBridge(materialization, preparations, bindings,
          new GuidedAssignmentStore(jdbc, contracts));
      if (installEmptyStore) {
        service.installTranslation(new GuidedTranslationStore(jdbc, contracts),
            new GuidedClarificationContextStore(jdbc, contracts));
      }
      com.citi.put.checks.studio.support.ObservedPublicationFixture.installGovernance(
          service, jdbc, "CSI-123", "orders", List.of("fixture"),
          List.of("https://orders.example.test"), null);
      ObjectNode start = contracts.object().put("schemaVersion", 3).put("testId", "test-1")
          .put("sourceRevision", "1").put("configurationRevision", "1")
          .put("environmentId", "fixture").put("applicationId", "orders");
      start.set("versions", versions());
      id = service.start(user, "start", start).path("sessionId").asText();
      ObjectNode intent = service.prepareAssignment(controller, id, "assign", "worker-1");
      ObjectNode claim = contracts.object().put("schemaVersion", 3).put("sessionId", id)
          .put("workerId", "worker-1").put("expectedFence", 0);
      claim.set("versions", versions());
      JsonNode owned = service.claim(worker, id, "claim", claim);
      ObjectNode receipt = contracts.object().put("schemaVersion", 3).put("status", "OWNED")
          .put("ownerFence", 1)
          .put("leaseExpiresAt", owned.path("owner").path("leaseExpiresAt").asText());
      receipt.set("assignment", intent);
      service.acknowledgeAssignment(controller, id, receipt);
      service.materialize(controller, id, 1, "materialize", new ObjectMapper().createArrayNode());
      // First materialization is revision1: V21 creates pending delivery only for revision>1.
      assertThat(context().path("contextRevision").asInt()).isEqualTo(1);
      assertThat(materialization.pendingDeliveryId(id)).isNull();
      for (String table : List.of("studio_guided_translation_state", "studio_guided_source_manifests",
          "studio_guided_translation_plans", "studio_guided_obligation_admissions",
          "studio_guided_actions", "studio_guided_preparations")) {
        assertThat(count(table)).as("negative fixture has no planner authority: %s", table).isZero();
      }
      repository.transaction(() -> {
        var current = repository.lock(id);
        assertThat(current.document().path("owner").path("workerId").asText()).isEqualTo("worker-1");
        assertThat(current.document().path("owner").path("fence").asInt()).isEqualTo(1);
        assertThat(Instant.parse(current.document().path("owner").path("leaseExpiresAt").asText()))
            .isAfter(NOW);
        assertThat(current.document().path("state").asText()).isEqualTo("PREPARING");
        return null;
      });
    }

    private ObjectNode versions() {
      return contracts.object().put("contract", 3);
    }

    private ObjectNode context() {
      return repository.transaction(() -> materialization.context(repository.lock(id).document()));
    }

    private String scope() {
      return repository.transaction(() -> repository.lock(id).document().path("scopeHash").asText());
    }

    private int count(String table) {
      return jdbc.queryForObject("SELECT count(*) FROM " + table + " WHERE session_id=?", Integer.class, id);
    }

    private void observe() {
      ObjectNode observation = documents.path("observation").deepCopy();
      observation.put("sessionId", id).put("scopeHash", scope()).put("generation", 1);
      service.observe(worker, id, 1, "observe", observation);
      assertThat(repository.transaction(() -> repository.lock(id).document().path("state").asText()))
          .isEqualTo("PROPOSING");
    }

    private ObjectNode question(boolean confirmation) {
      ObjectNode question = documents.path("clarification").deepCopy();
      question.put("sessionId", id).put("scopeHash", scope())
          .put("expiresAt", NOW.plusSeconds(20).toString());
      if (confirmation) {
        question.put("reason", "SIDE_EFFECT_CONFIRMATION").put("promptCode", "CONFIRM_ACTION");
      }
      contracts.validate("v3/clarification", question);
      return question;
    }

    private ObjectNode answer() {
      return contracts.object().put("schemaVersion", 3).put("clarificationId", "question-1")
          .put("choiceId", "choice-1").put("generation", 1);
    }

    private ObjectNode proposal() {
      ObjectNode proposal = documents.path("action").deepCopy();
      JsonNode progress = repository.transaction(() -> repository.lock(id).progress().deepCopy());
      proposal.put("sessionId", id).put("scopeHash", scope()).put("generation", 1)
          .put("evidenceHash", progress.path("evidenceHash").asText())
          .put("authoredInputsHash", context().path("authoredInputsHash").asText())
          .put("contextRevision", 1);
      proposal.set("executionEvidence", progress.path("executionEvidence"));
      proposal.remove("observationHash");
      proposal.putObject("operation").put("kind", "click");
      contracts.validate("action", proposal);
      return proposal;
    }

    private Map<String, List<String>> rows() {
      Map<String, List<String>> out = new LinkedHashMap<>();
      var tables = new java.util.TreeSet<String>(List.of(TABLES));
      tables.addAll(jdbc.queryForList(
          "SELECT table_name FROM information_schema.columns WHERE table_schema='public'"
              + " AND column_name='session_id' AND table_name LIKE 'studio_guided_%'", String.class));
      for (String table : tables) {
        assertThat(table).matches("studio_guided_[a-z_]+");
        out.put(table, jdbc.queryForList(
            "SELECT to_jsonb(t)::text FROM " + table + " t WHERE session_id=? ORDER BY 1", String.class, id));
      }
      out.put("studio_guided_commands", jdbc.queryForList(
          "SELECT to_jsonb(t)::text FROM studio_guided_commands t WHERE namespace=? ORDER BY 1", String.class, id));
      return out;
    }

    private void refuses(String site, java.util.function.Supplier<JsonNode> operation) {
      assertThat(count("studio_guided_translation_state")).isZero();
      Map<String, List<String>> before = rows();
      JsonNode[] accepted = new JsonNode[1];
      Throwable failure = catchThrowable(() -> accepted[0] = operation.get());
      // A returned operation must fail the explicit 409 assertion. Once a correct refusal is
      // observed, the SPI check separately catches admission guarding after authorization.
      assertThat(failure).as("F1_EXPECT_409 %s emptyStore=%s accepted=%s returnedState=%s",
              site, installEmptyStore, failure == null,
              accepted[0] == null ? "NONE" : accepted[0].path("state").asText("ACCEPTED"))
          .isInstanceOfSatisfying(GuidedSessionException.class, refusal -> {
            assertThat(refusal.status()).isEqualTo(409);
            assertThat(refusal.getMessage()).isEqualTo("RUNTIME_INCOMPATIBLE");
            assertThat(refusal.messageCode()).isEqualTo("RUNTIME_INCOMPATIBLE");
          });
      verify(access, never()).authorizeAction(any(), any(), any());
      verify(bindings, never()).validateAction(any(), any(), any());
      verify(preparations, never()).authorize(any(), any(), any(), any());
      assertThat(rows()).as("F1_ZERO_NEW_OR_CHANGED_ROWS %s", site).isEqualTo(before);
      assertThat(count("studio_guided_actions")).isZero();
      assertThat(count("studio_guided_obligation_admissions")).isZero();
    }
  }

  /** Test-only existing SPI. It grants no planner, step-complete or translation authority. */
  private static class NegativeAccess implements GuidedSessionAccess {
    NegativeAccess() {}

    @Override
    public Snapshot resolve(Authentication principal, JsonNode start) {
      assertThat(TransactionSynchronizationManager.isActualTransactionActive()).isTrue();
      ObjectNode session = documents.path("session").deepCopy();
      return new Snapshot((ObjectNode) session.path("scope"), session.path("sourceSteps"));
    }

    @Override
    public GuidedSourceReferenceValidator.Sources sources(JsonNode session) {
      checkSourceCurrent(session);
      return new GuidedSourceReferenceValidator.Sources("1", "1", Map.of(), Map.of());
    }

    @Override
    public void checkCurrent(JsonNode session) {
      assertThat(TransactionSynchronizationManager.isActualTransactionActive()).isTrue();
      GuidedSessionException.require(session.path("scope").path("csiId").asText().equals("CSI-123"),
          "FORBIDDEN_SCOPE", 403);
    }

    @Override
    public void checkSourceCurrent(JsonNode session) {
      checkCurrent(session);
      GuidedSessionException.require(
          session.path("scope").path("sourceRevision").asText().equals("1")
              && session.path("scope").path("configurationRevision").asText().equals("1"),
          "REVISION_CONFLICT", 409);
    }

    @Override
    public void checkUser(Authentication principal, JsonNode session) {
      checkCurrent(session);
      GuidedSessionException.require(
          principal instanceof BffAuthenticationToken && principal.isAuthenticated()
              && ((BffAuthenticationToken) principal).getPrincipal().userId()
                  .equals(session.path("scope").path("principalId").asText())
              && new CsiOwnershipEvaluator().canExecute(principal, "CSI-123"),
          "FORBIDDEN_SCOPE", 403);
    }

    @Override
    public String authorizeAction(JsonNode session, JsonNode proposal, JsonNode preparation) {
      // Default production SPI calculation; no positive executable planner permit is fabricated.
      return GuidedSessionAccess.super.authorizeAction(session, proposal, preparation);
    }
  }
