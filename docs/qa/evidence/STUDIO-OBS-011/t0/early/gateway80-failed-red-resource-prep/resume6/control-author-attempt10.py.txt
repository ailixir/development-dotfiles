def strict_facts(role,facts):
    schema=FACTS[role];P.need(type(facts)is dict and set(facts)==set(schema),'AUTHOR_EXACT_ROOT_FACTS')
    for name,expected in schema.items():
        value=facts[name]
        if expected=='CLOCK':P.need(type(value)in(int,float),'CLOCK_TYPE')
        elif expected in (list,dict):P.need(type(value)is expected and bool(value),'AUTHOR_EXACT_ROOT_FACTS')
        else:P.need(type(value)is type(expected) and value==expected,'AUTHOR_EXACT_ROOT_FACTS')
    return facts

def snapshot(system,k,auth_sha,mode,expected_exit,actor_facts=()):

    c=system.custody(N,0o700);rows=[];remaining=k['limits']['maximumArchiveBytes']

            st=c.info(name)

            if stat.S_ISDIR(st.st_mode):

                P.need(stat.S_ISREG(st.st_mode) and st.st_nlink==1 and row['mode']==0o600 and st.st_uid==system.uid(),'AUTHOR_STATE_MEMBER')
                P.need(0<=st.st_size<=min(k['limits']['maximumFileBytes'],remaining),'AUTHOR_STATE_BYTE_CAP')
                raw,actual=c.read(name,0o600,min(k['limits']['maximumFileBytes'],remaining),expected=st)

        c.check()
    finally:c.close_preserving()
    result=dict(kind='SOURCE06_CURRENT_N3_STATE',unit=k['subject']['label'],authoritySha256=auth_sha,sourceGeneration=6,

    # source-produced raw-map SHA and narrow header; never root-built input maps.

def execute(system,a,k,request,source_buffers,source_raw,records,producer_sha,sink):
    role=request.get('role') if type(request)is dict else None
    P.need(role in NAMES and role not in ONCE,'AUTHOR_ONCE_NO_REWRITE');ONCE.add(role)
    sink.require_finalizer();sink.resources=system.resources;sink.stage='CONTROL_AUTHOR'
    actual_pid,actual_parent=system.pid(),system.ppid()
    sink.owner=type('ActualControlAuthor',(),{})();sink.owner.provisional_client=dict(pid=actual_pid,ppid=actual_parent,start=None,argvHash=None);sink.owner.utilities=[];sink.owner.prequalified=False
    c=None;primary=None
    try:
        name,value=assemble(system,a,k,request,source_buffers,source_raw,records,producer_sha)
        P.need(value['designatedControlAuthor']['pid']==actual_pid and value['designatedControlAuthor']['ppid']==actual_parent,'CONTROL_AUTHOR_ACTUAL_IDS')
        sink.stage='CONTROL_AUTHOR_WRITE';c=system.custody(N,0o700);P.need(c.absent(name),'AUTHOR_ONCE_NO_REWRITE')
        raw=P.packed(value)+b'\n';P.need(len(raw)<=k['limits']['maximumJsonBytes'],'BOUNDED_JSON');c.write(name,raw,0o600)
        reread,info=c.read(name,0o600,k['limits']['maximumJsonBytes']);P.need(reread==raw,'INSTALL_POST_WRITE_EQUALITY')
        result=dict(kind='SOURCE06_CURRENT_CONTROL_AUTHOR_RESULT',controlPath=str(N/name),controlSha256=P.sha(raw),role=role,
            designatedControlAuthor=value['designatedControlAuthor'],pendingActors=[row for row in value['typedPriorActors'] if row['actualStatus'] is None],
            producerActualStatus=None,outerCompletionOwed=True,sourceManifestSha256=P.source_subject(a),roleContractSha256=a['roleContractSha256'],
            namespaceState=snapshot(system,k,request['releaseSha256'],'CONTROL_AUTHOR',0,value['typedPriorActors']))
        return result
    except BaseException as error:primary=error;sink.primary(error);raise
    finally:
        if c is not None:
            try:c.close()
            except BaseException as cleanup:
                sink.cleanup(cleanup)
                if primary is None:sink.primary(cleanup);raise
