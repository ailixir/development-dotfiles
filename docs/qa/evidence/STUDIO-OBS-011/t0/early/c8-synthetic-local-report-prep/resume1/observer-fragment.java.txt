  private final class P3PublicObservation {
    private final java.util.concurrent.atomic.AtomicBoolean consumed = new java.util.concurrent.atomic.AtomicBoolean();
    private String subject;
    private String cookie;
    private boolean ambiguous;
    private volatile Thread worker;
    private volatile P3PublicExchange exchange;

    void bind(String id, String ownerCookie) {
      if (subject != null) { ambiguous = true; }
      else { subject = id; cookie = ownerCookie; }
    }

    void observe(String id, String state) {
      final long start = System.nanoTime();
      if (subject == null || !subject.equals(id) || !consumed.compareAndSet(false, true)) { return; }
      boolean interrupted = Thread.currentThread().isInterrupted();
      ObjectNode record = p3PublicRecord(state);
      ObjectNode accounting = p3NetworkNotStarted();
      try {
        JsonNode control = p3ObserverControl();
        if (interrupted) { throw new IOException("INTERRUPTED"); }
        if (ambiguous || p3OwnedGuidedCookies.size() != 1) { record.put("captureStatus", "AMBIGUOUS"); }
        else if (cookie == null || !cookie.equals(p3Cookie) || !cookie.equals(p3OwnedGuidedCookies.get(id))) {
          record.put("captureStatus", "INVALID");
        } else if ("FINALIZING".equals(state)) { record.put("captureStatus", "NOT_REPRODUCED"); }
        else if (!Set.of("FAILED", "EXPIRED", "UNKNOWN_OUTCOME", "CANCELLED").contains(state)) {
          record.put("captureStatus", "NO_DECISION");
        } else {
          if (!UUID.fromString(id).toString().equals(id) || cookie.length() > 8192
              || !cookie.matches("[\\x21-\\x7e]+") || cookie.indexOf(';') >= 0) { throw new IOException("INPUT"); }
          exchange = new P3PublicExchange(start, control, port, id, cookie, state);
          if (System.nanoTime() - start >= 12000000000L) { throw new IOException("NO_WORK_BUDGET"); }
          P3_PUBLIC_OWNED.add(exchange);
          worker = Thread.ofVirtual().name("p3-selected-public-read").unstarted(exchange);
          exchange.worker = worker;
          exchange.started = true;
          worker.start();
          p3JoinUntil(worker, start, 12000000000L);
          if (worker.isAlive()) { exchange.cancel(); }
          p3JoinUntil(worker, start, 14000000000L);
          boolean terminated = !worker.isAlive();
          accounting = exchange.accounting(terminated, Thread.currentThread().isInterrupted());
          if (terminated && exchange.accept.get() && exchange.result != null
              && exchange.closes == exchange.acquired && System.nanoTime() - start <= 14000000000L) {
            record = exchange.result.deepCopy();
          }
        }
        record.set("networkAccounting", accounting);
        // NOT_STARTED zero-request branches are valid. An unresolved started worker forbids COMMIT.
        boolean closed = !accounting.path("workerStarted").booleanValue()
            || ("TERMINATED".equals(accounting.path("workerTermination").textValue())
                && accounting.path("acquiredSockets").intValue() == accounting.path("successfulSocketCloses").intValue()
                && !accounting.path("deadlineExceeded").booleanValue()
                && !accounting.path("callerInterrupted").booleanValue());
        p3Relay(record, control, start, closed);
      } catch (InterruptedException failure) {
        interrupted = true;
        if (exchange != null && worker != null && worker.isAlive()) { exchange.cancel(); }
        // A bounded second join is attempted with the ORIGINAL network cutoff; never reset it.
        try { if (worker != null) { p3JoinUntil(worker, start, 14000000000L); } }
        catch (InterruptedException repeated) { interrupted = true; }
      } catch (Exception | AssertionError failure) {
        // No private message/body/stack, no fallback writer, no oracle replacement.
      } finally {
        if (interrupted) { Thread.currentThread().interrupt(); }
      }
    }
  }

  private static void p3JoinUntil(Thread thread, long start, long cutoff) throws InterruptedException {
    long left = cutoff - (System.nanoTime() - start);
    if (left > 0 && thread.isAlive()) {
      thread.join(left / 1000000L, (int) (left % 1000000L));
    }
  }

  private static JsonNode p3StrictJson(byte[] bytes) throws IOException {
    try {
      String text = StandardCharsets.UTF_8.newDecoder()
          .onMalformedInput(java.nio.charset.CodingErrorAction.REPORT)
          .onUnmappableCharacter(java.nio.charset.CodingErrorAction.REPORT)
          .decode(java.nio.ByteBuffer.wrap(bytes)).toString();
      JsonNode node = p3StrictParser().readTree(text);
      if (node == null || !node.isObject()) { throw new IOException("JSON_OBJECT"); }
      return node;
    } catch (java.nio.charset.CharacterCodingException failure) { throw new IOException("UTF8"); }
  }

    try { return java.util.HexFormat.of().formatHex(java.security.MessageDigest.getInstance("SHA-256").digest(raw)); }
    catch (java.security.NoSuchAlgorithmException failure) { throw new IOException("SHA"); }
  }

  private static final class P3PublicExchange implements Runnable {
    final long start;
    final JsonNode control;
    final int port;
    final String id, cookie, state;
    final java.util.concurrent.atomic.AtomicBoolean accept = new java.util.concurrent.atomic.AtomicBoolean(true);
    final java.util.concurrent.atomic.AtomicBoolean cancellation = new java.util.concurrent.atomic.AtomicBoolean();
    volatile Thread worker;
    volatile boolean started, completed;
    void cancel() {
      accept.set(false);
      if (cancellation.compareAndSet(false,true) && worker != null) { worker.interrupt(); }
    }
    volatile int acquired, closes, connects, requests;
    volatile java.net.Socket raw;
    volatile javax.net.ssl.SSLSocket tls;
    volatile ObjectNode result;
    P3PublicExchange(long start, JsonNode control, int port, String id, String cookie, String state) {
      this.start = start; this.control = control; this.port = port; this.id = id; this.cookie = cookie; this.state = state;
    }
    int remaining() throws IOException {
      long left = 12000000000L - (System.nanoTime() - start);
      if (left <= 0 || !accept.get() || Thread.currentThread().isInterrupted()) { throw new IOException("WORK_DEADLINE"); }
      return (int) Math.max(1, Math.min(12000, (left + 999999L) / 1000000L));
    }
    @Override public void run() {
      ObjectNode candidate = null;
      try {
        remaining();
        raw = new java.net.Socket(java.net.Proxy.NO_PROXY); acquired++;
        raw.setSoTimeout(remaining()); connects++;
        raw.connect(new java.net.InetSocketAddress(java.net.InetAddress.getByAddress(new byte[] {127,0,0,1}), port), remaining());
        remaining();
        tls = (javax.net.ssl.SSLSocket) TLS.getSocketFactory().createSocket(raw, "localhost", port, true); acquired++;
        javax.net.ssl.SSLParameters parameters = tls.getSSLParameters();
        parameters.setEndpointIdentificationAlgorithm("HTTPS"); parameters.setApplicationProtocols(new String[] {"http/1.1"});
        tls.setSSLParameters(parameters); tls.setSoTimeout(remaining()); tls.startHandshake(); remaining();
        String protocol = tls.getApplicationProtocol();
        if (!protocol.isEmpty() && !"http/1.1".equals(protocol)) { throw new IOException("ALPN"); }
        byte[] request = ("GET /api/v1/studio/guided-sessions/" + id + "/result HTTP/1.1\r\nHost: localhost:" + port
            + "\r\nCookie: put_session=" + cookie + "\r\nAccept: application/json\r\nAccept-Encoding: identity\r\nConnection: close\r\n\r\n").getBytes(StandardCharsets.US_ASCII);
        remaining(); requests++; tls.getOutputStream().write(request); tls.getOutputStream().flush(); completed = true; remaining();
        P3HttpEntity entity = new P3HttpEntity(tls, this);
        JsonNode body = p3StrictJson(entity.read()); remaining();
        candidate = p3PublicRecord(state);
        if (!body.path("schemaVersion").isIntegralNumber() || !body.path("schemaVersion").canConvertToInt()
            || body.path("schemaVersion").intValue() != 3 || !id.equals(body.path("sessionId").textValue())
            || (entity.status == 200 && (!id.equals(body.path("executionId").textValue()) || !"guided".equals(body.path("mode").textValue()) || body.has("state")))
            || (entity.status == 202 && (body.has("outcome") || body.has("executionId") || body.has("mode")))) {
          candidate.put("captureStatus", "INVALID");
        } else {
          String outcome = p3PublicEnum(body.path(entity.status == 200 ? "outcome" : "state"), P3_PUBLIC_STATES);
          candidate.put("publicOutcomeOrState", outcome);
          if (!state.equals(outcome)) { candidate.put("captureStatus", "MISMATCH"); }
          else if (entity.status == 202) { candidate.put("captureStatus", "INVALID"); candidate.put("publicOutcomeOrState", "UNKNOWN"); }
          else {
            candidate.put("captureStatus", "CAPTURED");
            if ("FAILED".equals(state)) {
              String code = p3PublicEnum(body.path("errorCode"), P3_PUBLIC_CODES);
              candidate.put("errorCode", code); candidate.put("codeQualified", !"UNKNOWN".equals(code));
            }
          }
        }
      } catch (Exception | AssertionError failure) {
        candidate = null;
      } finally {
        // The worker owns closes. The assertion caller never closes a possibly blocked TLS socket.
        if (tls != null) { try { tls.close(); closes++; } catch (Exception failure) { accept.set(false); } }
        if (raw != null) { try { raw.close(); closes++; } catch (Exception failure) { accept.set(false); } }
        if (candidate != null && accept.get() && System.nanoTime() - start <= 12000000000L) { result = candidate; }
      }
    }
    ObjectNode accounting(boolean terminated, boolean interrupted) {
      ObjectNode a = p3NetworkNotStarted();
      a.put("workerStarted", started); a.put("workerTermination", terminated ? "TERMINATED" : "UNRESOLVED");
      a.put("acquiredSockets", acquired); a.put("successfulSocketCloses", closes); a.put("connectCalls", connects);
      a.put("requestStarts", requests); a.put("requestCompleted", completed); a.put("callerInterrupted", interrupted);
      long elapsed = System.nanoTime() - start;
      a.put("deadlineExceeded", elapsed > 12000000000L);
      if (terminated && elapsed <= 14000000000L) { a.put("networkCompletionElapsedNs", elapsed); }
      return a;
    }
  }

  private static final class P3HttpEntity {
    final javax.net.ssl.SSLSocket socket;
    final P3PublicExchange owner;
    final java.io.InputStream input;
    final byte[] body = new byte[262144];
    final byte[] transfer = new byte[8192];
    int count, status;
    P3HttpEntity(javax.net.ssl.SSLSocket socket, P3PublicExchange owner) throws IOException {
      this.socket = socket; this.owner = owner; input = socket.getInputStream();
    }
    int one() throws IOException { socket.setSoTimeout(owner.remaining()); int b = input.read(); owner.remaining(); return b; }
    String line(int limit, int[] total) throws IOException {
      byte[] bytes = new byte[limit]; int used = 0;
      while (true) {
        int b = one();
        if (b < 0 || ++total[0] > total[1]) { throw new IOException("LINE_EOF_CAP"); }
        if (b == 13) {
          int lf = one();
          if (lf != 10 || ++total[0] > total[1]) { throw new IOException("CRLF"); }
          return new String(bytes, 0, used, StandardCharsets.US_ASCII);
        }
        if (b == 10 || b < 32 || b > 126 || used == limit) { throw new IOException("LINE_GRAMMAR_CAP"); }
        bytes[used++] = (byte) b;
      }
    }
    void amount(int length) throws IOException {
      if (length < 0 || length > body.length - count) { throw new IOException("ENTITY_CAP"); }
      int left = length;
      while (left > 0) {
        socket.setSoTimeout(owner.remaining());
        int n = input.read(transfer, 0, Math.min(left, transfer.length)); owner.remaining();
        if (n <= 0 || n > left || n > body.length - count) { throw new IOException("ENTITY_SHORT"); }
        System.arraycopy(transfer, 0, body, count, n); count += n; left -= n;
      }
    }
    byte[] read() throws IOException {
      int[] total = {0,16384};
      String first = line(16384, total);
      if (!first.matches("HTTP/1\\.[01] (200|202)( [\\x20-\\x7e]*)?")) { throw new IOException("STATUS"); }
      status = Integer.parseInt(first.substring(9,12));
      java.util.Map<String,String> headers = new java.util.HashMap<>();
      while (true) {
        String text = line(16384, total); if (text.isEmpty()) { break; }
        int colon = text.indexOf(':');
        if (colon <= 0 || !text.substring(0, colon).matches("[!#$%&'*+.^_`|~0-9A-Za-z-]+")) { throw new IOException("HEADER"); }
        String key = text.substring(0,colon).toLowerCase(java.util.Locale.ROOT);
        if (headers.putIfAbsent(key,text.substring(colon+1).trim()) != null) { throw new IOException("DUPLICATE_HEADER"); }
      }
      String length = headers.get("content-length"), encoding = headers.get("transfer-encoding");
      if (!"identity".equalsIgnoreCase(headers.getOrDefault("content-encoding","identity"))
          || (length != null && encoding != null)) { throw new IOException("AMBIGUOUS_ENCODING"); }
      if (encoding != null) {
        if (!"chunked".equalsIgnoreCase(encoding)) { throw new IOException("TRANSFER_ENCODING"); }
        while (true) {
          String chunk = line(128, new int[] {0,130});
          if (!chunk.matches("[0-9a-fA-F]{1,8}")) { throw new IOException("CHUNK_SIZE"); }
          long n = Long.parseLong(chunk,16);
          if (n > body.length - count) { throw new IOException("CHUNK_CAP"); }
          if (n == 0) {
            int[] trailers = {0,8192};
            while (true) {
              String trailer = line(8192,trailers); if (trailer.isEmpty()) { break; }
              // No trailer values are interpreted; ambiguity/security relevant trailers are refused.
              if (!trailer.matches("[!#$%&'*+.^_`|~0-9A-Za-z-]+:[\\x20-\\x7e]*")
                  || trailer.regionMatches(true,0,"Content-Length:",0,15)
                  || trailer.regionMatches(true,0,"Transfer-Encoding:",0,18)) { throw new IOException("TRAILER"); }
            }
            break;
          }
          amount((int) n); if (one() != 13 || one() != 10) { throw new IOException("CHUNK_END"); }
        }
      } else if (length != null) {
        if (!length.matches("0|[1-9][0-9]{0,8}")) { throw new IOException("LENGTH"); }
        long n = Long.parseLong(length); if (n > body.length) { throw new IOException("LENGTH_CAP"); }
        amount((int) n);
      } else {
        while (true) {
          socket.setSoTimeout(owner.remaining());
          int n = input.read(transfer,0,Math.min(transfer.length,body.length-count+1)); owner.remaining();
          if (n < 0) { break; }
          if (n == 0 || n > body.length - count) { throw new IOException("CLOSE_ENTITY_CAP"); }
          System.arraycopy(transfer,0,body,count,n); count += n;
        }
      }
      owner.remaining(); return java.util.Arrays.copyOf(body,count);
    }
  }

  private static void p3Relay(ObjectNode record, JsonNode control, long start, boolean networkClosed) throws IOException {
    if (System.nanoTime() - start >= 15000000000L || Thread.currentThread().isInterrupted()) { throw new IOException("HANDOFF_DEADLINE"); }
    String controlSha = System.getenv("CHECKS_PUBLIC_OBSERVER_CONTROL_SHA256");
    byte[] projection = (record.toString() + "\n").getBytes(StandardCharsets.UTF_8);
    String projectionSha = p3PublicSha(projection);
    ObjectNode data = JSON.createObjectNode(); data.put("protocolVersion",2); data.put("controlSha256",controlSha);
    data.put("enrollmentNonce",control.path("enrollmentNonce").textValue()); data.set("observation",record);
    data.set("networkAccounting",record.path("networkAccounting"));
    try (java.nio.channels.SocketChannel channel = java.nio.channels.SocketChannel.open(java.net.StandardProtocolFamily.UNIX);
         java.nio.channels.Selector selector = java.nio.channels.Selector.open()) {
      channel.configureBlocking(false); channel.register(selector,java.nio.channels.SelectionKey.OP_CONNECT);
      channel.connect(java.net.UnixDomainSocketAddress.of("/private/tmp/obs011-pub-observe01/r.sock"));
      while (!channel.finishConnect()) { p3Select(channel,selector,start,java.nio.channels.SelectionKey.OP_CONNECT); }
      p3WriteFrame(channel,selector,start,data.toString().getBytes(StandardCharsets.UTF_8),16384);
      JsonNode ack = p3StrictJson(p3ReadFrame(channel,selector,start,256));
      java.util.HashSet<String> names = new java.util.HashSet<>(); ack.fieldNames().forEachRemaining(names::add);
      if (!names.equals(Set.of("protocolVersion","controlSha256","projectionSha256","status"))
          || !ack.path("protocolVersion").isIntegralNumber() || !ack.path("protocolVersion").canConvertToInt() || ack.path("protocolVersion").intValue() != 2
          || !controlSha.equals(ack.path("controlSha256").textValue()) || !projectionSha.equals(ack.path("projectionSha256").textValue())
          || !"WRITTEN_CUSTODIED".equals(ack.path("status").textValue()) || !networkClosed) { throw new IOException("ACK"); }
      long elapsed = System.nanoTime() - start;
      if (elapsed >= 15000000000L || Thread.currentThread().isInterrupted()) { throw new IOException("COMMIT_DEADLINE"); }
      ObjectNode commit = JSON.createObjectNode(); commit.put("protocolVersion",2); commit.put("controlSha256",controlSha);
      commit.put("projectionSha256",projectionSha); commit.put("status","CALLER_SAW_CUSTODIED_ACK"); commit.put("callerElapsedNs",elapsed);
      p3WriteFrame(channel,selector,start,commit.toString().getBytes(StandardCharsets.UTF_8),512);
      if (System.nanoTime() - start >= 15000000000L) { throw new IOException("COMMIT_LATE"); }
    }
  }

  private static void p3Select(java.nio.channels.SocketChannel c, java.nio.channels.Selector s, long start, int op) throws IOException {
    long left = 15000000000L - (System.nanoTime() - start);
    if (left <= 0 || Thread.currentThread().isInterrupted()) { throw new IOException("RELAY_DEADLINE"); }
    c.keyFor(s).interestOps(op); s.select(Math.max(1,Math.min(100,(left+999999L)/1000000L))); s.selectedKeys().clear();
    if (System.nanoTime() - start >= 15000000000L) { throw new IOException("RELAY_OVERRUN"); }
  }
  private static void p3WriteFrame(java.nio.channels.SocketChannel c, java.nio.channels.Selector s, long start, byte[] raw, int cap) throws IOException {
    if (raw.length <= 0 || raw.length > cap) { throw new IOException("FRAME_CAP"); }
    java.nio.ByteBuffer buffer = java.nio.ByteBuffer.allocate(4+raw.length); buffer.putInt(raw.length).put(raw).flip();
    while (buffer.hasRemaining()) { if (c.write(buffer) == 0) { p3Select(c,s,start,java.nio.channels.SelectionKey.OP_WRITE); }
      if (System.nanoTime() - start >= 15000000000L) { throw new IOException("WRITE_DEADLINE"); } }
  }
  private static byte[] p3ReadFrame(java.nio.channels.SocketChannel c, java.nio.channels.Selector s, long start, int cap) throws IOException {
    java.nio.ByteBuffer size = java.nio.ByteBuffer.allocate(4); p3ReadBuffer(c,s,start,size);
    size.flip(); long n = Integer.toUnsignedLong(size.getInt());
    if (n == 0 || n > cap) { throw new IOException("FRAME_CAP"); }
    java.nio.ByteBuffer data = java.nio.ByteBuffer.allocate((int) n); p3ReadBuffer(c,s,start,data); return data.array();
  }
  private static void p3ReadBuffer(java.nio.channels.SocketChannel c, java.nio.channels.Selector s, long start, java.nio.ByteBuffer target) throws IOException {
    while (target.hasRemaining()) { int n = c.read(target); if (n < 0) { throw new IOException("RELAY_EOF"); }
      if (n == 0) { p3Select(c,s,start,java.nio.channels.SelectionKey.OP_READ); }
      if (System.nanoTime() - start >= 15000000000L) { throw new IOException("READ_DEADLINE"); } }
  }
