  /**
   * B1B-01 (ledger 20 resume; OPEN matrix rows SUCCEEDED-restart, ADMITTED, DISPATCHING): a
   * replacement coordinator resumes at the next unconsumed prefix coordinate, re-dispatches an
   * existing ADMITTED admission once, and marks an existing DISPATCHING admission UNKNOWN without a
   * resend.
   */
  @Test
  void incrementalPrefixRecoveryUsesNextUnconsumedCoordinate() throws Exception {
    // R1: restart between continuations.
    try (var parent = b1Parent(new B1Spec().edit4())) {
      var cut = B1Cut.arm(B1Cut.Kind.PLANS_ENTRY, parent.id, null, 3);
      assertThat(cut.entered.await(45, java.util.concurrent.TimeUnit.SECONDS)).isTrue();
      assertThat(cut.failure).isNull();
      JsonNode program = sessions.transaction(() -> sessions.incremental().program(parent.id));
      JsonNode expected = sessions.transaction(() -> sessions.incremental().nextAction(parent.id));
      assertThat(expected)
          .as("the held continuation is the only one; the next coordinate is unambiguous")
          .isNotNull();
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_actions WHERE session_id=?"
                      + " AND document->>'state'='SUCCEEDED'",
                  Integer.class,
                  parent.id))
          .as("a non-trivial restart point")
          .isPositive();
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_actions WHERE session_id=?"
                      + " AND document->>'state' IN ('ADMITTED','DISPATCHING')",
                  Integer.class,
                  parent.id))
          .isZero();
      String before =
          jdbc.queryForObject(
              "SELECT jsonb_agg(to_jsonb(a) ORDER BY step_id,ordinal,attempt_ordinal)::text"
                  + " FROM studio_guided_actions a WHERE session_id=?",
              String.class,
              parent.id);
      int requests = b1Count("studio_guided_prefix_requests", parent.id);
      int model = MODEL_WIRE.size();
      int legacy = LEGACY_MODEL_CALLS.get();
      b1ReplacementContinue();
      var added =
          sessions.transaction(() -> sessions.actions(parent.id)).stream()
              .filter(a -> !before.contains(a.path("admission").path("admissionId").asText()))
              .toList();
      assertThat(added).hasSize(1);
      assertThat(added.getFirst().path("coordinate")).isEqualTo(expected.path("coordinate"));
      assertThat(added.getFirst().path("state").asText()).isEqualTo("SUCCEEDED");
      assertThat(added.getFirst().path("dispatchCount").asInt()).isEqualTo(1);
      assertThat(added.getFirst().path("attemptOrdinal").asInt()).isZero();
      assertThat(
              jdbc.queryForObject(
                  "SELECT jsonb_agg(to_jsonb(a) ORDER BY step_id,ordinal,attempt_ordinal)::text"
                      + " FROM studio_guided_actions a WHERE session_id=? AND admission_id<>?",
                  String.class,
                  parent.id,
                  added.getFirst().path("admission").path("admissionId").asText()))
          .as("earlier coordinates are never re-dispatched or rewritten")
          .isEqualTo(before);
      assertThat(b1Count("studio_guided_prefix_requests", parent.id)).isEqualTo(requests + 1);
      assertThat(MODEL_WIRE).hasSize(model);
      assertThat(LEGACY_MODEL_CALLS.get()).isEqualTo(legacy);
      cut.release.countDown();
      b1AwaitRow(
          "SELECT count(*) FROM studio_guided_relearn_switches WHERE session_id=?",
          parent.id,
          Duration.ofSeconds(60));
      for (JsonNode action : program.path("actions")) {
        var same =
            sessions.transaction(() -> sessions.actions(parent.id)).stream()
                .filter(a -> a.path("coordinate").equals(action.path("coordinate")))
                .toList();
        assertThat(same).hasSize(1);
        assertThat(same.getFirst().path("state").asText()).isEqualTo("SUCCEEDED");
        assertThat(same.getFirst().path("dispatchCount").asInt()).isEqualTo(1);
        assertThat(same.getFirst().path("operationHash")).isEqualTo(action.path("operationHash"));
      }
      assertThat(
              sessions.transaction(() -> sessions.actions(parent.id)).stream()
                  .map(a -> a.path("executionEvidence").path("contextId").asText())
                  .distinct()
                  .toList())
          .hasSize(1);
      assertThat(cut.failure).isNull();
    }
    // R2: existing ADMITTED. The real admit committed; consume has not been entered.
    try (var parent = b1Parent(new B1Spec().edit4().nativeMode("b1b-recover", 1, 0))) {
      var cut = B1Cut.arm(B1Cut.Kind.ADMIT_COMMIT, parent.id, b1Coordinate(1, 0), 0);
      assertThat(cut.entered.await(45, java.util.concurrent.TimeUnit.SECONDS)).isTrue();
      assertThat(cut.failure).isNull();
      String admission = cut.admissionId;
      JsonNode admitted = b1Action(parent.id, admission);
      assertThat(admitted.path("state").asText()).isEqualTo("ADMITTED");
      assertThat(admitted.path("dispatchCount").asInt()).isZero();
      assertThat(
              b1Native(parent.id).stream()
                  .filter(e -> e.path("admissionId").asText().equals(admission))
                  .toList())
          .isEmpty();
      int effects = RECEIPTS.size();
      b1ReplacementContinue(); // recovery: advancePrefix existing ADMITTED -> dispatch.dispatch
      JsonNode recovered = b1Action(parent.id, admission);
      assertThat(recovered.path("state").asText()).isEqualTo("SUCCEEDED");
      assertThat(recovered.path("dispatchCount").asInt()).isEqualTo(1);
      assertThat(b1NativeCount(parent.id, admission, "NATIVE_COMPLETED")).isEqualTo(1);
      String consumeKey = "prefix-dispatch-" + recovered.path("idempotencyKey").asText();
      assertThat(b1ConsumeCommands(parent.id, consumeKey)).isEqualTo(1);
      cut.release.countDown(); // the original sender: same key/digest, no second runner send
      long end = System.nanoTime() + Duration.ofSeconds(20).toNanos();
      while (b1Count("studio_guided_relearn_switches", parent.id) == 0 && System.nanoTime() < end) {
        Thread.sleep(20);
      }
      assertThat(b1NativeCount(parent.id, admission, "NATIVE_COMPLETED")).isEqualTo(1);
      assertThat(b1NativeCount(parent.id, admission, "DISPATCH_ENTER")).isEqualTo(1);
      assertThat(b1ConsumeCommands(parent.id, consumeKey)).isEqualTo(1);
      assertThat(b1AttemptOneActions(parent.id)).isZero();
      assertThat(RECEIPTS.size()).as("a fill is not a submit").isEqualTo(effects);
      assertThat(cut.failure).isNull();
    }
    // R3: existing DISPATCHING. The real consume committed; the transport send has not started.
    try (var parent = b1Parent(new B1Spec().edit4().nativeMode("b1b-recover", 1, 0))) {
      var cut = B1Cut.arm(B1Cut.Kind.CONSUME_COMMIT, parent.id, b1Coordinate(1, 0), 0);
      assertThat(cut.entered.await(45, java.util.concurrent.TimeUnit.SECONDS)).isTrue();
      assertThat(cut.failure).isNull();
      String admission = cut.admissionId;
      assertThat(cut.ledgerState).isEqualTo("DISPATCHING");
      b1ReplacementContinue(); // recovery: advancePrefix existing DISPATCHING -> lostAcknowledgment
      assertThat(parent.state()).isEqualTo("UNKNOWN_OUTCOME");
      p3AwaitTeardown(parent.id); // the runtime stops before the single held capability returns
      cut.release.countDown();
      Thread.sleep(2000); // bounded wait for the original sender to reach its refused transport
      JsonNode action = b1Action(parent.id, admission);
      assertThat(action.path("state").asText()).isEqualTo("UNKNOWN_OUTCOME");
      assertThat(action.path("outcomeCode").asText()).isEqualTo("ACK_LOST");
      assertThat(action.path("dispatchCount").asInt()).isEqualTo(1);
      assertThat(
              b1Native(parent.id).stream()
                  .filter(
                      e ->
                          e.path("admissionId").asText().equals(admission)
                              && e.path("phase").asText().startsWith("NATIVE_"))
                  .toList())
          .as("no native effect and no resend")
          .isEmpty();
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_commands WHERE namespace=?"
                      + " AND route='v2-consume'",
                  Integer.class,
                  parent.id))
          .isEqualTo(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_actions WHERE session_id=?"
                      + " AND (document->>'dispatchCount')::int=1",
                  Integer.class,
                  parent.id));
      assertThat(b1AttemptOneActions(parent.id)).isZero();
      assertThat(b1Count("studio_guided_model_attempts", parent.id)).isZero();
    }
  }

  /**
   * B1B-02 (ledger 20 concurrent duplicate transition): four exact byte duplicates of the runner's
   * authentic PRE_ADMISSION receipt, then the runner's own callback, commit exactly one receipt,
   * one DRIFT switch and one SOURCE_PLAN reservation, with no repeated dispatch.
   */
  @Test
  void incrementalConcurrentAuthenticatedDriftCommitsOneSwitch() throws Exception {
    try (var parent =
        b1Parent(
            new B1Spec().edit4().mixedDom("drift-auth-pre").nativeMode("drift-auth-pre", 1, 0))) {
      ObjectNode authentic = b1AwaitDriftCapture(parent.id);
      assertThat(authentic.path("receipt").path("phase").asText()).isEqualTo("PRE_ADMISSION");
      String prefixBefore = b1ActionsAt(parent.id, 0, 0) + b1ActionsAt(parent.id, 0, 1);
      long fence = authentic.path("receipt").path("ownerFence").asLong();
      String worker = token("owning-worker", "worker-1", "studio-guided-controller");
      var go = new java.util.concurrent.CountDownLatch(1);
      try (var pool = java.util.concurrent.Executors.newVirtualThreadPerTaskExecutor()) {
        var replies = new ArrayList<java.util.concurrent.Future<HttpResponse<String>>>();
        for (int i = 0; i < 4; i++) {
          replies.add(
              pool.submit(
                  () -> {
                    go.await();
                    return incrementalDriftProbe(parent.id, fence, authentic, worker, TLS);
                  }));
        }
        go.countDown();
        for (var reply : replies) {
          assertThat(reply.get(20, java.util.concurrent.TimeUnit.SECONDS).statusCode())
              .isEqualTo(202);
        }
      }
      INCREMENTAL_ACK_RELEASE.get(parent.id).countDown(); // the runner's own fifth delivery
      awaitFinalizing(parent.id);
      assertThat(b1Count("studio_guided_no_invocation_receipts", parent.id)).isEqualTo(1);
      var switched =
          jdbc.queryForMap(
              "SELECT reason,effective_step_index,next_ordinal,drift_receipt_hash"
                  + " FROM studio_guided_relearn_switches WHERE session_id=?",
              parent.id);
      assertThat(switched.get("reason")).isEqualTo("DRIFT");
      assertThat(((Number) switched.get("effective_step_index")).intValue()).isEqualTo(1);
      assertThat(((Number) switched.get("next_ordinal")).intValue()).isZero();
      assertThat(switched.get("drift_receipt_hash"))
          .isEqualTo(authentic.path("receipt").path("receiptHash").asText());
      assertThat(b1SourcePlanAttempts(parent.id)).isEqualTo(1);
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(DISTINCT request_id) FROM studio_guided_model_attempts"
                      + " WHERE session_id=? AND kind='SOURCE_PLAN'",
                  Integer.class,
                  parent.id))
          .isEqualTo(1);
      assertThat(b1AttemptsBeforeSwitch(parent.id)).isZero();
      assertThat(b1ActionsAt(parent.id, 0, 0) + b1ActionsAt(parent.id, 0, 1))
          .isEqualTo(prefixBefore);
      var fill =
          sessions.transaction(() -> sessions.actions(parent.id)).stream()
              .filter(a -> a.path("coordinate").equals(b1Coordinate(1, 0)))
              .toList();
      assertThat(fill).hasSize(1);
      assertThat(fill.getFirst().path("attemptOrdinal").asInt()).isZero();
      assertThat(fill.getFirst().path("dispatchCount").asInt()).isEqualTo(1);
      assertThat(
              b1NativeCount(
                  parent.id,
                  fill.getFirst().path("admission").path("admissionId").asText(),
                  "NATIVE_COMPLETED"))
          .isEqualTo(1);
      JsonNode result =
          publicCall("/api/v1/studio/guided-sessions/" + parent.id + "/result", null, 200, null);
      assertThat(result.path("relearnedFrom"))
          .isEqualTo(C.object().put("stepId", b1Step(1)).put("driftSwitchStepId", b1Step(1)));
    }
  }

  /**
   * B1B-03 (ledger 48; ledger 54 callback-tier joins; B3 routes 9, 10, 13, 14): every forged
   * transport, contract and identity join of an authentic drift receipt is refused with zero
   * writes; only the runner's own authenticated callback commits; late callbacks are inert or
   * refused; a cancelled session and a naturally expired owner lease refuse the authentic body.
   */
  @Test
  void incrementalDriftAuthorityRejectsEveryExactJoin() throws Exception {
    for (String phase : List.of("ADMITTED", "PRE")) {
      boolean admitted = phase.equals("ADMITTED");
      var spec = new B1Spec().edit4();
      if (admitted) {
        spec.nativeMode("drift-auth", 1, 0);
      } else {
        spec.mixedDom("drift-auth-pre").nativeMode("drift-auth-pre", 1, 0);
      }
      try (var parent = b1Parent(spec)) {
        ObjectNode authentic = b1AwaitDriftCapture(parent.id);
        long started = System.nanoTime();
        assertThat(authentic.path("receipt").path("phase").asText())
            .isEqualTo(admitted ? "ADMITTED_PRE_INVOCATION" : "PRE_ADMISSION");
        long fence = authentic.path("receipt").path("ownerFence").asLong();
        String path = "/internal/v2/studio/guided-sessions/" + parent.id + "/outcomes";
        String foreignPath = "/internal/v2/studio/guided-sessions/" + parent.base + "/outcomes";
        String worker = token("owning-worker", "worker-1", "studio-guided-controller");
        String pending = incrementalPendingSnapshot(parent.id);
        String immutable = incrementalImmutableSnapshot(parent.id);
        int model = MODEL_WIRE.size();
        SSLContext noCertificate = b1NoClientCertificate();
        var rows = new java.util.LinkedHashMap<String, B1Probe>();
        rows.put(
            "no-mtls",
            new B1Probe(authentic, worker, noCertificate, fence, path, 401, "UNAUTHENTICATED"));
        rows.put("no-auth", new B1Probe(authentic, null, TLS, fence, path, 401, "UNAUTHENTICATED"));
        rows.put(
            "alg-hs256",
            new B1Probe(
                authentic,
                b1Token(b1Claims("owning-worker", "worker-1"), "HS256", true),
                TLS,
                fence,
                path,
                401,
                "UNAUTHENTICATED"));
        rows.put(
            "alg-none-sig",
            new B1Probe(
                authentic,
                b1Token(b1Claims("owning-worker", "worker-1"), "none", false),
                TLS,
                fence,
                path,
                401,
                "UNAUTHENTICATED"));
        long now = Instant.now().getEpochSecond();
        rows.put(
            "expired",
            new B1Probe(
                authentic,
                b1Token(
                    b1Claims("owning-worker", "worker-1")
                        .put("iat", now - 400)
                        .put("exp", now - 200),
                    "RS256",
                    true),
                TLS,
                fence,
                path,
                401,
                "UNAUTHENTICATED"));
        rows.put(
            "aud",
            new B1Probe(
                authentic,
                b1Token(
                    b1Claims("owning-worker", "worker-1").put("aud", "studio-runner"),
                    "RS256",
                    true),
                TLS,
                fence,
                path,
                401,
                "UNAUTHENTICATED"));
        rows.put(
            "role-controller",
            new B1Probe(
                authentic,
                token("backend-controller", "controller-1", "studio-guided-controller"),
                TLS,
                fence,
                path,
                403,
                "FORBIDDEN_SCOPE"));
        rows.put(
            "other-worker",
            new B1Probe(
                authentic,
                token("owning-worker", "other-worker", "studio-guided-controller"),
                TLS,
                fence,
                path,
                403,
                "FORBIDDEN_SCOPE"));
        rows.put(
            "fence+1",
            new B1Probe(authentic, worker, TLS, fence + 1, path, 403, "FORBIDDEN_SCOPE"));
        rows.put(
            "cross-session",
            new B1Probe(authentic, worker, TLS, fence, foreignPath, 422, "INVALID_CONTRACT"));
        // B3 routes 9/10/14 (assigned to B1): exact original body; transport/assignment varied
        // only.
        rows.put(
            "alg-hs256-hmac",
            new B1Probe(
                authentic,
                b1HmacConfusionToken(b1Claims("owning-worker", "worker-1")),
                TLS,
                fence,
                path,
                401,
                "UNAUTHENTICATED"));
        rows.put(
            "no-auth-no-cert",
            new B1Probe(authentic, null, noCertificate, fence, path, 401, "UNAUTHENTICATED"));
        rows.put(
            "wrong-assignment-path",
            new B1Probe(authentic, worker, TLS, fence, foreignPath, 422, "INVALID_CONTRACT"));
        ObjectNode otherAssignment = authentic.deepCopy();
        ((ObjectNode) otherAssignment.path("receipt")).put("sessionId", parent.base);
        rows.put(
            "wrong-assignment-receipt",
            new B1Probe(
                b1Rehash(otherAssignment), worker, TLS, fence, path, 422, "INVALID_CONTRACT"));
        var text = JSON.getNodeFactory();
        ObjectNode schema = authentic.deepCopy();
        ((ObjectNode) schema.path("receipt")).put("schemaVersion", 2);
        rows.put(
            "schemaVersion",
            new B1Probe(schema, worker, TLS, fence, path, 409, "RUNTIME_INCOMPATIBLE"));
        ObjectNode claims = authentic.deepCopy();
        claims.set("modelClaims", C.object());
        rows.put(
            "model-claims", new B1Probe(claims, worker, TLS, fence, path, 422, "INVALID_CONTRACT"));
        rows.put(
            "native-started",
            new B1Probe(
                b1Forge(authentic, "nativeInvocationStarted", text.booleanNode(true)),
                worker,
                TLS,
                fence,
                path,
                422,
                "INVALID_CONTRACT"));
        rows.put(
            "attempt1",
            new B1Probe(
                b1Forge(authentic, "attemptOrdinal", text.numberNode(1)),
                worker,
                TLS,
                fence,
                path,
                422,
                "INVALID_CONTRACT"));
        ObjectNode hash = authentic.deepCopy();
        ((ObjectNode) hash.path("receipt")).put("receiptHash", "0".repeat(64));
        rows.put("hash", new B1Probe(hash, worker, TLS, fence, path, 409, "EVIDENCE_MISMATCH"));
        // D1 is fixed (verifyPrefixReceipt binds exactly 11 fields): each bound field is refused
        // by Java with STALE_FENCE before any write.
        var bound = new java.util.LinkedHashMap<String, JsonNode>();
        bound.put("workerId", text.textNode("foreign-worker"));
        bound.put("ownerFence", text.numberNode(fence + 1));
        bound.put("scopeHash", text.textNode("0".repeat(64)));
        bound.put("contextId", text.textNode("foreign-context"));
        bound.put(
            "contextRevision",
            text.numberNode(authentic.path("receipt").path("contextRevision").asInt() + 1));
        bound.put("sourceRevision", text.textNode("999"));
        bound.put("configurationRevision", text.textNode("999"));
        bound.put("authoredInputsHash", text.textNode("0".repeat(64)));
        bound.put("prefixProgramHash", text.textNode("0".repeat(64)));
        for (var field : bound.entrySet()) {
          rows.put(
              "bound-" + field.getKey(),
              new B1Probe(
                  b1Forge(authentic, field.getKey(), field.getValue()),
                  worker,
                  TLS,
                  fence,
                  path,
                  409,
                  "STALE_FENCE"));
        }
        // R3 pin semantics: these three are not Java-bound; only the V21 receipt guard/FK refuses
        // them. The static trace maps that refusal through GuidedInternalController.unexpected to
        // UNKNOWN_OUTCOME. The exact code is pinned; a first-run difference is a finding to rule.
        for (String field : List.of("requestDigest", "operationHash", "evidenceHash")) {
          rows.put(
              "trigger-" + field,
              new B1Probe(
                  b1Forge(authentic, field, text.textNode("0".repeat(64))),
                  worker,
                  TLS,
                  fence,
                  path,
                  409,
                  "UNKNOWN_OUTCOME"));
        }
        String readAdmission =
            jdbc.queryForObject(
                "SELECT admission_id FROM studio_guided_actions WHERE session_id=? AND step_id=?"
                    + " AND ordinal=1",
                String.class,
                parent.id,
                b1Step(0));
        String readPreparation =
            jdbc.queryForObject(
                "SELECT document->>'preparationId' FROM studio_guided_actions"
                    + " WHERE session_id=? AND admission_id=?",
                String.class,
                parent.id,
                readAdmission);
        if (admitted) {
          rows.put(
              "crossed-prep",
              new B1Probe(
                  b1Forge(authentic, "preparationId", text.textNode(readPreparation)),
                  worker,
                  TLS,
                  fence,
                  path,
                  409,
                  "EVIDENCE_MISMATCH"));
          rows.put(
              "crossed-adm",
              new B1Probe(
                  b1Forge(authentic, "admissionId", text.textNode(readAdmission)),
                  worker,
                  TLS,
                  fence,
                  path,
                  409,
                  "IDEMPOTENCY_CONFLICT"));
          ObjectNode succeeded = authentic.deepCopy();
          ((ObjectNode) succeeded.path("ledger"))
              .put("state", "SUCCEEDED")
              .put("outcomeCode", "OK");
          rows.put(
              "ledger-state",
              new B1Probe(succeeded, worker, TLS, fence, path, 409, "EVIDENCE_MISMATCH"));
          ObjectNode nullLedger = authentic.deepCopy();
          nullLedger.putNull("ledger");
          rows.put(
              "null-ledger",
              new B1Probe(nullLedger, worker, TLS, fence, path, 422, "INVALID_CONTRACT"));
        } else {
          ObjectNode withLedger = authentic.deepCopy();
          withLedger.set("ledger", C.object().put("state", "FAILED"));
          rows.put(
              "pre-with-ledger",
              new B1Probe(withLedger, worker, TLS, fence, path, 422, "INVALID_CONTRACT"));
          rows.put(
              "pre-admission-id",
              new B1Probe(
                  b1Forge(authentic, "admissionId", text.textNode(readAdmission)),
                  worker,
                  TLS,
                  fence,
                  path,
                  422,
                  "INVALID_CONTRACT"));
        }
        for (var row : rows.entrySet()) {
          B1Probe probe = row.getValue();
          var reply =
              b1RawOutcome(
                  probe.path(),
                  probe.fence(),
                  probe.body().toString(),
                  probe.bearer(),
                  probe.tls());
          assertThat(reply.statusCode()).as(row.getKey()).isEqualTo(probe.status());
          assertThat(C.parse(reply.body()).path("code").asText())
              .as(row.getKey())
              .isEqualTo(probe.code());
          assertThat(incrementalPendingSnapshot(parent.id)).as(row.getKey()).isEqualTo(pending);
          assertThat(incrementalImmutableSnapshot(parent.id)).as(row.getKey()).isEqualTo(immutable);
        }
        assertThat(b1Count("studio_guided_no_invocation_receipts", parent.id)).isZero();
        assertThat(
                jdbc.queryForObject(
                    "SELECT status FROM studio_guided_prefix_requests WHERE session_id=?"
                        + " AND request_digest=?",
                    String.class,
                    parent.id,
                    authentic.path("receipt").path("requestDigest").asText()))
            .isEqualTo(admitted ? "PREPARED" : "ISSUED");
        assertThat(parent.state()).isEqualTo("REPLAYING_PREFIX");
        assertThat(MODEL_WIRE).hasSize(model);
        assertThat(System.nanoTime() - started)
            .as("released inside the unchanged 15s transport deadline")
            .isLessThan(Duration.ofSeconds(12).toNanos());
        INCREMENTAL_ACK_RELEASE.get(parent.id).countDown(); // positive: the runner's own callback
        b1AwaitRow(
            "SELECT count(*) FROM studio_guided_relearn_switches WHERE session_id=?",
            parent.id,
            Duration.ofSeconds(20));
        assertThat(
                jdbc.queryForObject(
                    "SELECT receipt_hash FROM studio_guided_no_invocation_receipts"
                        + " WHERE session_id=?",
                    String.class,
                    parent.id))
            .isEqualTo(authentic.path("receipt").path("receiptHash").asText());
        assertThat(
                jdbc.queryForObject(
                    "SELECT reason FROM studio_guided_relearn_switches WHERE session_id=?",
                    String.class,
                    parent.id))
            .isEqualTo("DRIFT");
        var identical = b1RawOutcome(path, fence, authentic.toString(), worker, TLS);
        assertThat(identical.statusCode()).as("late identical duplicate is inert").isEqualTo(202);
        var different =
            b1RawOutcome(
                path,
                fence,
                b1Forge(authentic, "reason", text.textNode("LOCATOR_AMBIGUOUS")).toString(),
                worker,
                TLS);
        assertThat(different.statusCode()).isEqualTo(409);
        assertThat(C.parse(different.body()).path("code").asText())

  private static final List<B1Cut> B1_CUTS = new CopyOnWriteArrayList<>();
  private static final ThreadLocal<B1PlansCall> B1_PLANS_CALL = new ThreadLocal<>();
  private static final java.util.concurrent.ConcurrentMap<String, List<B1PlansCall>> B1_PLANS =
      new java.util.concurrent.ConcurrentHashMap<>();
  private static final java.util.concurrent.ConcurrentMap<String, List<String>>
      B1_OBSERVED_ORIGINS = new java.util.concurrent.ConcurrentHashMap<>();
  private static final AtomicInteger B1_REDIRECTS = new AtomicInteger();
  private static final String B1_INTERNAL = "/internal/v2/studio/guided-sessions/";
  private static final String B1_UNAPPROVED = "https://unapproved.invalid/b1c";
  private static final String B1_SUBMIT_HANDLER =
      "document.getElementById('done').hidden=false;fetch('/receipt',{method:'POST',"
          + "body:document.getElementById('destination').value});";
  private static volatile boolean b1Recording;
  private static volatile boolean b1GuardsPage;
  private static volatile Process b1StalledWorker;

  /** One internal /plans request of an armed session: its key, final status and refresh commit. */
  private static final class B1PlansCall {
    final String key;
    volatile boolean refreshCommitted;
    volatile int status;

    B1PlansCall(String key) {
      this.key = key;
    }
  }

  private record B1Probe(
      JsonNode body,
      String bearer,
      SSLContext tls,
      long fence,
      String path,
      int status,
      String code) {}

  /** T1: request-entry and outermost-commit holds; at most two cuts are armed at once. */
  private static final class B1Cut {
    enum Kind {
      PLANS_ENTRY,
      ADMIT_COMMIT,
      CONSUME_COMMIT,
      PLAN_CONTINUATION_COMMIT
    }

    final Kind kind;
    final String sessionId;
    final String requestKey;
    final JsonNode coordinate;
    final int plansOrdinal;
    final java.util.concurrent.CountDownLatch entered = new java.util.concurrent.CountDownLatch(1);
    final java.util.concurrent.CountDownLatch release = new java.util.concurrent.CountDownLatch(1);
    final AtomicInteger selections = new AtomicInteger();
    final AtomicInteger plansSeen = new AtomicInteger();
    volatile String admissionId;
    volatile String ledgerState;
    volatile long enteredAtNanos;
    volatile Throwable failure;

    private B1Cut(
        Kind kind, String sessionId, JsonNode coordinate, String requestKey, int plansOrdinal) {
      this.kind = kind;
      this.sessionId = sessionId;
      this.coordinate = coordinate;
      this.requestKey = requestKey;
      this.plansOrdinal = plansOrdinal;
    }

    static B1Cut arm(Kind kind, String sessionId, JsonNode coordinate, int plansOrdinal) {
      return arm(kind, sessionId, coordinate, null, plansOrdinal);
    }

    static B1Cut arm(
        Kind kind, String sessionId, JsonNode coordinate, String requestKey, int plansOrdinal) {
      assertThat(c5Capture).as("B1 cuts never overlap C5 sampling").isNull();
      assertThat(b1aCallbackCut).as("B1 cuts never overlap the B1a cut").isNull();
      assertThat(B1_CUTS.size()).isLessThan(2);
      var cut = new B1Cut(kind, sessionId, coordinate, requestKey, plansOrdinal);
      B1_CUTS.add(cut);
      return cut;
    }

    boolean plansRequest(jakarta.servlet.http.HttpServletRequest http) {
      return "POST".equals(http.getMethod())
          && http.getRequestURI().equals(B1_INTERNAL + sessionId + "/plans");
    }

    void onPlansRequest() {
      if (kind == Kind.PLANS_ENTRY
          && plansSeen.incrementAndGet() >= plansOrdinal
          && selections.getAndIncrement() == 0) {
        hold("PLANS_ENTRY");
      }
    }

    private boolean selects(Object result) {
      B1PlansCall call = B1_PLANS_CALL.get();
      JsonNode won = b1WonLedger(result);
      return switch (kind) {
        case ADMIT_COMMIT ->
            result instanceof ObjectNode action
                && sessionId.equals(action.path("sessionId").asText())
                && "ADMITTED".equals(action.path("state").asText())
                && coordinate.equals(action.path("coordinate"))
                && b1OnStack(B1A_SESSION_SERVICE, "admitChecked");
        case CONSUME_COMMIT ->
            won != null
                && sessionId.equals(won.path("sessionId").asText())
                && coordinate.equals(won.path("coordinate"));
        case PLAN_CONTINUATION_COMMIT ->
            result == null
                && call != null
                && requestKey.equals(call.key)
                && b1DirectPlanContinuation();
        case PLANS_ENTRY -> false;
      };
    }

    void afterOutermostCommit(Object result) {
      try {
        if (!selects(result) || selections.getAndIncrement() != 0) {
          return;
        }
        // Observe the typed ledger only; a consumed capability is never saved, cloned or resent.
        JsonNode won = b1WonLedger(result);
        JsonNode ledger = won != null ? won : result instanceof ObjectNode action ? action : null;
        if (ledger != null) {
          admissionId = ledger.path("admission").path("admissionId").asText();
          ledgerState = ledger.path("state").asText();
        }
      } catch (RuntimeException | AssertionError observerFailure) {
        failure = observerFailure;
        return;
      }
      hold(kind.name());
    }

    private void hold(String phase) {
      enteredAtNanos = System.nanoTime();
      entered.countDown();
      try {
        if (!release.await(30, java.util.concurrent.TimeUnit.SECONDS)) {
          failure = new IllegalStateException("B1 cut deadline " + phase);
        }
      } catch (InterruptedException interrupted) {
        Thread.currentThread().interrupt();
        failure = interrupted;
      }
    }
  }

  /** The ledger of an actual WonConsumption result, or null; the capability is never touched. */
  private static JsonNode b1WonLedger(Object result) {
    return result
            instanceof
            com.citi.put.checks.studio.guided.session.GuidedSessionService.WonConsumption won
        ? won.ledger()
        : null;
  }

  /** RuntimeHooks seam gate: armed only while a B1 parent records or a B1 cut is armed. */
  private static boolean b1Armed() {
    return b1Recording || !B1_CUTS.isEmpty();
  }

  /** Runs after an actual outermost commit with no transaction active; never alters the result. */
  private static void b1AfterCommit(Object result) {
    B1PlansCall call = B1_PLANS_CALL.get();
    if (call != null
        && result != null
        && b1OnStack(B1A_SESSION_SERVICE, "committedObservationRefresh")) {
      call.refreshCommitted = true;
    }
    for (B1Cut cut : B1_CUTS) {
      cut.afterOutermostCommit(result);
    }
  }

  private static boolean b1OnStack(String cls, String method) {
    return StackWalker.getInstance()
        .walk(
            frames ->
                frames.anyMatch(
                    f -> f.getClassName().startsWith(cls) && f.getMethodName().equals(method)));
  }

  /** The plan() continuation transaction only; never bootstrap, advancePrefix or a helper. */
  private static boolean b1DirectPlanContinuation() {
    return StackWalker.getInstance()
        .walk(
            stream -> {
              var frames = stream.map(f -> f.getClassName() + "#" + f.getMethodName()).toList();
              for (int at = 0; at < frames.size(); at++) {
                if (!frames.get(at).startsWith(B1A_SESSION_SERVICE)
                    || !frames.get(at).endsWith("#translationTransaction")) {
                  continue;
                }
                for (int next = at + 1; next < frames.size(); next++) {
                  if (frames.get(next).startsWith(B1A_SESSION_SERVICE)) {
                    continue; // proxy frames
                  }
                  return frames.get(next).equals(B1A_TRANSLATION + "GuidedTranslationService#plan");
                }
              }
              return false;
            });
  }

  /** Filter seam gate: POST /plans and /observations of guided sessions while B1 is armed. */
  private static boolean b1Selected(jakarta.servlet.http.HttpServletRequest http) {
    if (!b1Armed() || !"POST".equals(http.getMethod())) {

  /**
   * Records each /plans call (key, final status, whether it committed an observation refresh) and
   * each /observations callback origin, and applies an armed PLANS_ENTRY hold. The original request
   * continues unchanged; only an /observations body is read through a caching wrapper.
   */
  private static void b1Filter(
      jakarta.servlet.http.HttpServletRequest http,
      jakarta.servlet.ServletResponse response,
      B1Chain chain)
      throws IOException, jakarta.servlet.ServletException {
    String uri = http.getRequestURI();
    String id = uri.substring(B1_INTERNAL.length(), uri.lastIndexOf('/'));
    if (uri.endsWith("/observations")) {
      var wrapped = new org.springframework.web.util.ContentCachingRequestWrapper(http, 4194304);
      try {
        chain.proceed(wrapped);
      } finally {
        String origin = "UNPARSED";
        try {
          JsonNode body =
              C.parse(new String(wrapped.getContentAsByteArray(), StandardCharsets.UTF_8));
          if (body.path("observation").path("origin").isTextual()) {
            origin = body.path("observation").path("origin").asText();
          }
        } catch (RuntimeException unparsed) {
          origin = "UNPARSED";
        }
        B1_OBSERVED_ORIGINS
            .computeIfAbsent(id, ignored -> new CopyOnWriteArrayList<>())
            .add(origin);
      }
      return;
    }
    var call = new B1PlansCall(http.getHeader("Idempotency-Key"));
    B1_PLANS_CALL.set(call);
    try {
      for (B1Cut cut : B1_CUTS) {
        if (cut.plansRequest(http)) {
          cut.onPlansRequest();
        }
      }
      chain.proceed(http);
    } finally {
      B1_PLANS_CALL.remove();
      call.status = ((jakarta.servlet.http.HttpServletResponse) response).getStatus();
      B1_PLANS.computeIfAbsent(id, ignored -> new CopyOnWriteArrayList<>()).add(call);
    }
  }

    @Override
    public void close() throws Exception {
      var failures = new ArrayList<Throwable>();
      try {
        b1ResumeWorker();
      } catch (Exception | AssertionError failure) {
        failures.add(failure);
      }
      for (B1Cut cut : B1_CUTS) {
        cut.release.countDown();
      }
      var held = INCREMENTAL_ACK_RELEASE.remove(id);
      if (held != null) {

      INCREMENTAL_ACK_ENTERED.remove(id);

      forgetCachedCoreIdentity();

        refreshFixtureCredentials();
        cancel();

        failures.add(failure);
      }

        p3AwaitTeardown(id);

        }
      }

    }
  }

  /** T6: SIGSTOP of the current owned guided worker only; no clock, lease or cadence change. */
  private void b1StallWorker() throws Exception {
    Process worker = guidedWorker;
    assertThat(worker).isNotNull();
    assertThat(worker.isAlive()).isTrue();
    assertThat(
            OWNED_FIXTURE_PROCESSES.stream()
                .anyMatch(
                    p ->
                        p.process == worker
                            && p.role.equals("guided")
                            && p.generation == ownedGeneration))
        .as("only the current owned guided worker may be stalled")
        .isTrue();
    assertThat(worker.pid()).isNotEqualTo(ProcessHandle.current().pid());
    b1Signal(worker, "-STOP");
    b1StalledWorker = worker;
  }

  private void b1ResumeWorker() throws Exception {
    Process worker = b1StalledWorker;
    if (worker == null) {
      return;
    }
    b1StalledWorker = null;
    if (worker.isAlive()) {
      b1Signal(worker, "-CONT");
    }
  }

  private static void b1Signal(Process worker, String signal) throws Exception {
    var kill =
        new ProcessBuilder("kill", signal, Long.toString(worker.pid()))
            .redirectErrorStream(true)
            .start();
    assertThat(kill.waitFor(5, java.util.concurrent.TimeUnit.SECONDS)).isTrue();
    assertThat(kill.exitValue()).isZero();
  }

  /** Waits for the original owner lease to lapse on both the DB clock and the Java clock. */
  private Instant b1AwaitNaturalLeaseExpiry(String id) throws Exception {
    Thread.sleep(1000); // a renewal already on the wire when the worker stopped may still commit
    JsonNode owner =
        sessions.transaction(() -> sessions.lock(id).document().path("owner").deepCopy());
    Instant lease = Instant.parse(owner.path("leaseExpiresAt").asText());
    long end = System.nanoTime() + Duration.ofSeconds(45).toNanos();
    while (true) {
      JsonNode now =
          sessions.transaction(() -> sessions.lock(id).document().path("owner").deepCopy());
      assertThat(now)
          .as("a stalled owner never renews, re-fences or changes worker")
          .isEqualTo(owner);
      Boolean dbExpired =
          jdbc.queryForObject(
              "SELECT (document->'owner'->>'leaseExpiresAt')::timestamptz < clock_timestamp()"
                  + " FROM studio_guided_sessions WHERE session_id=?",
              Boolean.class,
              id);
      if (Boolean.TRUE.equals(dbExpired) && Instant.now().isAfter(lease)) {
        return lease;
      }
      assertThat(System.nanoTime()).as("original lease must lapse naturally").isLessThan(end);
      Thread.sleep(100);
    }
  }

  private static String b1Token(ObjectNode claims, String alg, boolean sign) throws Exception {
    var encoder = Base64.getUrlEncoder().withoutPadding();
    String unsigned =
        encoder.encodeToString(
                C.object()
                    .put("alg", alg)
                    .put("typ", "JWT")
                    .toString()
                    .getBytes(StandardCharsets.UTF_8))
            + "."
            + encoder.encodeToString(claims.toString().getBytes(StandardCharsets.UTF_8));
    if (!sign) {
      return unsigned + "." + encoder.encodeToString(new byte[] {1});
    }
    Signature signature = Signature.getInstance("SHA256withRSA");
    signature.initSign(ISSUER.getPrivate());
    signature.update(unsigned.getBytes(StandardCharsets.UTF_8));
    return unsigned + "." + encoder.encodeToString(signature.sign());
  }

  /** RS256 to HS256 confusion: HMAC keyed with the deployed public key's encoded bytes. */
  private static String b1HmacConfusionToken(ObjectNode claims) throws Exception {
    var encoder = Base64.getUrlEncoder().withoutPadding();
    String unsigned =
        encoder.encodeToString(
                C.object()
                    .put("alg", "HS256")
                    .put("typ", "JWT")
                    .toString()
                    .getBytes(StandardCharsets.UTF_8))
            + "."
            + encoder.encodeToString(claims.toString().getBytes(StandardCharsets.UTF_8));
    var mac = javax.crypto.Mac.getInstance("HmacSHA256");
    mac.init(new javax.crypto.spec.SecretKeySpec(ISSUER.getPublic().getEncoded(), "HmacSHA256"));
    return unsigned
        + "."
        + encoder.encodeToString(mac.doFinal(unsigned.getBytes(StandardCharsets.UTF_8)));
  }

    var trusts = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
    trusts.init(trustStore);
    var context = SSLContext.getInstance("TLS");
    context.init(null, trusts.getTrustManagers(), null);
    return context;
  }

  /**
   * F3 residual risk 1: every observation callback of the mixed session names an approved origin.
   * The mixed session may send none (a refused redirect); the base session of the same case proves
   * the capture was live.

  private void b1AssertApprovedObservationOrigins(String id, String liveWitness) {
    assertThat(B1_OBSERVED_ORIGINS.getOrDefault(liveWitness, List.of()))
        .as("the observation capture was live for this case")
        .isNotEmpty();

        sessions.transaction(
            () -> sessions.lock(id).document().path("scope").path("approvedOrigins").deepCopy());

              java.util.stream.StreamSupport.stream(approved.spliterator(), false)
                  .anyMatch(allowed -> allowed.asText().equals(origin)))

          .isTrue();

  /** F3 residual risk 2: the actual terminal label, value-free, for the wording ruling. */
  private void b1RecordTerminalLabel(String method, String variant, String id) throws Exception {
    JsonNode progress = sessions.transaction(() -> sessions.lock(id).progress().deepCopy());
    ObjectNode row =
        C.object()
            .put("method", method)
            .put("variant", variant)
            .put(
                "state",
                sessions.transaction(() -> sessions.lock(id).document().path("state").asText()))
            .put("errorCode", progress.path("errorCode").asText(""));
    var outcomes = row.putArray("actions");
    for (JsonNode action : sessions.transaction(() -> sessions.actions(id))) {
      outcomes
          .addObject()
          .put(
              "stepIndex",
              INCREMENTAL_STEPS.indexOf(action.path("coordinate").path("stepId").asText()))
          .put("ordinal", action.path("coordinate").path("ordinal").asInt())
          .put("state", action.path("state").asText())
          .put("outcomeCode", action.path("outcomeCode").asText(""));
    }
    Path file = TEMP.resolve("obs015-prefix-sync-b1-terminal-labels.jsonl");
    String prior =
        Files.exists(file, java.nio.file.LinkOption.NOFOLLOW_LINKS) ? Files.readString(file) : "";
    assertThat(prior.length()).isLessThanOrEqualTo(65536);
    b1PrivateWrite(file, prior + row + "\n");
  }

  private JsonNode b1AwaitTerminal(String id, Duration limit) throws Exception {
    long end = System.nanoTime() + limit.toNanos();
    String state;
    do {
      state = sessions.transaction(() -> sessions.lock(id).document().path("state").asText());
      if (List.of("FAILED", "UNKNOWN_OUTCOME", "CANCELLED", "EXPIRED").contains(state)) {

  }

  /** Counts planning/finalize diagnostics after {@code from} with this code and this frame. */
  private static long b1Diagnostics(int from, String code, String frame) throws Exception {
    long count = 0;
    for (int n = from + 1; n <= FINALIZE_DIAGNOSTICS.get(); n++) {
      Path file = TEMP.resolve("finalize-exception-" + n + ".json");
      if (!Files.isRegularFile(file)) {
        continue;
      }
      JsonNode row = C.parse(Files.readString(file));
      if (!row.path("exceptionCode").asText().equals(code)) {
        continue;
      }
      for (JsonNode at : row.path("stackFrames")) {
        if ((at.path("class").asText() + "#" + at.path("method").asText()).equals(frame)) {
          count++;
          break;
        }
      }
    }
    return count;
  }

  private static void b1SubstituteReceipt(
      ObjectNode raw, java.util.function.Consumer<ObjectNode> change) {
    ObjectNode incremental = (ObjectNode) raw.path("incrementalEvidence");
    ObjectNode receipt = (ObjectNode) incremental.path("driftReceipt");
    change.accept(receipt);
    receipt.remove("receiptHash");
    receipt.put("receiptHash", GuidedContracts.hash(receipt));
    ((ObjectNode) incremental.path("switch"))
        .put("driftReceiptHash", receipt.path("receiptHash").asText());
  }

  private void b1AssertFailedResult(B1Parent parent, JsonNode result) {
    assertThat(parent.state()).isEqualTo("FAILED");
    assertThat(result.path("outcome").asText()).isEqualTo("FAILED");
    assertThat(result.path("learnedFromThisRun").asBoolean()).isFalse();
    assertThat(result.path("publication").path("status").asText()).isEqualTo("REFUSED");
    assertThat(result.path("currentReadiness").path("ready").asBoolean()).isFalse();
    assertThat(result.path("currentReadiness").path("code").asText()).isEqualTo("NOT_PUBLISHED");
    assertThat(b1Revisions(parent.test)).isEqualTo(1);
    assertThat(b1SourcePlanAttempts(parent.id)).isEqualTo(1);
    C.validate("v3/result", result);
    var contradictions = new java.util.LinkedHashMap<String, ObjectNode>();
    ObjectNode missing = result.deepCopy();
    missing.remove("relearnedFrom");
    contradictions.put("missing", missing);
    ObjectNode learned = result.deepCopy();
    learned.put("learnedFromThisRun", true);
    contradictions.put("learned-while-refused", learned);
    ObjectNode ready = result.deepCopy();
    ((ObjectNode) ready.path("currentReadiness")).put("ready", true);
    contradictions.put("ready-unpublished", ready);
    ObjectNode extra = result.deepCopy();
    ((ObjectNode) extra.path("relearnedFrom")).put("receiptHash", "0".repeat(64));
    contradictions.put("extra", extra);
    ObjectNode blank = result.deepCopy();
    ((ObjectNode) blank.path("relearnedFrom")).remove("stepId");
    contradictions.put("no-step", blank);
    for (var bad : contradictions.entrySet()) {
      assertThatThrownBy(() -> C.validate("v3/result", bad.getValue()))
          .as(bad.getKey())
          .isInstanceOf(GuidedSessionException.class)
          .hasMessage("INVALID_CONTRACT");
    }
  }
