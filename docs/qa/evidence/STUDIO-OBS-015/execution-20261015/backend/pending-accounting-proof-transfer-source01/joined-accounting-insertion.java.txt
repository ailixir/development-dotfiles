      // This is the planner-owned append window. Protect only the synchronous accounting
      // checkpoint from lease renewal; no latch wait, negative transaction or ACK runs under this lock.
      sessions.transaction(() -> {
        var locked = sessions.lock(id);
        assertThat(plans.state(id).path("continuation").asText()).isEqualTo("MATERIALIZE");
        assertThat(new com.citi.put.checks.studio.guided.session.GuidedIncrementalStore(jdbc, C).pending(id)).isFalse();
        var materialized = jdbc.queryForMap(
            "SELECT live,context_revision FROM studio_guided_materializations WHERE session_id=?", id);
        assertThat(materialized.get("live")).isEqualTo(Boolean.TRUE);
        assertThat(((Number) materialized.get("context_revision")).intValue()).isEqualTo(2);
        assertThat(jdbc.queryForObject("SELECT count(*) FROM studio_guided_actions WHERE session_id=?"
            + " AND document->>'state' IN ('ADMITTED','DISPATCHING','UNKNOWN_OUTCOME')", Integer.class, id)).isZero();
        assertThat(jdbc.queryForObject("SELECT count(*) FROM studio_guided_preparations WHERE session_id=?"
            + " AND admission_id IS NULL AND expires_at>clock_timestamp()", Integer.class, id)).isZero();
        String pendingDelivery = materialization.pendingDeliveryId(id);
        assertThat(pendingDelivery).isNotNull();
        JsonNode issued = service.runnerContext(controller, id, fence);
        assertThat(issued.path("contextRevision").asInt()).isEqualTo(2);
        assertThat(jdbc.queryForObject("SELECT context_hash FROM studio_guided_materialization_deliveries"
            + " WHERE session_id=? AND delivery_id=?::uuid AND context_revision=2 AND status='PENDING'",
            String.class, id, pendingDelivery)).isEqualTo(GuidedContracts.hash(issued));
        var headers = jdbc.queryForList("SELECT context_id,authored_inputs_hash,previous_inputs_hash,input_count"
            + " FROM studio_guided_materialization_versions WHERE session_id=? ORDER BY context_revision", id);
        assertThat(headers).hasSize(2);
        assertThat(headers.get(1).get("context_id")).isEqualTo(headers.get(0).get("context_id"));
        assertThat(headers.get(1).get("previous_inputs_hash")).isEqualTo(headers.get(0).get("authored_inputs_hash"));
        int originalInputCount = ((Number) headers.get(0).get("input_count")).intValue();
        assertThat(((Number) headers.get(1).get("input_count")).intValue()).isGreaterThan(originalInputCount);
        JsonNode originalInputs = C.parse(jdbc.queryForObject("SELECT authored_inputs::text"
            + " FROM studio_guided_materialization_snapshots WHERE session_id=? AND context_revision=1", String.class, id));
        for (int inputIndex = 0; inputIndex < originalInputCount; inputIndex++)
          assertThat(issued.path("authoredInputs").path("inputs").get(inputIndex)
              .equals(originalInputs.path("inputs").get(inputIndex)))
              .as("appended context retains every immutable original input").isTrue();
        java.util.function.Supplier<Long> nativeEventCount = () -> {
          try {
            return Files.readAllLines(TEMP.resolve("obs015-native-events.jsonl")).stream()
                .filter(line -> !line.isBlank()).map(C::parse)
                .filter(event -> event.path("sessionId").asText().equals(id)).count();
          } catch (IOException failure) { throw new java.io.UncheckedIOException(failure); }
        };
        java.util.function.Supplier<JsonNode> accountingSnapshot = () -> {
          var actual = sessions.lock(id);
          ObjectNode snapshot = C.object();
          snapshot.set("document", actual.document().deepCopy());
          snapshot.set("progress", actual.progress().deepCopy());
          snapshot.set("translation", plans.state(id).deepCopy());
          snapshot.set("context", service.runnerContext(controller, id, fence));
          snapshot.put("pendingDeliveryId", materialization.pendingDeliveryId(id));
          snapshot.put("fullOwnedTestState", b1ScopedSnapshot(UUID.fromString(locked.document().path("testId").asText())));
          snapshot.set("assignments", C.parse(jdbc.queryForObject(
              "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                  + " FROM studio_guided_assignments t WHERE session_id=?", String.class, id)));
          snapshot.put("providerBodies", MODEL_WIRE.size()).put("providerKinds", MODEL_KINDS.size())
              .put("providerFailures", PROVIDER_FAILURES.get()).put("modelResponses", MODEL_RESPONSE_FACTS.size())
              .put("committedModelReservations", COMMITTED_MODEL_BUDGET_WITNESSES.size())
              .put("legacyCalls", LEGACY_MODEL_CALLS.get()).put("effects", RECEIPTS.size())
              .put("navigations", NAVIGATIONS.get()).put("pageReadValues", VALUES.size())
              .put("ordinaryReads", INCREMENTAL_ORDINARY_READS.get()).put("replayInvocations", REPLAY_INVOCATIONS.size())
              .put("nativeEvents", nativeEventCount.get());
          return snapshot;
        };
        JsonNode beforeAccounting = accountingSnapshot.get();
        for (int accountingCall = 0; accountingCall < 2; accountingCall++) {
          assertThat(service.reconcileLateModelUsage(controller, id, fence)).isTrue();
          assertThat(accountingSnapshot.get().equals(beforeAccounting))
              .as("pending normalized delivery permits accounting without any captured state/provider/native change")
              .isTrue();
        }
        return null;
      });
