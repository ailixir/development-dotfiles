  private static String submitControl() {
    String effect =
        "document.querySelector('[aria-label=Confirmation]').hidden="
            + "false;fetch('/translation-submit',{method:'POST',body:JSON.stringify({"
            + "destination:document.querySelectorAll('textarea')[0].value,"
            + "backup:document.querySelectorAll('textarea')[1].value})})";
    String control = formSubmit;
    if (control == null) {
      return "<button onclick=\""
          + effect
          + "\">"
          + (guardTarget == null ? "Submit" : guardTarget)
          + "</button>";
    }
    // AR-SEC001-FM I-1/I-2: the OBS-007 login shape (a form with no accessible name, so the
    // collector exports no form node and every control has formId null), and a submit button
    // associated through the form= attribute from outside its form.
    if (control.equals("form-attribute")) {
      return "<form id='order' onsubmit=\"event.preventDefault();"
          + effect
          + "\"></form><button form='order'>Submit</button>";
    }
    if (control.startsWith("unlabelled-")) {
      return "<form onsubmit=\"event.preventDefault();"
          + effect
          + "\">"
          + (control.equals("unlabelled-input")
              ? "<input type='submit' value='Submit'>"
              : "<button>Submit</button>")
          + "</form>";
    }
    return "<form aria-label='Order form' onsubmit=\"event.preventDefault();"
        + effect
        + "\">"
        + (control.equals("input")
            ? "<input type='submit' value='Submit'>"
            : "<button>Submit</button>")
        + "</form>";
  }

      assertThat(

          .isEqualTo("POLICY_DENIED");
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_model_attempts WHERE session_id=? AND"
                      + " status='REJECTED'",
                  Integer.class,
                  id))
          .isPositive();
      for (String table : List.of("studio_guided_actions", "studio_guided_obligation_admissions")) {
        assertThat(
                jdbc.queryForObject(
                    "SELECT count(*) FROM "
                        + table

                            ? "step_id=?"
                            : "proposal->'coordinate'->>'stepId'=?"),
                    Integer.class,
                    id,
                    THREE))
            .as("no third-step row in %s", table)
            .isZero();
      }
      assertThat(
              jdbc.queryForObject(
                  "SELECT count(*) FROM studio_guided_preparations WHERE session_id=?",
                  Integer.class,
                  id))
          .as("the refused guard call performed no preparation")
          .isEqualTo(preparationsBeforeRefusal);
      assertThat(SUBMITS.get()).as("no native click effect").isZero();
      assertThat(audit.list)
          .anyMatch(
              event ->
                  event.getFormattedMessage().equals("GUIDED_ACTION_REFUSED")

      governedStance("ALLOW");
    }
  }

  private void clarifyFirstAsksOnceThenContinues(String control) throws Exception {
    formSubmit = control;
    askField = false;
    try {
      governedStance("CLARIFY_FIRST");
      String id = startSession();
      invoke(id, "plans");
      int confirmations = 0;
      for (int n = 0; n < 24; n++) {
        String state = sessionState(id);
        if (state.equals("FINALIZING") || TERMINAL_STATES.contains(state)) {
          break;
        }
        if (state.equals("AWAITING_CLARIFICATION")) {
          confirmations++;
          assertThat(SUBMITS.get()).as("asked before the first side-effecting dispatch").isZero();
          assertThat(submitStepAdmissions(id)).isZero();
          JsonNode question = publicGet("/" + id + "/clarification");
          assertThat(question.path("reason").asText()).isEqualTo("SIDE_EFFECT_CONFIRMATION");
          assertThat(question.path("promptCode").asText()).isEqualTo("CONFIRM_ACTION");
          assertThat(question.path("choices")).hasSize(1);
          JsonNode choice = question.path("choices").get(0);
          assertThat(choice.path("labelSource").asText()).isEqualTo("AUTHORED");
          assertThat(choice.path("safeLabel").asText()).isEqualTo(TEXT.get(2));
          publicPost(
              "/" + id + "/clarify",
              C.object()
                  .put("schemaVersion", 3)
                  .put("clarificationId", question.path("clarificationId").asText())
                  .put("generation", question.path("generation").asInt())
                  .put("choiceId", choice.path("choiceId").asText()));
          continue;
        }
        invoke(id, "proposals");
      }
      assertThat(sessionState(id)).isEqualTo("FINALIZING");
      assertThat(confirmations).as("one confirmation per session").isEqualTo(1);
      assertThat(SUBMITS.get()).as("the confirmed submit ran once").isEqualTo(1);
      assertThat(
              repository.transaction(
                  () ->
                      repository
                          .lock(id)
                          .progress()
                          .path("sideEffectConfirmation")
                          .path("status")
                          .asText()))
          .isEqualTo("CONFIRMED");
    } finally {
      governedStance("ALLOW");
    }
  }

  private static void mutateHostileResponse(JsonNode request, ObjectNode response) {
    if (hostileMutation == null
        || !request.path("kind").asText().equals("NEXT_ACTION")
        || !response.path("status").asText().equals("ACTION")) {
      return;
    }
    ObjectNode action = (ObjectNode) response.path("action");
    String kind = action.path("operation").path("kind").asText();
    if (hostileMutation.equals("assertion")) {
      if (kind.equals("assert")) {
        action
            .withObject("/operation")
            .put("predicate", "hasValue")
            .set(
                "expected",
                C.object()
                    .put("kind", "authoredLiteral")
                    .put(
                        "sourceInputId",
                        request.path("context").path("inputs").get(0).path("inputId").asText()));
      }
    } else {
      if (kind.equals("fill")) {
        if (hostileMutation.equals("value")) {
          action
              .withObject("/operation")
              .set(
                  "value",
                  C.object()
                      .put("kind", "authoredLiteral")
                      .put(
                          "sourceInputId",
                          request.path("context").path("inputs").get(0).path("inputId").asText()));
        } else {
          String wanted = hostileMutation.equals("operation") ? "Submit" : "Backup destination";
          for (JsonNode node : request.path("context").path("observation").path("nodes")) {
            if (node.path("name").asText().equals(wanted)) {
              var target = action.putObject("target").put("nodeId", node.path("nodeId").asText());
              target
                  .putObject("locator")
                  .put("kind", "role")
                  .put("role", node.path("role").asText())
                  .put("name", wanted)
                  .put("exact", true);
            }
          }
          if (hostileMutation.equals("operation")) {
            action.putObject("operation").put("kind", "click");
          }
        }
      }
    }
    // Every hostile replacement remains in the closed typed model language.
    C.validate("v3/guided-model-response", response);
  }

  private static void afterCommitReceipt(Runnable receipt) {
    org.springframework.transaction.support.TransactionSynchronizationManager
        .registerSynchronization(
            new org.springframework.transaction.support.TransactionSynchronization() {
              @Override
              public void afterCompletion(int status) {
                if (status == STATUS_COMMITTED) {
                  java.util.concurrent.CompletableFuture.runAsync(receipt).join();
                }
              }
            });
  }

  private HttpResponse<String> invokeRaw(String id, String route) throws Exception {

        HttpRequest.newBuilder(
                URI.create(
                    "https://localhost:"
                        + port
                        + "/internal/v2/studio/guided-sessions/"
                        + id
                        + "/"
                        + route))
            .header(
                "Authorization",
                "Bearer "
                    + Environment.token(
                        "backend-controller", "controller-1", "studio-guided-controller"))
            .header("X-Owner-Fence", Long.toString(fence))

    if (context.path("obligations").isEmpty()) {
      return response
          .put("status", "COMPLETE")
          .putNull("obligationId")
          .putNull("action")
          .putNull("question");
    }
    JsonNode obligation = context.path("obligations").get(0);
    String op = obligation.path("operationKind").asText();
    JsonNode evidence = context.path("executionEvidence");
    ObjectNode action =
        C.object()
            .put("schemaVersion", 3)
            .put("sessionId", context.path("sessionId").asText())
            .put("scopeHash", context.path("scopeHash").asText())
            .put("generation", evidence.path("generation").asInt())
            .put("evidenceHash", GuidedContracts.hash(evidence))
            .put("authoredInputsHash", context.path("authoredInputsHash").asText())
            .put("contextRevision", context.path("contextRevision").asInt());
    action.set("coordinate", context.path("coordinate"));
    action.set("executionEvidence", evidence);
    ObjectNode target =
        action
            .putObject("target")
            .put("documentId", evidence.path("documentId").asText())
            .put("generation", evidence.path("generation").asInt());
    if (op.equals("navigate")) {
      target.put("kind", "page");
      action
          .putObject("operation")
          .put("kind", op)
          .put("destinationInputId", context.path("inputs").get(0).path("inputId").asText());
    } else {
      JsonNode node = null;
      for (JsonNode candidate : context.path("observation").path("nodes")) {
        if (candidate.path("name").asText().equals(obligation.path("targetIntent").asText())) {
          node = candidate;
        }
      }
      if (node == null) {
        throw new IllegalStateException("Controlled expected node missing");
      }
      target.put("nodeId", node.path("nodeId").asText());
      ObjectNode locator =
          target
              .putObject("locator")
              .put("kind", "role")
              .put("role", node.path("role").asText())
              .put("name", node.path("name").asText())
              .put("exact", true);
      ObjectNode operation = action.putObject("operation").put("kind", op);
      if (op.equals("read")) {
        operation
            .put("bindingId", context.path("needs").get(0).path("bindingId").asText())
            .putObject("recipe")
            .put("kind", "visibleText")
            .put("classification", "permittedBusinessData")
            .put("policyRuleId", "business-read");
      }
      if (op.equals("assert")) {
        operation.put("predicate", "visible").putNull("expected");
      }
      if (op.equals("fill")) {
        if (askField) {
          askField = false;
          response.put("status", "CLARIFICATION").putNull("obligationId").putNull("action");
          var question = response.putObject("question").put("reason", "AMBIGUOUS_TARGET");
          var choices = question.putArray("choices");
          for (int n = 0; n < 2; n++) {
            JsonNode selected = node;
            if (n == 1) {
              for (JsonNode other : context.path("observation").path("nodes")) {
                if (other.path("name").asText().equals("Backup destination")) {
                  selected = other;
                }
              }
            }
            ObjectNode meaning =
                C.object()
                    .put("obligationId", obligation.path("obligationId").asText())
                    .put("nodeId", selected.path("nodeId").asText())
                    .putNull("bindingId");
            meaning
                .putObject("locator")
                .put("kind", "role")
                .put("role", selected.path("role").asText())
                .put("name", selected.path("name").asText())
                .put("exact", true);
            choices
                .addObject()
                .put("choiceId", "choice-" + n)
                .put("safeLabel", "Destination")
                .set("meaning", meaning);
          }
          return response;
        }
        operation
            .putObject("value")
            .put("kind", "binding")
            .put("bindingId", context.path("needs").get(0).path("bindingId").asText());
      }
    }
    response
        .put("status", "ACTION")
        .put("obligationId", obligation.path("obligationId").asText())
        .putNull("question")
        .set("action", action);
    // The provider selects only target/operation. All bridge authority is constructed server-side.
    action.remove(
        java.util.List.of(
            "schemaVersion",
            "sessionId",
            "scopeHash",
            "coordinate",
            "generation",
            "executionEvidence",
            "evidenceHash",
            "authoredInputsHash",
            "contextRevision"));
    target.remove(java.util.List.of("documentId", "generation"));
    if (op.equals("click") && missingGuardTarget) {
      target.put("nodeId", "node-1-999999");
    }
    if (op.equals("click")) {
      lastSubmit = response.deepCopy();
    }
    return response;
  }

  private static ObjectNode obligation(
      String step, int occurrence, String kind, String target, Integer reference, String need) {
    ObjectNode item =
        C.object()
            .put("stepId", step)
            .put("occurrence", occurrence)
            .put("operationKind", kind)
            .put("targetIntent", target)
            .putNull("predicate");
    if (reference == null) {
      item.putNull("referenceIndex");
    } else {
      item.put("referenceIndex", reference);
    }
    if (need == null) {
      item.putNull("needId");
    } else {
      item.put("needId", need);
    }
    return item;
  }

    void closeWorker() throws Exception {
      releaseClaimResponse.countDown();
      releaseRefreshCallback.countDown();
      Throwable failure = null;
      try {
        if (worker != null) {
          worker.destroy();
          if (!worker.waitFor(5, TimeUnit.SECONDS)) {
            worker.destroyForcibly();
          }
        }
      } catch (Throwable cleanup) {
        failure = cleanup;
      }
      try {
        if (workerLog != null) {
          assertThat(Files.readString(workerLog))
              .doesNotContain(CAPTURE, PASSWORD_ECHO, ORDINARY_ECHO.strip());
        }
      } catch (Throwable privacy) {
        if (failure == null) {
          failure = privacy;
        } else {
          failure.addSuppressed(privacy);
        }
      }
      rethrowCleanupFailure(failure);
    }

    private static void rethrowCleanupFailure(Throwable failure) throws Exception {
      if (failure instanceof Exception exception) {
        throw exception;
      }
      if (failure instanceof Error error) {
        throw error;
      }
      if (failure != null) {
        throw new IllegalStateException("Fixture cleanup failed", failure);
      }
    }

    static String token(String role, String subject, String audience) {
      try {
        long now = Instant.now().getEpochSecond();
        String header =
            Base64.getUrlEncoder()
                .withoutPadding()
                .encodeToString(
                    "{\"alg\":\"RS256\",\"typ\":\"JWT\"}".getBytes(StandardCharsets.UTF_8));
        String claims =
            CONTRACTS
                .object()
                .put("iss", "fixture-issuer")
                .put("sub", subject)
                .put("aud", audience)
                .put("role", role)
                .put("iat", now)
                .put("exp", now + 300)
                .toString();
        String signed =
            header
                + "."
                + Base64.getUrlEncoder()
                    .withoutPadding()
                    .encodeToString(claims.getBytes(StandardCharsets.UTF_8));
        Signature signature = Signature.getInstance("SHA256withRSA");
        signature.initSign(ISSUER.getPrivate());
        signature.update(signed.getBytes(StandardCharsets.US_ASCII));
        return signed
            + "."
            + Base64.getUrlEncoder().withoutPadding().encodeToString(signature.sign());
      } catch (Exception failure) {
        throw new IllegalStateException(failure);
      }
    }

    static void command(String... arguments) throws Exception {

    }

    static void await(java.util.function.BooleanSupplier condition, long timeout) throws Exception {
      long end = System.nanoTime() + TimeUnit.MILLISECONDS.toNanos(timeout);
      while (!condition.getAsBoolean() && System.nanoTime() < end) {
        Thread.sleep(25);
      }
      assertThat(condition.getAsBoolean()).isTrue();
    }
