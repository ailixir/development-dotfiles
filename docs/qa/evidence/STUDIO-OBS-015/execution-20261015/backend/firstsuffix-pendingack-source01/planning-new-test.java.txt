  @Test
  void genuineAppendedContextAllowsAccountingWhileNormalizedDeliveryIsPending() throws Exception {
    String id = startSession();
    Throwable primary = null;
    boolean deliveryAttempted = false;
    try {
      invoke(id, "plans");
      assertThat(repository.transaction(
          () -> repository.lock(id).progress().path("translationBootstrap").path("phase").asText()))
          .isEqualTo("READY");
      assertThat(PROVIDER_KINDS).containsExactly("SOURCE_PLAN");
      assertThat(NAVIGATION_DESTINATIONS).isEmpty();
      assertThat(repository.transaction(() -> repository.actions(id))).isEmpty();
      JsonNode initial = sessions.runnerContext(fixture.controller, id, 1);
      assertThat(initial.path("contextRevision").asInt()).isEqualTo(1);
      assertThat(repository.transaction(() -> materialization.pendingDeliveryId(id))).isNull();
      String alternate = URL + "?alternative=two";
      int start = TEXT.get(0).indexOf(alternate);
      assertThat(start).isGreaterThanOrEqualTo(0);
      assertThat(TEXT.get(0).substring(start, start + alternate.length())).isEqualTo(alternate);
      ObjectNode reference = C.object()
          .put("referenceKind", "sourceSpan").put("sourceRevision", "1").put("stepId", ONE)
          .put("startUtf16", start).put("endUtf16", start + alternate.length())
          .put("kind", "destination").put("intendedSlot", "navigate.destination");
      assertThat(java.util.stream.StreamSupport.stream(
          initial.path("authoredInputs").path("inputs").spliterator(), false)
          .anyMatch(input -> input.path("provenance").equals(reference))).isFalse();
      var references = C.object().putArray("references");
      references.add(reference);
      JsonNode appended = sessions.materialize(
          fixture.controller, id, 1, "append-accounting-" + UUID.randomUUID(), references);
      assertThat(appended.path("contextRevision").asInt()).isEqualTo(2);
      assertThat(appended.path("contextId")).isEqualTo(initial.path("contextId"));
      assertThat(appended.path("sourceRevision")).isEqualTo(initial.path("sourceRevision"));
      assertThat(appended.path("configurationRevision")).isEqualTo(initial.path("configurationRevision"));
      assertThat(appended.path("authoredInputs").path("inputs"))
          .hasSize(initial.path("authoredInputs").path("inputs").size() + 1);
      String pendingDelivery = repository.transaction(
          () -> { repository.lock(id); return materialization.pendingDeliveryId(id); });
      assertThat(pendingDelivery).isNotNull();
      assertThat(jdbc.queryForObject(
          "SELECT status FROM studio_guided_materialization_deliveries"
              + " WHERE session_id=? AND delivery_id=?::uuid AND context_revision=2",
          String.class, id, pendingDelivery)).isEqualTo("PENDING");
      assertThat(jdbc.queryForObject(
          "SELECT live FROM studio_guided_materializations WHERE session_id=? AND context_revision=2",
          Boolean.class, id)).isTrue();
      JsonNode pendingContext = sessions.runnerContext(fixture.controller, id, 1);
      assertThat(pendingContext.equals(appended)).isTrue();
      int providerCalls = PROVIDER_KINDS.size();
      int providerBodies = PROVIDER_BYTES.size();
      int navigationCalls = NAVIGATION_DESTINATIONS.size();
      java.util.function.Supplier<JsonNode> snapshot = () -> repository.transaction(() -> {
        var current = repository.lock(id);
        ObjectNode value = C.object();
        value.set("document", current.document().deepCopy());
        value.set("progress", current.progress().deepCopy());
        value.set("translationState", store.state(id).deepCopy());
        for (String table : List.of("studio_guided_materializations",
            "studio_guided_materialization_versions", "studio_guided_materialization_inputs",
            "studio_guided_materialization_deliveries", "studio_guided_assignments",
            "studio_guided_actions", "studio_guided_model_attempts")) {
          value.set(table, C.parse(jdbc.queryForObject(
              "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                  + " FROM " + table + " t WHERE session_id=?", String.class, id)));
        }
        value.set("commands", C.parse(jdbc.queryForObject(
            "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                + " FROM studio_guided_commands t WHERE namespace=?", String.class, id)));
        return value;
      });
      JsonNode beforeAccounting = snapshot.get();
      for (int accountingCall = 0; accountingCall < 2; accountingCall++) {
        assertThat(sessions.reconcileLateModelUsage(fixture.controller, id, 1)).isTrue();
        assertThat(snapshot.get().equals(beforeAccounting))
            .as("accounting leaves the real pending revision2 and all persisted boundary facts unchanged")
            .isTrue();
        assertThat(repository.transaction(() -> materialization.pendingDeliveryId(id)))
            .isEqualTo(pendingDelivery);
        assertThat(sessions.runnerContext(fixture.controller, id, 1).equals(pendingContext))
            .as("accounting cannot acknowledge or replace the real pending current context")
            .isTrue();
        assertThat(PROVIDER_KINDS).hasSize(providerCalls);
        assertThat(PROVIDER_BYTES).hasSize(providerBodies);
        assertThat(NAVIGATION_DESTINATIONS).hasSize(navigationCalls);
      }
      org.assertj.core.api.Assertions.assertThatThrownBy(() -> sessions.reserveModel(
          fixture.controller, id, 1, "pending-appended-ack-refusal", 100, 100))
          .isInstanceOf(GuidedSessionException.class).hasMessage("REVISION_CONFLICT");
      assertThat(snapshot.get().equals(beforeAccounting))
          .as("pending-ACK model refusal writes no boundary facts or model/native work").isTrue();
      assertThat(PROVIDER_KINDS).hasSize(providerCalls);
      assertThat(PROVIDER_BYTES).hasSize(providerBodies);
      assertThat(NAVIGATION_DESTINATIONS).hasSize(navigationCalls);
      // Bootstrap's earlier READY marker is not an acknowledgement of appended context revision2.
      deliveryAttempted = true;
      JsonNode acknowledgement = dispatch.pushContext(fixture.controller, id, 1);
      C.validate("v3/bridge#/definitions/ContextAcknowledgement", acknowledgement);
      assertThat(acknowledgement.path("deliveryId").asText()).isEqualTo(pendingDelivery);
      assertThat(acknowledgement.path("contextRevision").asInt()).isEqualTo(2);
      assertThat(acknowledgement.path("contextHash").asText()).isEqualTo(GuidedContracts.hash(pendingContext));
      assertThat(repository.transaction(() -> materialization.pendingDeliveryId(id))).isNull();
      JsonNode persisted = C.parse(jdbc.queryForObject(
          "SELECT ack::text FROM studio_guided_materialization_deliveries"
              + " WHERE session_id=? AND delivery_id=?::uuid AND status='ACKNOWLEDGED'",
          String.class, id, pendingDelivery));
      assertThat(persisted.equals(acknowledgement)).isTrue();
      assertThat(PROVIDER_KINDS).hasSize(providerCalls);
      assertThat(PROVIDER_BYTES).hasSize(providerBodies);
      assertThat(NAVIGATION_DESTINATIONS).hasSize(navigationCalls);
    } catch (Exception | AssertionError failure) {
      primary = failure;
      throw failure;
    } finally {
      // On an earlier assertion failure, attempt the genuine pending delivery once before the
      // existing @AfterEach cancellation/worker teardown. Never retry an uncertain delivery.
      if (!deliveryAttempted) {
        try {
          String pending = repository.transaction(
              () -> { repository.lock(id); return materialization.pendingDeliveryId(id); });
          if (pending != null) dispatch.pushContext(fixture.controller, id, 1);
        } catch (Exception | AssertionError cleanup) {
          if (primary != null) primary.addSuppressed(cleanup);
          else throw cleanup;
        }
      }
    }
  }
