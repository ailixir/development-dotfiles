    // This is the real stop-after-materialization boundary: delivery has not been acknowledged.
    String pendingDelivery =
        repository.transaction(
            () -> {
              repository.lock(id);
              return materialization.pendingDeliveryId(id);
            });
    assertThat(pendingDelivery).isNotNull();
    assertThat(sessionState(id)).isEqualTo("PREPARING");
    int providerCallsBeforeAccounting = PROVIDER_KINDS.size();
    int providerBodiesBeforeAccounting = PROVIDER_BYTES.size();
    int navigationCallsBeforeAccounting = NAVIGATION_DESTINATIONS.size();
    java.util.function.Supplier<JsonNode> pendingAccountingSnapshot =
        () ->
            repository.transaction(
                () -> {
                  var current = repository.lock(id);
                  ObjectNode snapshot = C.object();
                  snapshot.set("document", current.document().deepCopy());
                  snapshot.set("progress", current.progress().deepCopy());
                  snapshot.set("translationState", store.state(id).deepCopy());
                  snapshot.set("actions", C.parse(jdbc.queryForObject(
                      "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                          + " FROM studio_guided_actions t WHERE session_id=?", String.class, id)));
                  snapshot.set("deliveries", C.parse(jdbc.queryForObject(
                      "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                          + " FROM studio_guided_materialization_deliveries t WHERE session_id=?", String.class, id)));
                  snapshot.set("commands", C.parse(jdbc.queryForObject(
                      "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                          + " FROM studio_guided_commands t WHERE namespace=?", String.class, id)));
                  snapshot.set("modelAttempts", C.parse(jdbc.queryForObject(
                      "SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY to_jsonb(t)::text),'[]'::jsonb)::text"
                          + " FROM studio_guided_model_attempts t WHERE session_id=?", String.class, id)));
                  return snapshot;
                });
    JsonNode beforePendingAccounting = pendingAccountingSnapshot.get();
    for (int accountingCall = 0; accountingCall < 2; accountingCall++) {
      assertThat(sessions.reconcileLateModelUsage(fixture.controller, id, 1)).isTrue();
      assertThat(pendingAccountingSnapshot.get().equals(beforePendingAccounting))
          .as("accounting leaves genuine pending delivery/bootstrap/counters and immutable attempts unchanged")
          .isTrue();
      assertThat(repository.transaction(() -> materialization.pendingDeliveryId(id)))
          .isEqualTo(pendingDelivery);
      assertThat(sessions.runnerContext(fixture.controller, id, 1).equals(oldContext))
          .as("accounting cannot acknowledge, replace or advance the current pending context")
          .isTrue();
      assertThat(PROVIDER_KINDS).hasSize(providerCallsBeforeAccounting);
      assertThat(PROVIDER_BYTES).hasSize(providerBodiesBeforeAccounting);
      assertThat(NAVIGATION_DESTINATIONS).hasSize(navigationCallsBeforeAccounting);
    }
    // Real accounting authority is not authoring authority: ACK still gates model reservation.
    org.assertj.core.api.Assertions.assertThatThrownBy(
            () -> sessions.reserveModel(
                fixture.controller, id, 1, "pending-ack-accounting-refusal", 100, 100))
        .isInstanceOf(GuidedSessionException.class)
        .hasMessage("REVISION_CONFLICT");
    assertThat(pendingAccountingSnapshot.get().equals(beforePendingAccounting))
        .as("the refused pending-ACK reservation writes no session/delivery/command/attempt/action bytes")
        .isTrue();
    assertThat(PROVIDER_KINDS).hasSize(providerCallsBeforeAccounting);
    assertThat(PROVIDER_BYTES).hasSize(providerBodiesBeforeAccounting);
    assertThat(NAVIGATION_DESTINATIONS).hasSize(navigationCallsBeforeAccounting);
