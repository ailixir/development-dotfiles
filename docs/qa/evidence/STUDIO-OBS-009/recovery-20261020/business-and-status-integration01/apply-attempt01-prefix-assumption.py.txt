from pathlib import Path
import json,hashlib,stat,datetime
O=Path('docs/qa/evidence/STUDIO-OBS-009/recovery-20261020/business-and-status-integration01')
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
pre=json.loads((O/'nine-target-preflight.json').read_text());before=json.loads((O/'actual-before.json').read_text())
# Joint recheck completes before the first actual write.
for r in pre['rows']:
 assert sha(r['path'])==r['beforeSha256'];assert sha(r['candidatePath'])==r['afterSha256'];assert sha(r['archivedCandidatePath'])==r['afterSha256'];assert oct(stat.S_IMODE(Path(r['path']).stat().st_mode))==r['beforeMode']
for r in before['files']+before['dist']['files']:assert sha(r['path'])==r['sha256'],r['path']
for r in json.loads((O/'protected-before.json').read_text()):assert sha(r['path'])==r['sha256'] and oct(stat.S_IMODE(Path(r['path']).stat().st_mode))==r['mode'],r['path']
for r in json.loads((O/'dependencies-before.json').read_text()):
 p=Path(r['path']);assert sha(p)==r['sha256'] and p.stat().st_size==r['bytes'] and oct(stat.S_IMODE(p.stat().st_mode))==r['mode'] and p.is_symlink()==r['symlink'];assert not p.is_symlink() or str(p.readlink())==r['linkTarget']
for r in json.loads((O/'provenance-pins.json').read_text()):assert sha(r['path'])==r['sha256']
result={'startedAtUtc':now(),'allNineAndFrozenInputsRecheckedBeforeFirstActualWrite':True,'operations':[],'exception':None}
try:
 for r in pre['rows']:
  p=Path(r['path']);assert sha(p)==r['beforeSha256'];b=Path(r['archivedCandidatePath']).read_bytes();assert hashlib.sha256(b).hexdigest()==r['afterSha256'];p.write_bytes(b)
  assert sha(p)==r['afterSha256'];assert oct(stat.S_IMODE(p.stat().st_mode))==r['beforeMode']
  result['operations'].append({'path':r['path'],'beforeSha256':r['beforeSha256'],'afterSha256':sha(p),'mode':r['beforeMode'],'bytesWritten':len(b),'exit':0,'atUtc':now()})
  (O/'application.json').write_text(json.dumps(result,indent=2)+'\n')
 result['exit']=0
except Exception as e:
 result['exception']={'type':type(e).__name__,'detail':str(e)};result['exit']=1;raise
finally:
 result['endedAtUtc']=now();(O/'application.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'exit':result['exit'],'actualTargetsWritten':len(result['operations'])}))
